/**
 * 🆕 承認処理のコアロジック(HTMLを組み立てない、シンプルな結果オブジェクトを返す)。
 * doPost経由(フォールバック)・google.script.run経由(通常経路)の両方から呼ばれる共通処理。
 * 戻り値: { success, code, message, approverDisplayName, userEmail }
 */
function approveByToken_(token, userEmail) {
  const props = PropertiesService.getScriptProperties();
  const key = "appr_" + token;
  const dataStr = props.getProperty(key);

  if (!dataStr) return { success: false, code: "invalid", message: "このトークンは存在しません。" };
  const data = JSON.parse(dataStr);
  if (data.status === "approved") return { success: false, code: "already", message: "この依頼はすでに承認されています。" };

  if (userEmail) {
    const approvers = getApprovers_();
    if (!approvers.includes(userEmail)) {
      return { success: false, code: "forbidden", message: `${userEmail} には承認権限がありません。` };
    }
  }

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("CO31_Responses");
    const rows = tokenRowIds_(data);
    if (rows.length === 0) throw new Error("トークンに承認対象の行が含まれていません。");

    const approverDisplayName = getApproverNameByEmail_(userEmail) || userEmail || "運送部門担当者";

    // 🆕 PDF生成・メール送信は重く実行時間上限に抵触するため、ここではマークだけ付ける。
    //    実際の処理は processApprovalQueue_ がバックグラウンドで行う(独立した実行枠を使える)。
    let marked = 0;
    let firstIid = "";
    rows.forEach(function(r) {
      const st = String(sheet.getRange(r, 30).getValue() || "").trim();
      if (st === "CO30作成待ち" || st === "送信完了" || st === "取消") return;
      // 承認者名は行番号ではなく内部依頼ID(不変)に紐づける
      const iid = assignInternalRequestId_(sheet, r);
      if (!firstIid) firstIid = iid;
      props.setProperty("apprby_id_" + iid, approverDisplayName);
      sheet.getRange(r, 30).setValue("承認");
      marked++;
    });

    if (firstIid) {
      props.deleteProperty("apprfail_" + firstIid);
      props.deleteProperty("apprnotified_" + firstIid);
    }

    // 行への書き込みが成功してからトークンを承認済みにする(途中失敗時は再承認できるようにするため)
    data.status = "approved";
    data.approvedAt = Date.now();
    data.approvedBy = userEmail || "不明";
    props.setProperty(key, JSON.stringify(data));

    scheduleApprovalWorker_(5000);

    return {
      success: true, queued: true, markedCount: marked,
      approverDisplayName: approverDisplayName, userEmail: userEmail || "",
      partialFailure: false, failedCount: 0
    };
  } catch (err) {
    return { success: false, code: "error", message: err.message };
  }
}

/** doPost(素のフォームPOST)からのフォールバック用。HTMLページを組み立てて返す。 */

    return buildSimplePage_("❌ 承認処理でエラーが発生しました", result.message + "\n\nスプレッドシートから手動で承認してください。");
  }

  const html = `<!DOCTYPE html>
<html lang="ja"><head><meta charset="UTF-8"><meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>承認完了</title>
<style>body{font-family:-apple-system,sans-serif;text-align:center;padding:60px 20px;background:#f4f6f7;}
.icon{font-size:64px;margin-bottom:16px;} h1{color:#27ae60;font-size:22px;} p{color:#666;font-size:14px;line-height:1.6;}</style>
</head><body>
<div class="icon">${result.partialFailure ? "⚠️" : "✅"}</div><h1>承認が完了しました</h1>

/**

 */
function runScheduledCO30Batch_() {
  const lock = LockService.getScriptLock();

    console.log("CO30定期実行: 他の処理が実行中のため、この回はスキップしました。");
    return;
  }
  try {
    const result = exportAllCO30Combined();
    console.log(result.message);
    if (result.supersededCount > 0) {
      try {
        sendMail_({
          to: "2026trucking@gmail.com",
          subject: `【要確認】CO30自動作成で${result.supersededCount}件の差し替えがありました`,
          body: `運送・用具部門 担当者様\n\n定期実行によるCO-30の一括作成で、修正前の内容が既に現場で処理済みだった物品が${result.supersededCount}件ありました。\n該当の行は削除せず、備考欄に「⚠️内容修正により差し替え」の印を付けて保持しています。\n\nダッシュボードで内容をご確認ください。`

  } finally {

  }

/** 自動作成トリガーを設定する共通ロジック(4時間おき)。既に設定済みなら二重に作らない。 */
function enableCO30Trigger_() {
  const existing = ScriptApp.getProjectTriggers().filter(function(t) {
    return t.getHandlerFunction() === 'runScheduledCO30Batch_';
  });
  if (existing.length > 0) {
    return { success: true, enabled: true, message: "✅ 既に自動作成は設定済みです(4時間おき)。" };
  }
  ScriptApp.newTrigger('runScheduledCO30Batch_').timeBased().everyHours(4).create();
  return { success: true, enabled: true, message: "✅ CO-30の自動作成を4時間おきに設定しました。" };
}

/** 自動作成トリガーを解除する共通ロジック */
function disableCO30Trigger_() {
  const triggers = ScriptApp.getProjectTriggers().filter(function(t) {
    return t.getHandlerFunction() === 'runScheduledCO30Batch_';
  });
  if (triggers.length === 0) {
    return { success: true, enabled: false, message: "自動作成は設定されていません。" };
  }
  triggers.forEach(function(t) { ScriptApp.deleteTrigger(t); });
  return { success: true, enabled: false, message: `✅ 自動作成(${triggers.length}件)を解除しました。` };
}

}

function submitCO31Form(formData) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("CO31_Responses");
  if (!sheet) throw new Error("エラー：シート「CO31_Responses」が見つかりません。");

  const team = determineTeam_(formData.fromLocation);
  const timestamp = new Date();

  let arrivalFullText = formData.arrivalDateStr || "";
  if (formData.arrivalTimeStr) { arrivalFullText += " " + formData.arrivalTimeStr; } else { arrivalFullText += " 指定なし"; }
  if (formData.arrivalDetailTimeStr && formData.arrivalDetailTimeStr.trim() !== "") { arrivalFullText += " " + formData.arrivalDetailTimeStr.trim(); } else { arrivalFullText += " なし"; }

  // フォームの品目スロット(1〜20)を順番どおりに読み取る
  const slots = [];
  for (let n = 1; n <= 20; n++) {
    slots.push({
      qty:  formData["qty" + n]  || "",
      name: formData["item" + n] || "",
      size: formData["size" + n] || ""
    });
  }

  // 🆕 1行分(最大5品目・物品1〜5列のみ)のrowFullを組み立てる。物品6〜20列(index33-77)は新形式では使わない
  function buildRow_(chunk) {
    const rowFull = new Array(CO31_TOTAL_COLS).fill("");
    rowFull[0] = timestamp;
    rowFull[2] = formData.docDate || "";
    rowFull[3] = formData.fromLocation || "";
    rowFull[4] = formData.deptName || "";
    rowFull[5] = formData.toLocation || "";
    rowFull[6] = arrivalFullText;
    rowFull[7] = formData.cost || "";               // 費用: 空欄OK
    const firstBlocks = [8, 11, 14, 17, 20];
    firstBlocks.forEach(function(base, idx) {
      const it = chunk[idx] || { qty: "", name: "", size: "" };
      rowFull[base]     = it.qty  || "";
      rowFull[base + 1] = it.name || "";
      rowFull[base + 2] = it.size || "";
    });
    rowFull[24] = formData.totalNote || "";
    rowFull[25] = formData.approver || "";
    rowFull[29] = "承認待ち";
    rowFull[30] = team;
    rowFull[31] = formData.contactInfo || "";        // AF: 担当者名・連絡先
    rowFull[32] = formData.contactEmail || "";       // AG: 担当者メール
    return rowFull;
  }

  function chunkItemsAll_(chunk) {
    return chunk.filter(function(it) { return it && (it.name || it.qty); });
  }

  const isEdit = formData.editRow && parseInt(formData.editRow, 10) >= 2;
  // 🆕 1送信=1トークン=1通: このトークンで承認されるべき全行と、メール本文用データを集める
  let tokenRowIds = [];
  let emailFormData = null;

  if (isEdit) {
    const anchorRow = parseInt(formData.editRow, 10);
    const anchorData = sheet.getRange(anchorRow, 1, 1, CO31_TOTAL_COLS).getValues()[0];
    const groupId = String(anchorData[CO31_COL_GROUPID_IDX] || "").trim();

    if (!groupId) {
      // ---- 旧形式(親IDなし・1行=1依頼・最大20品目)の修正: 従来通り1行に上書き ----
      const rowFull = buildRow_(slots.slice(0, 5));
      for (let n = 6; n <= 20; n++) {                // AH〜: 物品6〜20(旧形式のみ)
        const base = 33 + (n - 6) * 3;
        const it = slots[n - 1];
        rowFull[base]     = it.qty  || "";
        rowFull[base + 1] = it.name || "";
        rowFull[base + 2] = it.size || "";
      }
      const internalId = anchorData[1];
      rowFull[1] = internalId ? String(internalId).trim() : "";
      rowFull[CO31_COL_DISPLAYPAGE_IDX] = anchorData[CO31_COL_DISPLAYPAGE_IDX] || "";

      const changed = detectChangedFields_(anchorData, rowFull);
      rowFull[CO31_COL_CHANGED_IDX] = JSON.stringify(changed);

      sheet.getRange(anchorRow, 1, 1, CO31_TOTAL_COLS).setValues([rowFull]);
      invalidateTokensForRow_(anchorRow);

      tokenRowIds = [anchorRow];
      emailFormData = Object.assign({}, formData);
      emailFormData.itemsAll = collectItems_(rowFull).map(function(it) { return { name: it.name, qty: it.qty }; });
      emailFormData.changedSummary = buildChangeSummary_(changed, collectItems_(rowFull));
    } else {
      // ---- 🆕 新形式(親IDで束ねた複数行)の修正 ----
      const groupRows = getGroupRowNumbers_(sheet, groupId);
      const rowCount = groupRows.length;

      const existingSlots = slots.slice(0, rowCount * 5);
      const extraFilled = slots.slice(rowCount * 5).filter(function(it) { return it.name || it.qty; });

      // 🆕 グループ全体(既存ページ+追加分)の品目が0件になる修正は禁止する。全削除は「取消」で行ってもらう。
      const totalRemaining = existingSlots.filter(function(it) { return it && (it.name || it.qty); }).length + extraFilled.length;
      if (totalRemaining === 0) {
        throw new Error("すべての品目が削除されています。依頼全体が不要な場合は、修正ではなく「取消」をご利用ください。");
      }

      // ---- フェーズ1: 品目が全て空になったページを特定する(この時点ではまだ何も書き換えない) ----
      const emptyPageRows = []; // { rowNum, oldDisplay, oldInternalId }
      const remainingChunks = []; // 残すページの新しい品目チャンク(元の並び順を維持)
      for (let k = 0; k < rowCount; k++) {
        const chunk = existingSlots.slice(k * 5, k * 5 + 5);
        const filledCount = chunk.filter(function(it) { return it && (it.name || it.qty); }).length;
        if (filledCount === 0) {
          const rn = groupRows[k];
          const oldR = sheet.getRange(rn, 1, 1, CO31_TOTAL_COLS).getValues()[0];
          emptyPageRows.push({
            rowNum: rn,
            oldDisplay: String(oldR[CO31_COL_DISPLAYPAGE_IDX] || "").trim(),
            oldInternalId: oldR[1] ? String(oldR[1]).trim() : ""
          });
        } else {
          remainingChunks.push(chunk);
        }
      }

      // ---- フェーズ2: 空になったページを削除する(既にCO30化済みなら現物側に「取消済み」の印を付けてから削除) ----
      if (emptyPageRows.length > 0) {
        emptyPageRows.forEach(function(p) {
          if (p.oldInternalId) markCO30ItemsAsCancelled_(p.oldInternalId); // 既存無ければ何もしない(安全)
          invalidateTokensForRow_(p.rowNum);
        });
        // 行番号のズレを防ぐため、必ず降順(下の行から)で削除する
        emptyPageRows.slice().sort(function(a, b) { return b.rowNum - a.rowNum; }).forEach(function(p) {
          try { sheet.deleteRow(p.rowNum); } catch (e) {}
        });
        try {
          const printFolder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
          emptyPageRows.forEach(function(p) { if (p.oldDisplay) removeOldDriveFilesForRequest_(printFolder, p.oldDisplay); });
        } catch (e) {}
      }

      // ---- フェーズ3: 削除後の実際の行番号を取り直し、残ったページに新しい内容を書き込む ----
      const groupRowsAfter = getGroupRowNumbers_(sheet, groupId);
      const changedRows = [];
      const pendingUnchangedRows = [];
      const summaryParts = [];

      emptyPageRows.forEach(function(p) {
        summaryParts.push(`【ページ削除】${p.oldDisplay ? "No." + p.oldDisplay + " " : ""}このページの品目がすべて削除されたため、ページ自体を削除しました。`);
      });

      for (let k = 0; k < groupRowsAfter.length; k++) {
        const rowNum = groupRowsAfter[k];
        const oldRow = sheet.getRange(rowNum, 1, 1, CO31_TOTAL_COLS).getValues()[0];
        const oldStatus = String(oldRow[29] || "").trim();
        const chunk = remainingChunks[k];
        const rowFull = buildRow_(chunk);
        rowFull[1] = oldRow[1] ? String(oldRow[1]).trim() : "";
        rowFull[CO31_COL_GROUPID_IDX] = groupId;
        rowFull[CO31_COL_DISPLAYPAGE_IDX] = oldRow[CO31_COL_DISPLAYPAGE_IDX] || "";

        const changed = detectChangedFields_(oldRow, rowFull);
        if (changed.fields.length === 0 && changed.itemIndices.length === 0) {
          if (oldStatus === "承認待ち") pendingUnchangedRows.push(rowNum);
          continue;
        }

        rowFull[CO31_COL_CHANGED_IDX] = JSON.stringify(changed);
        sheet.getRange(rowNum, 1, 1, CO31_TOTAL_COLS).setValues([rowFull]);
        invalidateTokensForRow_(rowNum);
        changedRows.push(rowNum);

        const dispNo = String(oldRow[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
        const pagePrefix = groupRowsAfter.length > 1 ? `【${k + 1}ページ目${dispNo ? "・No." + dispNo : ""}】\n` : "";
        summaryParts.push(pagePrefix + buildChangeSummary_(changed, collectItems_(rowFull)));
      }

      // ---- フェーズ4: 既存ページを超える追加分を、新しいページとして末尾に追記する ----
      const newRowsAdded = [];
      if (extraFilled.length > 0) {
        for (let i = 0; i < extraFilled.length; i += 5) {
          const chunk = extraFilled.slice(i, i + 5);
          const rowFull = buildRow_(chunk);
          rowFull[CO31_COL_GROUPID_IDX] = groupId;
          sheet.appendRow(rowFull);
          const newRowNum = sheet.getLastRow();
          newRowsAdded.push(newRowNum);
          const newPageIdx = groupRowsAfter.length + Math.floor(i / 5) + 1;
          const addedNames = chunk.filter(function(it) { return it.name; }).map(function(it) { return it.name; }).join("、");
          summaryParts.push(`【${newPageIdx}ページ目・新規追加】\n・追加された品目: ${addedNames || "(品名なし)"}`);
        }
      }

      pendingUnchangedRows.forEach(function(rn) { invalidateTokensForRow_(rn); });
      tokenRowIds = changedRows.concat(pendingUnchangedRows).concat(newRowsAdded).sort(function(a, b) { return a - b; });

      if (tokenRowIds.length === 0) {
        // 🆕 ページ削除だけが発生し、残りのページに変更が無く再承認の必要も無いケース。
        //    再承認は不要だが、削除の事実は運送・用具部門【と依頼部門本人】の双方に知らせる。
        const delDeptEmail = getDeptEmail_(formData.deptName || "");
        const delContactEmail = (formData.contactEmail && formData.contactEmail.includes("@")) ? formData.contactEmail : "";
        const delCc = [];
        if (delDeptEmail) delCc.push(delDeptEmail);
        if (delContactEmail && delCc.indexOf(delContactEmail) === -1) delCc.push(delContactEmail);
        try {
          const delMailOpts = {
            to: "2026trucking@gmail.com",
            subject: `【CO-31 ページ削除】${formData.deptName || "不明部門"}の依頼から一部ページが削除されました`,
            body: `運送・用具部門 御中\n${formData.deptName || "不明部門"} 御中\n\n${formData.deptName || "不明部門"}からの修正操作により、以下のページが削除されました(残りのページに変更はありません)。\n\n${summaryParts.join("\n")}\n\n※削除したページの控えPDF・印刷済みの紙をお持ちの場合は、今後の手続きには使用せず破棄してください。`
          };
          if (delCc.length > 0) delMailOpts.cc = delCc.join(",");
          sendMail_(delMailOpts);
        } catch (e) {}
        return { success: true, edit: true, pagesDeleted: true, deletedCount: emptyPageRows.length };
      }

      emailFormData = Object.assign({}, formData);
      emailFormData.itemsAll = chunkItemsAll_(existingSlots).concat(extraFilled);
      emailFormData.changedSummary = summaryParts.join("\n");
    }
  } else {
    // ---- 🆕 新規: 空スロットを詰めて5品目ごとに行を分け、共通の親依頼IDで束ねる ----
    const filled = slots.filter(function(it) { return it.name || it.qty; });
    const chunks = [];
    for (let i = 0; i < filled.length; i += 5) chunks.push(filled.slice(i, i + 5));
    if (chunks.length === 0) chunks.push([]);

    const groupId = assignNewGroupId_(sheet);
    for (let k = 0; k < chunks.length; k++) {
      const rowFull = buildRow_(chunks[k]);
      rowFull[CO31_COL_GROUPID_IDX] = groupId;
      sheet.appendRow(rowFull);
      tokenRowIds.push(sheet.getLastRow());
    }

    emailFormData = Object.assign({}, formData);
    emailFormData.itemsAll = filled;
  }

  try {
    sendApprovalRequestEmail_(tokenRowIds, emailFormData, !!isEdit);
  } catch (emailErr) {
    console.error("承認依頼メールの送信に失敗しました: " + emailErr.message);
  }
  return { success: true, edit: !!isEdit };
}

/**
 * 🆕 CO31_Responsesを親依頼ID(AA列)単位にグルーピングして返す共通ヘルパー。
 * 親IDが空の行(旧形式)は「1行=1グループ」として扱う。
 * 返り値: 出現順のグループ配列。各グループは
 *   { row(先頭行), rows[], groupId, reqNos[], statuses[], items[], dept, docDate,
 *     fromLocation, toLocation, arrival, team, timestamp }
 */
function collectCO31Groups_() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!sheet || sheet.getLastRow() < 2) return [];
  const lastRow = sheet.getLastRow();
  const values = sheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();
  const arrivalDisp = sheet.getRange(2, 7, lastRow - 1, 1).getDisplayValues();

  const groups = {};
  const order = [];
  for (let i = 0; i < values.length; i++) {
    const r = values[i];
    if (!r[4]) continue;
    const gid = String(r[CO31_COL_GROUPID_IDX] || "").trim();
    const key = gid ? "g:" + gid : "r:" + (i + 2);
    if (!groups[key]) {
      groups[key] = {
        row: i + 2, rows: [], groupId: gid, reqNos: [], statuses: [], items: [],
        dept: String(r[4]).trim(),
        docDate: r[2] ? String(r[2]) : "",
        fromLocation: r[3] ? String(r[3]).trim() : "",
        toLocation: r[5] ? String(r[5]).trim() : "",
        arrival: arrivalDisp[i][0] || "",
        team: r[30] ? String(r[30]).trim() : "",
        timestamp: (r[0] instanceof Date) ? r[0].getTime() : 0
      };
      order.push(key);
    }
    const g = groups[key];
    g.rows.push(i + 2);
    const reqNo = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    if (reqNo) g.reqNos.push(reqNo);
    g.statuses.push(r[29] ? String(r[29]).trim() : "");
    collectItems_(r).filter(function(it) { return it.name; }).forEach(function(it) {
      g.items.push({ name: it.name, qty: it.qty });
    });
  }
  return order.map(function(key) { return groups[key]; });
}

function listAllCO31Requests_() {
  const out = collectCO31Groups_().map(function(g) {
    return {
      row: g.row,                       // 先頭行(アンカー)。修正・取消はこの行番号を起点に親IDで全行を辿る
      rows: g.rows,
      groupId: g.groupId,
      reqNo: g.reqNos.join("、"),       // 各行の番号を連結(例: "清1、清2"。未承認ならまだ空)
      dept: g.dept,
      docDate: g.docDate,
      fromLocation: g.fromLocation,
      toLocation: g.toLocation,
      arrival: g.arrival,
      status: aggregateGroupStatus_(g.statuses),
      team: g.team,
      items: g.items,
      timestamp: g.timestamp
    };
  });
  out.sort(function(a, b) { return b.timestamp - a.timestamp; });
  return out;
}

function getDeptRequests(deptName) {
  return listAllCO31Requests_().filter(function(r) { return r.dept === String(deptName).trim(); });
}

function getRequestDetail(row) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!sheet) throw new Error("CO31_Responsesが見つかりません");
  const r = sheet.getRange(row, 1, 1, CO31_TOTAL_COLS).getValues()[0];
  const groupId = String(r[CO31_COL_GROUPID_IDX] || "").trim();

  const items = [];
  const reqNos = [];
  const statuses = [];
  let rows = [row];

  if (groupId) {
    // 🆕 新形式: 親IDに紐づく全行から品目を行順に連結(各行は物品1〜5列のみ)
    rows = getGroupRowNumbers_(sheet, groupId);
    rows.forEach(function(rn) {
      const rr = (rn === row) ? r : sheet.getRange(rn, 1, 1, CO31_TOTAL_COLS).getValues()[0];
      [8, 11, 14, 17, 20].forEach(function(i) {
        items.push({ qty: String(rr[i] || ""), name: String(rr[i + 1] || ""), size: String(rr[i + 2] || "") });
      });
      const no = String(rr[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
      reqNos.push(no || ""); // 🆕 rows[i]とreqNos[i]が常に1対1対応するよう、未採番でも空文字列を入れる
      statuses.push(String(rr[29] || "").trim());
    });
  } else {
    // 旧形式: 1行から最大20品目を読む
    [8, 11, 14, 17, 20].forEach(function(i) {
      items.push({ qty: String(r[i] || ""), name: String(r[i + 1] || ""), size: String(r[i + 2] || "") });
    });
    for (let i = 33; i <= 75; i += 3) {
      items.push({ qty: String(r[i] || ""), name: String(r[i + 1] || ""), size: String(r[i + 2] || "") });
    }
    const no = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    reqNos.push(no || ""); // 🆕 新形式と同様、rows[i]とreqNos[i]が常に1対1対応するようにする
    statuses.push(String(r[29] || "").trim());
  }

  // フォームは常に20スロットあるため、20件になるまで空品目でパディングする
  while (items.length < 20) items.push({ qty: "", name: "", size: "" });

  return {
    row: rows[0],                                    // アンカーは常にグループ先頭行
    rows: rows,
    groupId: groupId,
    itemLimit: groupId ? rows.length * 5 : 20,       // 🆕 修正フォームで編集を許可する品目枠数
    reqNo: reqNos.filter(Boolean).join("、"),         // 表示用連結文字列は空文字列を除いてから繋げる
    reqNos: reqNos,                                   // 🆕 rows配列と同じ順序・同じ長さ(未採番の行は空文字列)
    docDate: String(r[2] || ""),
    fromLocation: String(r[3] || ""),
    deptName: String(r[4] || ""),
    toLocation: String(r[5] || ""),
    arrival: String(sheet.getRange(row, 7).getDisplayValue() || ""),
    cost: String(r[7] || ""),
    totalNote: String(r[24] || ""),
    approver: String(r[25] || ""),
    contactInfo: String(r[31] || ""),
    contactEmail: String(r[32] || ""),
    status: aggregateGroupStatus_(statuses),
    items: items
  };
}

/** 🆕 AD列(ステータス)の入力規則(プルダウン制限)に「取消」が含まれていない場合に自動追加する */
function ensureStatusValidationIncludesCancel_(sheet) {
  try {
    const rule = SpreadsheetApp.newDataValidation()
      .requireValueInList(["承認待ち", "承認", "送信完了", "CO30作成待ち", "取消"], true)
      .setAllowInvalid(false)
      .build();
    const lastRow = Math.max(sheet.getLastRow(), 1000); // 将来追加される行の分もあらかじめ余裕を持たせる
    sheet.getRange(2, 30, lastRow - 1, 1).setDataValidation(rule);
  } catch (e) {
    console.error("ステータス列の入力規則更新に失敗しました: " + e.message);
  }
}

/** 🆕 CO31が取消された際、既にCO30化(内部依頼IDで紐付け)されている物品の備考に「取消済み」の印を付ける */
function markCO30ItemsAsCancelled_(internalId) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO30_Inspection");
  if (!sheet || sheet.getLastRow() < 2) return 0;
  const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 16).getValues();
  let count = 0;
  for (let i = 0; i < values.length; i++) {
    const rowInternalId = values[i][15] ? String(values[i][15]).trim() : "";
    if (rowInternalId !== internalId) continue;
    const currentRemarks = values[i][14] ? String(values[i][14]).trim() : "";
    if (currentRemarks.indexOf("⚠️取消済み") !== -1) continue; // 既に印済みなら二重に付けない
    const newRemarks = currentRemarks ? `⚠️取消済み / ${currentRemarks}` : "⚠️取消済み";
    sheet.getRange(i + 2, 15).setValue(newRemarks); // O列=備考
    count++;
  }
  return count;
}

/**
 * 🩺 バックフィル用: 過去に「取消」になったが、まだCO30_Inspection側に印が付いていない依頼を
 * まとめて見つけて備考に印を付ける。Apps Scriptエディタから直接実行してください。
 * (取消済みの依頼はフォーム側にもう「取消」ボタンが出ないため、この関数で後追い対応する)
 */
function BACKFILL_markCancelledCO30Items() {
  const respSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!respSheet || respSheet.getLastRow() < 2) {
    SpreadsheetApp.getUi().alert("CO31_Responsesにデータがありません。");
    return;
  }
  const lastRow = respSheet.getLastRow();
  const values = respSheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();
  let totalMarked = 0;
  let cancelledRequestCount = 0;
  for (let i = 0; i < values.length; i++) {
    const status = String(values[i][29] || "").trim();
    if (status !== "取消") continue;
    const internalId = String(values[i][1] || "").trim();
    if (!internalId) continue;
    cancelledRequestCount++;
    totalMarked += markCO30ItemsAsCancelled_(internalId);
  }
  SpreadsheetApp.getUi().alert(`✅ 完了しました。\n取消済みの依頼: ${cancelledRequestCount}件を確認し、CO30_Inspection上の${totalMarked}件の物品に「⚠️取消済み」の印を新たに付けました。`);
}

function cancelRequest(row) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
    if (!sheet) return { success: false, error: "CO31_Responsesシートが見つかりません" };

    const rowNum = parseInt(row, 10);
    if (isNaN(rowNum) || rowNum < 2) return { success: false, error: "行番号が不正です(" + row + ")" };
    const lastRow = sheet.getLastRow();
    if (rowNum > lastRow) return { success: false, error: "対象の行が見つかりません(行" + rowNum + ")" };

    // 🆕 必要な範囲を1回だけまとめて読む(行ごとにgetRangeすると通信回数が増えて遅い)
    const cols = Math.min(CO31_TOTAL_COLS, sheet.getMaxColumns());
    const all = sheet.getRange(2, 1, lastRow - 1, cols).getValues();
    const get_ = function(rn) {
      const arr = (all[rn - 2] || []).slice();
      while (arr.length < CO31_TOTAL_COLS) arr.push("");
      return arr;
    };

    const oldRow = get_(rowNum);
    const groupId = String(oldRow[CO31_COL_GROUPID_IDX] || "").trim();

    let targetRows = [];
    if (groupId) {
      for (let i = 0; i < all.length; i++) {
        if (String(all[i][CO31_COL_GROUPID_IDX] || "").trim() === groupId) targetRows.push(i + 2);
      }
    }
    if (targetRows.length === 0) targetRows = [rowNum];

    try { ensureStatusValidationIncludesCancel_(sheet); } catch (e) {}

    // ★同期処理はここまで。ステータス更新だけを行い、重い処理は一切しない。
    const reqNos = [], oldStatuses = [], internalIds = [];
    targetRows.forEach(function(rn) {
      const rr = get_(rn);
      const no = String(rr[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
      if (no) reqNos.push(no);
      oldStatuses.push(String(rr[29] || "").trim());
      const iid = String(rr[1] || "").trim();
      if (iid) internalIds.push(iid);
      sheet.getRange(rn, 30).setValue("取消");
    });

    // 🆕 トークン無効化・CO30への印付け・Drive削除・通知メールはすべてバックグラウンドへ。
    //    同期で行うと実行時間の上限に達し、try-catchで捕捉できないエラーになる。
    try {
      const job = {
        dept: String(oldRow[4] || "不明部門").trim(),
        fromLoc: String(oldRow[3] || "").trim(),
        toLoc: String(oldRow[5] || "").trim(),
        contactInfo: String(oldRow[31] || "").trim(),
        contactEmail: String(oldRow[32] || "").trim(),
        rows: targetRows,
        internalIds: internalIds,
        reqNos: reqNos,
        oldStatuses: oldStatuses
      };
      PropertiesService.getScriptProperties()
        .setProperty("cancjob_" + Date.now() + "_" + rowNum, JSON.stringify(job));
      scheduleCancelWorker_(3000);
    } catch (e) {
      console.error("取消の後処理を予約できませんでした: " + e.message);
    }

    return { success: true, cancelledCount: targetRows.length };
  } catch (err) {
    const msg = (err && err.message) ? err.message : String(err);
    console.error("cancelRequestでエラー: " + msg);
    return { success: false, error: msg || "原因不明のエラーです。実行ログをご確認ください。" };
  }
}

/**
 * 🆕 検品スキャナー用の依頼一覧(シンプル版)。
 * 部門・物品名・数量・検品状況(物品コードごとにチェック済みか)だけを返す。
 * CO30_Inspectionの「内部依頼ID」列で紐付け、物品名でグルーピングする。
 * CO30がまだ作成されていない場合は codes が空配列になる。
 */
function listRequestsForScanner() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const respSheet = ss.getSheetByName("CO31_Responses");
  if (!respSheet || respSheet.getLastRow() < 2) return [];
  const lastRow = respSheet.getLastRow();
  const values = respSheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();

  // CO30_Inspectionを内部依頼ID(16列目)でグルーピング → 物品名ごとにコードをまとめる
  // 🆕 「検品済みか」は数量ベースではなく、そのコードの現在のライフサイクル状態で判定する(呼び出し側=scan.htmlで
  //     選択中のモードと比較して判定できるよう、生のlifecycle値をそのまま返す)
  const inspByReqId = {}; // { internalId: { itemName: [{code,lifecycle,remarks,location}] } }
  const inspSheet = ss.getSheetByName("CO30_Inspection");
  if (inspSheet && inspSheet.getLastRow() >= 2) {
    const iv = inspSheet.getRange(2, 1, inspSheet.getLastRow() - 1, 16).getValues();
    iv.forEach(function(r) {
      const code = r[1] ? String(r[1]).trim() : "";
      const name = r[2] ? String(r[2]).trim() : "";
      const internalId = r[15] ? String(r[15]).trim() : "";
      if (!code || !internalId) return;
      const location = r[0] ? String(r[0]).trim() : "";
      const lifecycle = r[11] ? String(r[11]).trim() : "未処理";
      const remarks = r[14] ? String(r[14]).trim() : "";
      if (!inspByReqId[internalId]) inspByReqId[internalId] = {};
      if (!inspByReqId[internalId][name]) inspByReqId[internalId][name] = [];
      inspByReqId[internalId][name].push({ code: code, lifecycle: lifecycle, remarks: remarks, location: location });
    });
  }

  const out = [];
  for (let i = 0; i < values.length; i++) {
    const r = values[i];
    if (!r[4]) continue;
    const status = r[29] ? String(r[29]).trim() : "";
    if (status === "取消") continue;

    const internalId = r[1] ? String(r[1]).trim() : "";
    const dept = String(r[4]).trim();
    const reqNo = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    const requestNote = String(r[24] || "").trim(); // 🆕 CO31の特記事項(Y列)
    const co30Groups = internalId ? inspByReqId[internalId] : null;

    let items;
    if (co30Groups) {
      items = Object.keys(co30Groups).map(function(name) {
        return { name: name, qty: String(co30Groups[name].length), codes: co30Groups[name] };
      });
    } else {
      items = collectItems_(r).filter(function(it) { return it.name; })
        .map(function(it) { return { name: it.name, qty: it.qty || "1", codes: [] }; });
    }

    out.push({
      reqNo: reqNo,
      dept: dept,
      requestNote: requestNote,
      items: items,
      timestamp: (r[0] instanceof Date) ? r[0].getTime() : 0
    });
  }
  out.sort(function(a, b) { return b.timestamp - a.timestamp; });
  return out;
}

function getEquipmentItems() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const masterSheet = ss.getSheetByName("Item_Master");
  if (!masterSheet) return [];
  const values = masterSheet.getDataRange().getValues();
  const items = [];
  for (let i = 1; i < values.length; i++) {
    const name    = values[i][0] ? String(values[i][0]).trim() : "";
    const maker   = values[i][1] ? String(values[i][1]).trim() : "";
    const purpose = values[i][2] ? String(values[i][2]).trim() : "";
    if (name !== "") items.push({ name: name, maker: maker, purpose: purpose });
  }
  return items;
}

/**
 * 🆕 Item_MasterのH列(場所名)・I列(住所)から、配送元フォームの候補と住所を取得する。
 * フォームの「配送元」プルダウンの候補として使う。
 */
function getLocationOptions_() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Item_Master");
  if (!sheet) return [];
  const values = sheet.getDataRange().getValues();
  const seen = {};
  const out = [];
  for (let i = 1; i < values.length; i++) {
    const name = values[i][7] ? String(values[i][7]).trim() : ""; // H列
    if (!name || seen[name]) continue;
    seen[name] = true;
    out.push({
      name: name,
      address: values[i][8] ? String(values[i][8]).trim() : "",   // I列
      code: values[i][6] ? String(values[i][6]).trim() : "",      // 🆕 G列=識別記号
      team: values[i][9] ? String(values[i][9]).trim() : ""       // 🆕 J列=担当TM
    });
  }
  return out;
}

/** getLocationOptions_()を「場所名→住所」のマップ形式にしたもの(CO30生成時の住所書き込みに使う) */
function getLocationAddressMap_() {
  const map = {};
  getLocationOptions_().forEach(function(loc) { map[loc.name] = loc.address; });
  return map;
}

/** google.script.runからフォーム(index.html)が呼び出す用。配送元候補一覧を返す。 */
function getFromLocationOptions() {
  return getLocationOptions_().map(function(loc) { return loc.name; });
}

 * 会場からのレンタル品を「どこから取ったか」記録する際の入力候補として使う。

/**
 * 🆕 1行分のCO31 PDF(PDF-A: 原本+コピー2 / PDF-B: コピー1)を生成して返す。
 * 採番(assignCO31DisplayNumber_)・internalId付与はここで行うが、
 * メール送信・Drive保存・ステータス更新は行わない(sendGroupCO31_側でまとめて実施)。
 */
function generateCO31PdfsForRow_(responseSheet, row, approverDisplayName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const templateSheet31 = ss.getSheetByName("CO31_Template");
  if (!templateSheet31) throw new Error("CO31_Templateシートが見つかりません。");

  const uniqueKey = "REQ_" + Utilities.formatDate(new Date(), "JST", "yyyyMMdd_HHmmss_SSS") + "_R" + row;

  const targetData = responseSheet.getRange(row, 1, 1, CO31_TOTAL_COLS).getValues()[0];
  const docDateStr      = targetData[2] ? String(targetData[2]) : "";
  const fromLocation    = targetData[3] ? String(targetData[3]).trim() : "未指定拠点";
  const toDept          = targetData[4] ? String(targetData[4]).trim() : "不明部門";
  const toLocation      = targetData[5];
  const requestDateTime = String(responseSheet.getRange(row, 7).getDisplayValue());
  const cost            = targetData[7];
  const totalQty        = targetData[23];
  const remarks         = targetData[24];
  const approverName    = targetData[25] ? String(targetData[25]).trim() : "責任者"; // 依頼部門の責任者氏名(フォーム入力値)
  const transportName   = approverDisplayName || "運送部門担当者"; // 🆕 承認ボタンを押した運送・用具部門担当者の氏名
  const contactInfo     = targetData[31] ? String(targetData[31]).trim() : "";
  const contactEmail    = targetData[32] ? String(targetData[32]).trim() : "";

  // 内部依頼ID(CO30連携用・不変)
  const internalId = assignInternalRequestId_(responseSheet, row);

  // 🆕 部門ごとの表示番号を採番する前に、旧番号を控えておく(修正時の古いDriveファイル特定用)
  const oldDisplayBeforeRevision = String(targetData[CO31_COL_DISPLAYPAGE_IDX] || "").trim();

  // 🆕 部門ごとの表示番号を採番(初回="設5"型 / 修正の再承認="設5-1"型)
  // 🆕 修正マーク(変更フィールドのJSON)がある場合だけ改訂版として番号を進める
  const changedRawForRev = targetData[CO31_COL_CHANGED_IDX];
  const isRevisionRequested = !!(changedRawForRev && String(changedRawForRev).trim());
  const numInfo = assignCO31DisplayNumber_(responseSheet, row, toDept, isRevisionRequested);
  const isRev = numInfo.isRevision;

  // 🆕 改訂番号を採番した「直後」にマークを消す。
  //    処理の最後で消すと、途中でタイムアウトした際に再処理で番号が二重に進んでしまう。
  if (isRev) {
    responseSheet.getRange(row, CO31_COL_CHANGED_IDX + 1).setValue("");
  }

  // 🆕 修正で変更されたフィールドを取得(初回承認時は空)
  let changedInfo = { fields: [], itemIndices: [] };
  try {
    const raw = targetData[CO31_COL_CHANGED_IDX];
    if (raw) changedInfo = JSON.parse(raw);
  } catch (e) {}
  function hl(key) { return isRev && changedInfo.fields && changedInfo.fields.indexOf(key) !== -1; }

  // 物品(最大20)→ 5品目/ページに分割
  let items = collectItems_(targetData).filter(function(it) { return it.name || it.qty; });
  if (items.length === 0) items = [{ qty: "", name: "", size: "" }];
  const pages = [];
  for (let i = 0; i < items.length; i += 5) pages.push(items.slice(i, i + 5));
  const totalPages = pages.length;

  const now = new Date();
  const autoSignDate = Utilities.formatDate(now, "JST", "yyyy/MM/dd");
  const autoSignTime = Utilities.formatDate(now, "JST", "HH:mm");

  const token = ScriptApp.getOAuthToken();
  const formattedDocDate = docDateStr ? Utilities.formatDate(new Date(docDateStr), "JST", "yyyy/MM/dd") : autoSignDate;

  let datePart = "", amPmPart = "指定なし", detailTimePart = "なし";
  if (requestDateTime) {
    if (requestDateTime.includes("午後")) { amPmPart = "午後"; let s = requestDateTime.split("午後"); datePart = s[0].trim(); detailTimePart = s[1] ? s[1].trim() : "なし"; }
    else if (requestDateTime.includes("午前")) { amPmPart = "午前"; let s = requestDateTime.split("午前"); datePart = s[0].trim(); detailTimePart = s[1] ? s[1].trim() : "なし"; }
    else { amPmPart = "指定なし"; datePart = requestDateTime.includes("指定なし") ? requestDateTime.split("指定なし")[0].trim() : requestDateTime.trim(); }
  }

  const pdfStorageBlobs = [];   // PDF-A: 原本+コピー2を1枚にまとめた保管・印刷用(運送部門宛)
  const pdfRequesterBlobs = []; // PDF-B: 単独1ページ構成の控え(コピー1・署名欄印字あり、依頼部門宛)

  for (let p = 0; p < totalPages; p++) {
    const pageSuffix = totalPages > 1 ? `_p${p + 1}` : "";
    // 印字する番号(複数ページなら「(1/3)」を添えるが、保存されている番号自体は全ページ共通)
    const printedNumber = totalPages > 1 ? `${numInfo.display} (${p + 1}/${totalPages})` : numInfo.display;

    // ---- 元になるテンプレート(この後、原本+コピーとして複製する) ----
    const tempSheet31 = templateSheet31.copyTo(ss);
    tempSheet31.setName(`TEMP_CO31_P${p + 1}_${uniqueKey}`);

    setCell_(tempSheet31, CO31_NUMBER_CELL, printedNumber, "@");  // 番号(修正時は自動でハイライト不要、番号自体が変わるので分かる)
    setCellHL_(tempSheet31, "S3", formattedDocDate, hl('docDate'), "@");
    setCellHL_(tempSheet31, "C4", fromLocation, hl('fromLocation'));
    setCell_(tempSheet31, "N4", toDept);        // 配送先(部門) ※部門は編集フォームでは固定のためハイライト対象外
    setCellHL_(tempSheet31, "M6", toLocation, hl('toLocation'));    // 配送先(場所)
    setCellHL_(tempSheet31, "M8", contactInfo, hl('contactInfo')); // 担当者名・連絡先
    setCellHL_(tempSheet31, "R9", cost || "", hl('cost'));

    setCellHL_(tempSheet31, "C9", datePart, hl('arrival'));
    if (!requestDateTime.includes("指定なし") && (requestDateTime.includes("午前") || requestDateTime.includes("午後"))) {
      if (amPmPart === "午前") setCellHL_(tempSheet31, "J9", detailTimePart !== "なし" ? detailTimePart : "✔", hl('arrival'));
      else if (amPmPart === "午後") setCellHL_(tempSheet31, "O9", detailTimePart !== "なし" ? detailTimePart : "✔", hl('arrival'));
    }

    const pageItems = pages[p];
    for (let i = 0; i < 5; i++) {
      const it = pageItems[i] || { qty: "", name: "", size: "" };
      const startRow = 12 + i;
      const globalIdx = p * 5 + i;
      const itemChanged = isRev && changedInfo.itemIndices && changedInfo.itemIndices.indexOf(globalIdx) !== -1;
      setCellHL_(tempSheet31, `B${startRow}`, it.qty || "", itemChanged);
      setCellHL_(tempSheet31, `E${startRow}`, it.name || "", itemChanged);
      setCellHL_(tempSheet31, `K${startRow}`, it.size || "", itemChanged);
    }
    if (p === 0) setCell_(tempSheet31, "Q12", totalQty || "");
    setCellHL_(tempSheet31, "C18", remarks || "特になし", hl('totalNote'));
    setCell_(tempSheet31, "C20", transportName || "運送部門担当者");
    setCell_(tempSheet31, "C22", autoSignDate);
    setCell_(tempSheet31, "G22", autoSignTime);
    setCellHL_(tempSheet31, "K20", approverName || "責任者 承認済", hl('approver'));

    // ---- PDF-A: 原本+コピー2を1枚にまとめたシートを作成(保管・印刷用。従来のprintTempSheet31と同じ構成) ----
    const printTempSheet31 = ss.insertSheet(`PRINT_TEMP_31_P${p + 1}_${uniqueKey}`);
    for (let col = 1; col <= 21; col++) printTempSheet31.setColumnWidth(col, tempSheet31.getColumnWidth(col));
    tempSheet31.getRange("A1:U25").copyTo(printTempSheet31.getRange("A1:U25"));
    tempSheet31.getRange("A1:U25").copyTo(printTempSheet31.getRange("A27:U51"));
    for (let r = 1; r <= 25; r++) {
      printTempSheet31.setRowHeight(r, tempSheet31.getRowHeight(r));
      printTempSheet31.setRowHeight(r + 26, tempSheet31.getRowHeight(r));
    }
    printTempSheet31.setRowHeight(26, 20);
    printTempSheet31.getRange("Q2").setValue("【原本】"); printTempSheet31.getRange("Q28").setValue("【コピー2】");
    printTempSheet31.getRange("C20").setValue(""); printTempSheet31.getRange("C22").setValue(""); printTempSheet31.getRange("G22").setValue("");
    printTempSheet31.getRange("C46").setValue(""); printTempSheet31.getRange("C48").setValue(""); printTempSheet31.getRange("G48").setValue("");
    SpreadsheetApp.flush();

    const url31Print = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printTempSheet31.getSheetId()}&r1=0&c1=0&r2=51&c2=21`;
    const storageBlob = UrlFetchApp.fetch(url31Print, { headers: { 'Authorization': `Bearer ${token}` } }).getBlob();
    storageBlob.setName(`CO-31_${toDept}_No${numInfo.display}${pageSuffix}.pdf`);
    pdfStorageBlobs.push(storageBlob);
    try { ss.deleteSheet(printTempSheet31); } catch (e) {}

    // ---- PDF-B: 依頼部門宛、単独1ページ構成(コピー1)。署名欄(C20/C22/G22)は空欄クリアせず、tempSheet31の値をそのまま印字 ----
    const printReqSheet31 = ss.insertSheet(`PRINT_REQ_31_P${p + 1}_${uniqueKey}`);
    for (let col = 1; col <= 21; col++) printReqSheet31.setColumnWidth(col, tempSheet31.getColumnWidth(col));
    tempSheet31.getRange("A1:U25").copyTo(printReqSheet31.getRange("A1:U25"));
    for (let r = 1; r <= 25; r++) {
      printReqSheet31.setRowHeight(r, tempSheet31.getRowHeight(r));
    }
    printReqSheet31.getRange("Q2").setValue("【コピー1】");
    SpreadsheetApp.flush();

    const url31Req = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printReqSheet31.getSheetId()}&r1=0&c1=0&r2=25&c2=21`;
    const requesterBlob = UrlFetchApp.fetch(url31Req, { headers: { 'Authorization': `Bearer ${token}` } }).getBlob();
    requesterBlob.setName(`CO-31_${toDept}_No${numInfo.display}${pageSuffix}.pdf`);
    pdfRequesterBlobs.push(requesterBlob);
    try { ss.deleteSheet(printReqSheet31); } catch (e) {}

    try { ss.deleteSheet(tempSheet31); } catch (e) {}
  }

  return {
    row: row,
    toDept: toDept,
    fromLocation: fromLocation,
    approverName: approverName,
    contactInfo: contactInfo,
    contactEmail: contactEmail,
    display: numInfo.display,
    isRev: isRev,
    oldDisplay: oldDisplayBeforeRevision,
    changeSummary: isRev ? buildChangeSummary_(changedInfo, items) : "",
    totalPages: totalPages,
    storageBlobs: pdfStorageBlobs,     // PDF-A: 原本+コピー2(保管・印刷用、運送部門宛)
    requesterBlobs: pdfRequesterBlobs  // PDF-B: コピー1(署名欄印字あり、依頼部門宛)
  };
}

/**
 * 🆕 グループ(複数行=複数ページ)のCO31をまとめて処理する本体。
 * 全行分のPDFを生成し、Drive保存・依頼部門宛1通・運送部門宛1通のメール送信・ステータス更新を行う。
 * 戻り値: { processedCount, failedPages: [{row, error}], displays: [...] }
 * 一部の行で失敗しても残りは処理を続け、失敗情報を戻り値で返す(呼び出し元が承認者に警告表示する)。
 */
function sendGroupCO31_(responseSheet, rows, approverDisplayName) {
  const truckingEmail = "2026trucking@gmail.com";
  const transportName = approverDisplayName || "運送部門担当者";
  const props = PropertiesService.getScriptProperties();

  const copyDone_  = function(r) { return props.getProperty("co31copy_" + r) === "1"; };
  const printDone_ = function(r) { return props.getProperty("co31print_" + r) === "1"; };

  // 🆕 前回の実行でメール送信まで終わっていた場合(ステータス更新直前に力尽きたケース)は、
  //    PDFの再生成もメールの再送もせず、ステータスの確定だけ行う。
  if (rows.length > 0 && rows.every(copyDone_) && rows.every(printDone_)) {
    rows.forEach(function(r) {
      const st = String(responseSheet.getRange(r, 30).getValue() || "").trim();
      if (st === "承認") {
        responseSheet.getRange(r, 30).setValue("CO30作成待ち");
        responseSheet.getRange(r, CO31_COL_CHANGED_IDX + 1).setValue("");
      }
      props.deleteProperty("co31copy_" + r);
      props.deleteProperty("co31print_" + r);
    });
    return { processedCount: rows.length, failedPages: [], displays: [], resumed: true };
  }

  const results = [];
  const failedPages = [];
  rows.forEach(function(row) {
    try {
      const status = String(responseSheet.getRange(row, 30).getValue() || "").trim();
      if (status === "CO30作成待ち" || status === "送信完了" || status === "取消") return;
      results.push(generateCO31PdfsForRow_(responseSheet, row, approverDisplayName));
    } catch (err) {
      failedPages.push({ row: row, error: err.message });
      console.error("CO31ページ処理に失敗しました(行" + row + "): " + err.message);
    }
  });

  if (results.length === 0) {
    return { processedCount: 0, failedPages: failedPages, displays: [] };
  }

  const first = results[0];
  const toDept = first.toDept;
  const resultRows = results.map(function(r) { return r.row; });
  const displays = results.map(function(r) { return r.display; });
  const displayJoined = displays.join("、");
  const anyRev = results.some(function(r) { return r.isRev; });
  const totalPageCount = results.reduce(function(s, r) { return s + r.totalPages; }, 0);
  const copyAlreadySent  = resultRows.every(copyDone_);
  const printAlreadySent = resultRows.every(printDone_);

  // ---- 印刷用PDFをGoogleドライブに保存 ----
  try {
    const printFolder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
    results.forEach(function(res) {
      if (res.isRev && res.oldDisplay) removeOldDriveFilesForRequest_(printFolder, res.oldDisplay);
    });
    results.forEach(function(res) {
      res.storageBlobs.forEach(function(blob) {
        removeDriveFilesByExactName_(printFolder, blob.getName()); // 🆕 再処理時の重複を防ぐ
        printFolder.createFile(blob);
      });
    });
  } catch (driveErr) {
    console.error("Google Driveへの保存に失敗しました(フォルダIDと共有権限を確認してください): " + driveErr.message);
  }

  // ---- 宛先 ----
  const deptEmailForCc = getDeptEmail_(toDept);
  let mailTo = (first.contactEmail && first.contactEmail.includes("@")) ? first.contactEmail : "";
  let mailCc = deptEmailForCc || "";
  if (!mailTo) { mailTo = mailCc || truckingEmail; mailCc = ""; }
  if (mailCc && mailCc === mailTo) mailCc = "";

  // ---- 変更サマリー ----
  let changeSummaryText = "";
