 *
 * 🆕 今回(4回目)の大規模改修:
 *   A. 番号体系を全面刷新
 *      - CO31_Responses B列: 内部依頼ID(全体通し番号。修正時も不変。CO30_Inspectionとの紐付け専用)
 *      - CO31_Responses 末尾列: 表示用ページ番号(部門ごと・PDF1枚ごとに1から連番。承認のたびに再計算)

 *      - CO30テンプレート AF3(結合 AF3:AK3) に「拠点ごと・PDF1枚ごとの通し番号」を印字(スクリプトプロパティで永続化)

// すべての自動メールはこのラッパー経由で送る（name/replyTo を一括付与）。
// ※内部は MailApp["sendEmail"] とブラケット記法。これにより手順2の一括置換で
//   ラッパー自身が置換されず、自己再帰（無限ループ）を防ぐ。
function sendMail_(recipientOrOptions, subject, body, options) {
  const opts = (typeof recipientOrOptions === 'string')
    ? Object.assign({ to: recipientOrOptions, subject: subject, body: body }, options || {})
    : Object.assign({}, recipientOrOptions);   // 既存の to/cc/subject/body/attachments はそのまま保持
  if (!opts.name)    opts.name = MAIL_SENDER_NAME;      // 呼び出し側で個別指定があればそちら優先
  if (!opts.replyTo) opts.replyTo = MAIL_REPLY_TO;
  MailApp["sendEmail"](opts);
}

function determineTeam_(fromLocation) {
  const loc = String(fromLocation || "").trim();
  if (!loc) return "運送";

  // 🆕 Item_MasterのH列(拠点名)・J列(担当TM)で拠点ごとに担当TMを指定できるようにする。
  //    同じ拠点が複数行ある場合は、最初に見つかった行の値を採用する。
  //    空欄の拠点は従来どおり「運送」扱い。拠点が増えてもコード変更なしで対応できる。
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Item_Master");
    if (sheet) {
      const values = sheet.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][7] || "").trim() !== loc) continue; // H列=拠点名
        const team = String(values[i][9] || "").trim();          // J列=担当TM
        if (team === "レンタル" || team === "運送") return team;
      }
    }
  } catch (e) {}

  // マスタに指定が無い場合のフォールバック(旧仕様の互換)
  if (loc === "コンベンションセンター") return "レンタル";
  return "運送";
}

// =========================================================
// 結合セル対応の安全な書き込みヘルパー
// =========================================================

function setCell_(sheet, a1, value, numberFormat) {
  const range = sheet.getRange(a1);
  const merged = range.getMergedRanges();
  const target = (merged && merged.length > 0) ? merged[0] : range;
  if (numberFormat) target.setNumberFormat(numberFormat);
  target.setValue(value);
}

/** 🆕 setCell_と同じだが、highlightがtrueならセル背景を黄色にする(修正箇所の可視化用) */
function setCellHL_(sheet, a1, value, highlight, numberFormat) {
  const range = sheet.getRange(a1);
  const merged = range.getMergedRanges();
  const target = (merged && merged.length > 0) ? merged[0] : range;
  if (numberFormat) target.setNumberFormat(numberFormat);
  target.setValue(value);
  target.setBackground(highlight ? "#fff3b0" : null);
}

function doGet(e) {
  if (e && e.parameter && e.parameter.page === 'dashboard') {
    return HtmlService.createHtmlOutputFromFile('dashboard')
      .setTitle('CO-30 監督者ダッシュボード')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
  }
  if (e && e.parameter && e.parameter.action === 'approvalPage') {
    const userEmail = Session.getActiveUser().getEmail();
    return showApprovalPage_(e.parameter.token, userEmail);
  }
  if (e && e.parameter && e.parameter.action) {
    return handleApiRequest_(e.parameter.action, e.parameter);
  }
  // 🆕 メール内の「修正リンク」から開かれた場合、mode/dept/reqNoをテンプレートに埋め込んで渡す
  // (Apps ScriptのWebアプリはiframe化されるため、クライアント側でURLパラメータを直接読めないことがある)
  const tmpl = HtmlService.createTemplateFromFile('index');
  tmpl.initialMode  = (e && e.parameter && e.parameter.mode)  ? e.parameter.mode  : '';
  tmpl.initialDept  = (e && e.parameter && e.parameter.dept)  ? e.parameter.dept  : '';
  tmpl.initialReqNo = (e && e.parameter && e.parameter.reqNo) ? e.parameter.reqNo : '';
  return tmpl.evaluate()
      .setTitle('CO-31 大会配送依頼 Webフォーム')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// doPost: 承認フォーム送信 / GitHub Pages API
// =========================================================

function doPost(e) {
  if (e.parameter && e.parameter.action === 'doApprove') {
    // 🆕 Session.getActiveUser()は組織外アカウントで空になることがあるため、
    //    承認ページでプルダウン選択された値を優先し、無ければセッションのメールにフォールバックする。
    const selectedEmail = e.parameter.approverEmail || "";
    const sessionEmail = Session.getActiveUser().getEmail();
    const userEmail = selectedEmail || sessionEmail;
    return processApprovalByToken_(e.parameter.token, userEmail);
  }
  try {
    const body = JSON.parse(e.postData.contents);
    return handleApiRequest_(body.action, body.payload || {});
  } catch (err) {
    return jsonResponse_({ success: false, error: "リクエストの解析に失敗: " + err.message });
  }
}

function handleApiRequest_(action, params) {
  try {
    if (action === 'getInspectionData') {
      const displayItems = getInspectionData(params.location || "", params.team || "");
      const allItems = getInspectionData(params.location || "", "");
      return jsonResponse_({ display: displayItems, all: allItems });
    }
    if (action === 'listRequests') {
      return jsonResponse_({ success: true, requests: listRequestsForScanner() });
    }
    if (action === 'listPrintFolderFiles') {
      const result = listPrintFolderFiles_();
      return jsonResponse_({ success: true, files: result.files, totalCount: result.totalCount });
    }
    if (action === 'updateInspectionQty') {
      const result = updateInspectionQty(
        params.location || "",
        params.itemCode,
        parseInt(params.newQty, 10),
        params.rowId ? parseInt(params.rowId, 10) : null
      );
      return jsonResponse_(result);
    }
    if (action === 'updateLifecycle') {
      // 🆕 5モード(未処理/配送検品済/配送済/回収検品済/返却済)いずれも同じAPIで処理
      const result = updateLifecycle(
        params.location || "",
        params.itemCode,
        params.lifecycle || "未処理"
      );
      return jsonResponse_(result);
    }
    if (action === 'updateFieldBatch') {
      // 🆕 拠点別・備品別ロット表示(親子チェック)からの一括更新用。ダッシュボードのupdateInspectionFieldBatchをJSON API経由で呼べるようにする中継。
      const rowIds = Array.isArray(params.rowIds) ? params.rowIds : String(params.rowIds || '').split(',').filter(Boolean);
      const result = updateInspectionFieldBatch(rowIds, params.field || '', params.value || '');
      return jsonResponse_(result);
    }
    if (action === 'updateRemarks') {
      // 🆕 スキャナー側(依頼一覧・スキャン履歴)から備考(どこから取ったか等)を更新する
      const result = updateInspectionRemarks_(params.location || "", params.itemCode, params.remarks || "");
      return jsonResponse_(result);
    }
    if (action === 'getRentalRemarkOptions') {
      // 🆕 Item_MasterのK列(レンタル備考候補)を返す
      return jsonResponse_({ success: true, options: getRentalRemarkOptions_() });
    }
    return jsonResponse_({ success: false, error: "不明なアクション: " + action });
  } catch (err) {
    return jsonResponse_({ success: false, error: err.message });
  }
}

function jsonResponse_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/** 🆕 印刷待ちフォルダ内のPDFを新しい順に返す(QRラベル印刷ページと統合した印刷センター用) */
/** 🆕 現在「取消」になっているCO31依頼の表示番号一覧を返す(印刷センターでの取消フラグ付け用) */
function getCancelledRequestNumbers_() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!sheet || sheet.getLastRow() < 2) return [];
  const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, CO31_TOTAL_COLS).getValues();
  const nums = [];
  values.forEach(function(r) {
    const status = String(r[29] || "").trim();
    if (status !== "取消") return;
    const reqNo = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    if (reqNo) nums.push(reqNo);
  });
  return nums;
}

function listPrintFolderFiles_() {
  try {
    const cancelledNumbers = getCancelledRequestNumbers_();
    const folder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
    const iter = folder.getFilesByType(MimeType.PDF);
    const files = [];
    while (iter.hasNext()) {
      const f = iter.next();
      const name = f.getName();
      const kind = name.indexOf("CO-30") === 0 || name.indexOf("CO30") !== -1 ? "CO30"
                 : (name.indexOf("CO31") !== -1 || name.indexOf("CO-31") !== -1) ? "CO31" : "その他";
      // 🆕 「No1」が「No10」等に誤マッチしないよう、後ろに数字が続かない場合だけ一致とみなす
      const cancelled = cancelledNumbers.some(function(num) {
        const esc = String(num).replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        return new RegExp("No" + esc + "(?!\\d)").test(name);
      });
      files.push({
        name: name,
        url: f.getUrl(),
        kind: kind,
        cancelled: cancelled,
        createdAt: f.getDateCreated().getTime()
      });
    }
    files.sort(function(a, b) { return b.createdAt - a.createdAt; });
    const CAP = 500; // 🆕 150→500に引き上げ
    return { files: files.slice(0, CAP), totalCount: files.length };
  } catch (e) {
    return { files: [], totalCount: 0 };
  }
}

    const qty  = rowArr[i]     ? String(rowArr[i]).trim()     : "";
    const name = rowArr[i + 1] ? String(rowArr[i + 1]).trim() : "";
    const size = rowArr[i + 2] ? String(rowArr[i + 2]).trim() : "";
    if (name || qty) items.push({ qty: qty, name: name, size: size });
  });
  for (let i = 33; i <= 75; i += 3) { // 物品6〜20
    const qty  = rowArr[i]     ? String(rowArr[i]).trim()     : "";
    const name = rowArr[i + 1] ? String(rowArr[i + 1]).trim() : "";
    const size = rowArr[i + 2] ? String(rowArr[i + 2]).trim() : "";
    if (name || qty) items.push({ qty: qty, name: name, size: size });
  }
  return items;
}

/** グループ内の行ステータスが混在している場合の表示用集約(未完了寄りを優先) */
function aggregateGroupStatus_(statuses) {
  const uniq = [];
  statuses.forEach(function(s) {
    if (s && uniq.indexOf(s) === -1) uniq.push(s);
  });
  if (uniq.length <= 1) return uniq[0] || "";
  const priority = ["承認待ち", "CO30作成待ち", "承認", "送信完了", "取消"];
  for (let i = 0; i < priority.length; i++) {
    if (uniq.indexOf(priority[i]) !== -1) return priority[i];
  }
  return uniq[0];
}

/**

function getDeptAbbrev_(deptName) {

/**
 * この部門で現在使われている最大の「基準番号」(末尾の -N 改訂サフィックスは除く)を調べる。
 */
function getMaxBaseNumberForDept_(sheet, deptName, excludeRow) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return 0;
  const values = sheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();
  let max = 0;
  for (let i = 0; i < values.length; i++) {
    const rowNum = i + 2;
    if (rowNum === excludeRow) continue;
    if (String(values[i][4] || "").trim() !== String(deptName).trim()) continue;
    const disp = String(values[i][CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    if (!disp) continue;
    const m = disp.match(/(\d+)/); // 略語の直後の数字部分=基準番号
    if (m) max = Math.max(max, parseInt(m[1], 10));
  }
  return max;
}

/**
 * 🆕 承認のたびに呼び出す。
 * 初回承認: 部門ごとに新しい基準番号を採番する。例: "設5"
 * 修正後の再承認: 既存の基準番号を維持したまま、末尾に改訂回数を付与する。
 *   例: "設5" → 1回目の修正で "設5-1" → 2回目の修正で "設5-2"
 * これにより、番号を見ただけで「修正版である」ことが一目で分かる。
 */
function assignCO31DisplayNumber_(sheet, row, deptName, isRevisionRequested) {
  const existing = String(sheet.getRange(row, CO31_COL_DISPLAYPAGE_IDX + 1).getValue() || "").trim();

  if (existing) {
    // 🆕 既に番号があっても、修正マークが無ければ「失敗した処理のやり直し」とみなし番号を据え置く。
    //    (これが無いと、再処理のたびに -1, -2 と改訂番号が進んでしまう)
    if (!isRevisionRequested) {
      return { display: existing, isRevision: false };
    }
    const m = existing.match(/^(.*?)(?:-(\d+))?$/);
    let base = existing, revision = 0;
    if (m) {
      base = m[1] || existing;
      revision = m[2] ? parseInt(m[2], 10) : 0;
    }
    const display = `${base}-${revision + 1}`;
    sheet.getRange(row, CO31_COL_DISPLAYPAGE_IDX + 1).setNumberFormat("@").setValue(display);
    return { display: display, isRevision: true };
  }

  const abbrev = getDeptAbbrev_(deptName) || String(deptName || "").trim().slice(0, 1);
  const maxSoFar = getMaxBaseNumberForDept_(sheet, deptName, row);
  const display = `${abbrev}${maxSoFar + 1}`;
  sheet.getRange(row, CO31_COL_DISPLAYPAGE_IDX + 1).setNumberFormat("@").setValue(display);
  return { display: display, isRevision: false };
}

/**
 * 🆕 修正(再承認)の際、旧データと新データを比較して変更されたフィールドを検出する。
 * 戻り値: { fields: ["fromLocation", ...], itemIndices: [0, 2, ...] }
 * fieldsは書式済みの項目名の配列、itemIndicesはcollectItems_後の物品リストの位置(0始まり)。
 */
function detectChangedFields_(oldRow, newRow) {
  const changed = { fields: [], itemIndices: [] };
  const scalarMap = {
    docDate: 2, fromLocation: 3, toLocation: 5, arrival: 6, cost: 7,
    totalNote: 24, approver: 25, contactInfo: 31, contactEmail: 32
  };
  Object.keys(scalarMap).forEach(function(key) {
    const idx = scalarMap[key];
    if (String(oldRow[idx] || "") !== String(newRow[idx] || "")) changed.fields.push(key);
  });

  const oldItems = collectItems_(oldRow);
  const newItems = collectItems_(newRow);
  const maxLen = Math.max(oldItems.length, newItems.length);
  for (let i = 0; i < maxLen; i++) {
    const o = oldItems[i] || { qty: "", name: "", size: "" };
    const n = newItems[i] || { qty: "", name: "", size: "" };
    if (o.qty !== n.qty || o.name !== n.name || o.size !== n.size) changed.itemIndices.push(i);
  }
  return changed;
}

/** 🆕 変更フィールドのキー→日本語ラベル */
const CO31_FIELD_LABELS_ = {
  docDate: "依頼日", fromLocation: "配送元", toLocation: "配送先(場所)",
  arrival: "搬入希望日時", cost: "費用", totalNote: "特記事項",
  approver: "責任者氏名", contactInfo: "担当者・連絡先", contactEmail: "担当者メール"
};

/**
 * 🆕 detectChangedFields_の結果を、メール本文に載せられる日本語の箇条書きサマリーに変換する。
 * newItemsには変更後(collectItems_後)の物品リストを渡す。
 */
function buildChangeSummary_(changedInfo, newItems) {
  const lines = [];
  (changedInfo.fields || []).forEach(function(key) {
    lines.push(`・${CO31_FIELD_LABELS_[key] || key}`);
  });
  if (changedInfo.itemIndices && changedInfo.itemIndices.length > 0) {
    const descs = changedInfo.itemIndices.map(function(idx) {
      const it = newItems[idx];
      return it && it.name ? `${idx + 1}品目目(${it.name})` : `${idx + 1}品目目(削除)`;
    });
    lines.push(`・物品: ${descs.join("、")}`);
  }
  return lines.length > 0 ? lines.join("\n") : "(変更箇所は検出されませんでした)";
}

// 🆕 拠点(Depot_Master)識別記号 と 用具番号の採番
// =========================================================

function getDepotCode_(depotName) {
  const name = String(depotName || "").trim();

  // 🆕 まずItem_MasterのH列(拠点名)・G列(識別記号)を見る。
  //    同じ拠点が複数行ある場合は、最初に見つかった行の値を採用する。
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Item_Master");
    if (sheet) {
      const values = sheet.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][7] || "").trim() !== name) continue; // H列=拠点名
        const code = String(values[i][6] || "").trim();           // G列=識別記号
        if (code) return code;
      }
    }
  } catch (e) {}

  // 従来のDepot_Master / Owner_MasterのC列も引き続き参照する(既存設定を無駄にしないため)
  const candidateSheets = ["Depot_Master", "Owner_Master"];
  for (let s = 0; s < candidateSheets.length; s++) {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(candidateSheets[s]);
    if (!sheet) continue;
    const values = sheet.getDataRange().getValues();
    for (let i = 1; i < values.length; i++) {
      if (String(values[i][0]).trim() === name) {
        const code = values[i][2] ? String(values[i][2]).trim() : "";
        if (code) return code;
      }
    }
  }

  // どこにも無ければ拠点名の先頭1文字を識別記号として使う
  return name.slice(0, 1) || "拠";
}

/** CO30_Inspectionに既にある物品コードから、この識別記号の最大連番を調べる(部門略語プレフィックスが前に付いても検出できるよう、先頭固定はしない) */
function getMaxToolSeqForDepotCode_(depotCode) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO30_Inspection");
  if (!sheet || sheet.getLastRow() < 2) return 0;
  const codes = sheet.getRange(2, 2, sheet.getLastRow() - 1, 1).getValues();
  const escaped = depotCode.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const re = new RegExp(escaped + "(\\d+)");
  let max = 0;
  codes.forEach(function(c) {
    const m = String(c[0] || "").match(re);
    if (m) max = Math.max(max, parseInt(m[1], 10));
  });
  return max;
}

/** CO30拠点ごとの通し番号(印刷用)。スクリプトプロパティで永続化し、バッチ生成をまたいで増え続ける。 */
function nextCO30PageNumber_(depotName) {
  const props = PropertiesService.getScriptProperties();
  const key = "co30pn_" + depotName;
  let cur = parseInt(props.getProperty(key) || "0", 10);
  cur += 1;
  props.setProperty(key, String(cur));
  return cur;
}

/**
 * 🆕 指定した拠点の「現在のCO30_Inspectionの内容」だけを使って、CO30 PDFを作り直す(再印刷)。
 * 用具番号・ライフサイクル・依頼番号など、既存のデータは一切変更しない(読み取るだけ)。
 * PDFは一度作ると内容が更新されない静止画のため、「番号や部門名が古いまま」といったズレに
 * 気づいたときに、最新の正しい内容で印刷用PDFだけを作り直すための機能。
 * ダッシュボードから google.script.run 経由で呼ばれる。
 */
function reprintCO30ForLocation(fromLocation) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const templateSheet30 = ss.getSheetByName("CO30_Template");
    // 🆕 別紙テンプレート。exportAllCO30Combinedと同じルールで、あれば49件/ページの別紙を使う。
    const templateSheetSuppl = ss.getSheetByName(CO30_SUPPL_SHEET_NAME);
    const inspSheet = ss.getSheetByName("CO30_Inspection");
    if (!templateSheet30 || !inspSheet) {
      return { success: false, message: "必要なシートが見つかりません。" };
    }

    // 現在のCO30_Inspectionから、この拠点の行をそのまま読み取る(番号などは一切再計算しない)
    const values = inspSheet.getDataRange().getValues();
    const flattenedLines = [];
    const remarksSet = {};
    for (let i = 1; i < values.length; i++) {
      const row = values[i];
      if (String(row[0] || "").trim() !== fromLocation) continue;
      const code = row[1] ? String(row[1]).trim() : "";
      if (!code) continue;
      const dept = row[8] ? String(row[8]).trim() : "";
      const reqNo = row[9] ? String(row[9]).trim() : "";
      const reqRef = reqNo;
      const internalId = row[15] ? String(row[15]).trim() : ""; // 🆕 別紙の備考重複表示防止に使用
      const requestNote = row[16] ? String(row[16]).trim() : "";
      if (requestNote && dept) remarksSet[`【${dept}】${requestNote}`] = true;
      flattenedLines.push({
        code: code,
        desc: row[2] ? String(row[2]).trim() : "",
        lenderDest: row[10] ? String(row[10]).trim() : "",
        reqRef: reqRef,
        internalId: internalId,       // 🆕
        requestNote: requestNote      // 🆕
      });
    }

    if (flattenedLines.length === 0) {
      return { success: false, message: `拠点「${fromLocation}」の物品がCO30_Inspectionに見つかりませんでした。` };
    }

    // 所有者情報(電話・住所)
    const ownerSheetCandidates = ["Depot_Master", "Owner_Master"];
    let ownerInfo = { phone: "", address: "" };
    for (let s = 0; s < ownerSheetCandidates.length; s++) {
      const sheet = ss.getSheetByName(ownerSheetCandidates[s]);
      if (!sheet) continue;
      const oData = sheet.getDataRange().getValues();
      for (let i = 1; i < oData.length; i++) {
        if (String(oData[i][0]).trim() === fromLocation) {
          ownerInfo = { phone: oData[i][1] || "", address: oData[i][2] || "" };
          break;
        }
      }
    }

    const truckingEmail = "2026trucking@gmail.com";
    const token = ScriptApp.getOAuthToken();
    const now = new Date();
    const autoSignDate = Utilities.formatDate(now, "JST", "yyyy/MM/dd");
    const autoSignTime = Utilities.formatDate(now, "JST", "HH:mm:ss");
    const uniqueKeyBase = "REPRINT_" + Utilities.formatDate(now, "JST", "yyyyMMdd_HHmmss");
    const linesPerPage = 5;

    // 🆕 exportAllCO30Combinedと同じ分割ルール: 1ページ目はCO30_Template(5件)、
    //    残りは別紙テンプレートがあればCO30_別紙_Template(49件/ページ)、無ければ本紙の繰り返し(5件/ページ)。
    const mainCount = Math.min(flattenedLines.length, linesPerPage);
    const remainingCount = flattenedLines.length - mainCount;
    const useSuppl = remainingCount > 0 && !!templateSheetSuppl;
    const remainderCapacity = useSuppl ? CO30_SUPPL_CAPACITY : linesPerPage;
    const remainderPages = remainingCount > 0 ? Math.ceil(remainingCount / remainderCapacity) : 0;
    const totalPages = 1 + remainderPages;
    const mainPagesCount = useSuppl ? 1 : totalPages;

    // ---- メインページ用の結合シート ----
    const printMainSheet = ss.insertSheet(`PRINT_CO30_RP_${uniqueKeyBase}`);
    if (printMainSheet.getMaxColumns() < 40) {
      printMainSheet.insertColumnsAfter(printMainSheet.getMaxColumns(), 40 - printMainSheet.getMaxColumns());
    }
    for (let col = 1; col <= 40; col++) {
      printMainSheet.setColumnWidth(col, templateSheet30.getColumnWidth(col));
    }

    // ---- 別紙用の結合シート(必要な場合のみ) ----
    let printSupplSheet = null;
    if (useSuppl) {
      printSupplSheet = ss.insertSheet(`PRINT_CO30SUPPL_RP_${uniqueKeyBase}`);
      const supplTemplateCols = templateSheetSuppl.getMaxColumns();
      if (printSupplSheet.getMaxColumns() < supplTemplateCols) {
        printSupplSheet.insertColumnsAfter(printSupplSheet.getMaxColumns(), supplTemplateCols - printSupplSheet.getMaxColumns());
      }
      for (let col = 1; col <= supplTemplateCols; col++) {
        printSupplSheet.setColumnWidth(col, templateSheetSuppl.getColumnWidth(col));
      }
    }

    let mainPageNumber = null;
    let mainCursorRow = 1;
    let supplCursorRow = 1;

    for (let p = 0; p < totalPages; p++) {
      const isLastPage = (p === totalPages - 1);

      if (p === 0) {

        const tempSheet30 = templateSheet30.copyTo(ss);
        tempSheet30.setName(`TEMP_CO30_RP_P${p + 1}_${uniqueKeyBase}`);
        if (tempSheet30.getMaxColumns() < 40) {

        tempSheet30.getRange("T31").setNumberFormat("@").setValue(`${p + 1} / ${totalPages}`);

        for (let idx = 0; idx < mainCount; idx++) {
          const lineData = flattenedLines[idx];
          const rowNum = CO30_ITEM_ROWS[idx];
          setCell_(tempSheet30, `${CO30_COL_TOOLNO}${rowNum}`, lineData.code);
          setCell_(tempSheet30, `${CO30_COL_DESC}${rowNum}`, lineData.desc);

          setCell_(tempSheet30, `${CO30_COL_REQREF}${rowNum}`, lineData.reqRef || "");

        }
        tempSheet30.getRange("A1:AN31").copyTo(printMainSheet.getRange(mainCursorRow, 1, 31, 40));
        try { ss.deleteSheet(tempSheet30); } catch (e) {}
        mainCursorRow += mainIsLastOfItsPdf ? 31 : 33;

      } else if (useSuppl) {

        const tempSheetSuppl = templateSheetSuppl.copyTo(ss);

        }

        const pageNumber = nextCO30PageNumber_(fromLocation);
        setCell_(tempSheet30, CO30_NUMBER_CELL, `${pageNumber}(再印刷)`, "@");
        tempSheet30.getRange("D3").setNumberFormat("@").setValue(autoSignDate);
        tempSheet30.getRange("E5").setValue(fromLocation);
        tempSheet30.getRange("AF5").setNumberFormat("@").setValue(ownerInfo.phone);
        tempSheet30.getRange("D7").setValue(ownerInfo.address);
        tempSheet30.getRange("D22").setValue(Object.keys(remarksSet).join("\n") || "特になし");
        tempSheet30.getRange("T31").setNumberFormat("@").setValue(`${p + 1} / ${totalPages}`);

        const startIdx = mainCount + (p - 1) * linesPerPage;
        const endIdx = Math.min(startIdx + linesPerPage, flattenedLines.length);
        for (let idx = startIdx; idx < endIdx; idx++) {
          const lineData = flattenedLines[idx];
          const rowNum = CO30_ITEM_ROWS[idx - startIdx];
          setCell_(tempSheet30, `${CO30_COL_TOOLNO}${rowNum}`, lineData.code);
          setCell_(tempSheet30, `${CO30_COL_DESC}${rowNum}`, lineData.desc);
          setCell_(tempSheet30, `${CO30_COL_LENDER}${rowNum}`, lineData.lenderDest || "");
          setCell_(tempSheet30, `${CO30_COL_REQREF}${rowNum}`, lineData.reqRef || "");

        for (let r = 1; r <= 31; r++) {

        }
        if (!isLastPage) {
          printMainSheet.setRowHeight(mainCursorRow + 31, 20);
          printMainSheet.setRowHeight(mainCursorRow + 32, 20);
        }
        tempSheet30.getRange("A1:AN31").copyTo(printMainSheet.getRange(mainCursorRow, 1, 31, 40));
        try { ss.deleteSheet(tempSheet30); } catch (e) {}
        mainCursorRow += isLastPage ? 31 : 33;
      }
    }

    SpreadsheetApp.flush();

    // ---- メインPDFを書き出す ----
    const mainRowsCount = mainCursorRow - 1;
    const urlMain = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printMainSheet.getSheetId()}&r1=0&c1=0&r2=${mainRowsCount}&c2=39`;
    const blobMain = UrlFetchApp.fetch(urlMain, { headers: { 'Authorization': `Bearer ${token}` } })
      .getBlob().setName(`CO-30_再印刷_[${fromLocation}]_${autoSignDate} ${autoSignTime}(全${mainPagesCount}頁).pdf`);
    try { ss.deleteSheet(printMainSheet); } catch (e) {}

    // ---- 別紙PDFを書き出す(ある場合のみ) ----
    let blobSuppl = null;
    if (useSuppl) {
      const supplRowsCount = supplCursorRow - 1;
      const supplColsForExport = 5; // 🆕 用具番号/説明/貸出先/備考/依頼番号の5列固定。getLastColumn()は環境によって不安定なため直接指定する
      const urlSuppl = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printSupplSheet.getSheetId()}&r1=0&c1=0&r2=${supplRowsCount}&c2=${supplColsForExport}`;
      blobSuppl = UrlFetchApp.fetch(urlSuppl, { headers: { 'Authorization': `Bearer ${token}` } })
        .getBlob().setName(`CO-30_別紙_再印刷_[${fromLocation}]_${autoSignDate} ${autoSignTime}(全${remainderPages}頁).pdf`);
      try { ss.deleteSheet(printSupplSheet); } catch (e) {}
    }

    try {
      const printFolder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
      removeOldCO30FilesForLocation_(printFolder, fromLocation); // 同じ拠点の古いCO-30(本紙・別紙とも)をゴミ箱へ
      printFolder.createFile(blobMain.copyBlob());
      if (blobSuppl) printFolder.createFile(blobSuppl.copyBlob());
    } catch (driveErr) {
      console.error("CO30再印刷のGoogle Drive保存に失敗しました: " + driveErr.message);
    }

    const supplNote = blobSuppl ? `\n\n※物品数が多いため、別紙(CO-30_別紙_再印刷_...)も別ファイルで添付しています。` : "";
    sendMail_({
      to: truckingEmail,
      subject: `【再印刷】CO-30 用具受取書(拠点: ${fromLocation})`,
      body: `運送・用具部門 担当者様\n\nダッシュボードから、拠点「${fromLocation}」のCO-30を最新のデータで再印刷しました。\n番号・検品状態など既存のデータは変更していません。古いPDFと差し替えてご利用ください。${supplNote}`,
      attachments: blobSuppl ? [blobMain, blobSuppl] : [blobMain]
    });

    return { success: true, message: `✅ 拠点「${fromLocation}」のCO-30を再印刷し、メール送信・Driveへの保存を行いました。` };
  } catch (err) {
    return { success: false, message: "❌ 再印刷に失敗しました: " + err.message };
  }
}

/**
 * 🆕 Dept_Masterで「運送・用具部門」の行を"行ごと"に走査し、各行のD列(メール)・E列(氏名)を
 * ペアで返す。以前は1セルにカンマ区切りで複数メールを入れていたが、
 * 承認者ごとに行を分けて入力する方式に変更した。
 * 例:
 *   運送・用具部門 | ... | ... | tanaka@example.com | 田村裕
 *   運送・用具部門 | ... | ... | gima@example.com    | 儀間悟
 */
function getApproversWithNames_() {
  const fallback = [{ email: "2026trucking@gmail.com", name: "" }];
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Dept_Master");
    if (!sheet) return fallback;
    const lastRow = sheet.getLastRow();
    if (lastRow < 1) return fallback;
    const values = sheet.getRange(1, 1, lastRow, 5).getValues(); // A〜E列

    // 🆕 A列(部門名)が結合セルになっている場合、Apps Scriptは結合範囲の1行目にしか値を返さない。
    //    2行目以降が見た目は「運送・用具部門」でも実際には空欄というケースに対応するため、
    //    結合範囲の中では上のセルの値を引き継がせる。
    const effectiveDeptA = values.map(function(r) { return String(r[0] || "").trim(); });
    const mergedRanges = sheet.getRange(1, 1, lastRow, 1).getMergedRanges();
    mergedRanges.forEach(function(mr) {
      const topRow = mr.getRow();
      const numRows = mr.getNumRows();
      const anchorVal = effectiveDeptA[topRow - 1];
      for (let r = topRow; r < topRow + numRows; r++) {
        effectiveDeptA[r - 1] = anchorVal;
      }
    });

    const list = [];
    for (let i = 0; i < values.length; i++) {
      if (effectiveDeptA[i] !== "運送・用具部門") continue;
      const cellD = values[i][3] ? String(values[i][3]).trim() : "";
      const cellE = values[i][4] ? String(values[i][4]).trim() : "";
      if (!cellD) continue;
      // 🆕 「1セルにカンマ区切りで複数メール」という旧形式が残っていても読めるようにする
      const emails = cellD.split(",").map(function(e) { return e.trim(); }).filter(function(e) { return e.includes("@"); });
      if (emails.length === 1) {
        list.push({ email: emails[0], name: cellE });
      } else if (emails.length > 1) {
        // 旧形式(1セルに複数メール)の場合、氏名を個別に紐づけられないため空のまま登録する
        emails.forEach(function(email) { list.push({ email: email, name: "" }); });
      }
    }
    return list.length > 0 ? list : fallback;
  } catch (e) {}
  return fallback;
}

function getApprovers_() {
  return getApproversWithNames_().map(function(a) { return a.email; });
}

/** 🆕 承認ボタンを押したアカウントのメールから、Dept_MasterのE列(氏名)を逆引きする(前後空白・大文字小文字ゆれを吸収) */
function getApproverNameByEmail_(email) {
  if (!email) return "";
  const target = String(email).trim().toLowerCase();
  const found = getApproversWithNames_().find(function(a) { return a.email.trim().toLowerCase() === target; });
  return found ? found.name : "";
}

// 🆕 1グループ=1トークン: トークンは複数行(rowIds)を束ねて承認する。
//    旧形式トークン({rowId: n})がScriptPropertiesに残っていても読めるよう、
//    読み取り側は必ず tokenRowIds_() を通して行番号の配列を取り出すこと。
function generateApprovalToken_(rowIds) {
  const token = Utilities.getUuid().replace(/-/g, "");
  PropertiesService.getScriptProperties().setProperty(
    "appr_" + token,
    JSON.stringify({ rowIds: rowIds, created: Date.now(), status: "pending" })
  );
  return token;
}

/** トークンデータから対象行番号の配列を取り出す(旧形式 {rowId} との後方互換) */
function tokenRowIds_(data) {
  if (data && data.rowIds && data.rowIds.length) return data.rowIds.map(function(r) { return Number(r); });
  if (data && data.rowId) return [Number(data.rowId)];
  return [];
}

function invalidateTokensForRow_(rowId) {
  invalidateTokensForRows_([rowId]);
}

/**
 * 🆕 複数行のトークンをまとめて無効化する。
 * getKeys()+getProperty()をキーの数だけ呼ぶと通信回数が膨大になるため、
 * getProperties()で一度に全件取得してからメモリ上で判定する。
 */
function invalidateTokensForRows_(rowIds) {
  const props = PropertiesService.getScriptProperties();
  const set = {};
  (rowIds || []).forEach(function(r) { set[Number(r)] = true; });
  if (Object.keys(set).length === 0) return;

  let all = {};
  try { all = props.getProperties(); } catch (e) { return; }

  Object.keys(all).forEach(function(k) {
    if (k.indexOf("co31copy_") === 0 || k.indexOf("co31print_") === 0) {
      const n = Number(k.substring(k.indexOf("_") + 1));
      if (set[n]) { try { props.deleteProperty(k); } catch (e) {} }
      return;
    }
    if (k.indexOf("appr_") !== 0) return;
    try {
      const d = JSON.parse(all[k]);
      if (d && d.status === "pending" && tokenRowIds_(d).some(function(r) { return set[r]; })) {
        props.deleteProperty(k);
      }
    } catch (e) {}
  });
}

/**
 * 🆕 トークンを介さず、行番号配列を直接承認する共通コア処理。
 * ダッシュボードからの一括承認で使う(メールリンク経由のapproveByToken_と処理内容は同じ)。
 */
function approveRowsDirect_(rows, approverEmail) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!sheet) return { success: false, message: "CO31_Responsesが見つかりません" };
  if (!rows || rows.length === 0) return { success: false, message: "対象行がありません" };

  const props = PropertiesService.getScriptProperties();
  const approverDisplayName = getApproverNameByEmail_(approverEmail) || approverEmail || "運送部門担当者";

  const targetRows = rows.filter(function(r) {
    const status = String(sheet.getRange(r, 30).getValue() || "").trim();
    return status !== "CO30作成待ち" && status !== "送信完了" && status !== "取消" && status !== "承認";
  });
  if (targetRows.length === 0) return { success: false, message: "承認待ちの行がありませんでした(既に処理済みの可能性があります)" };

  // 🆕 ここでもPDF生成はせず、キューに任せる
  let firstIid = "";
  targetRows.forEach(function(r) {
    const iid = assignInternalRequestId_(sheet, r);
    if (!firstIid) firstIid = iid;
    props.setProperty("apprby_id_" + iid, approverDisplayName);
    sheet.getRange(r, 30).setValue("承認");
    invalidateTokensForRow_(r);
  });
  if (firstIid) {
    props.deleteProperty("apprfail_" + firstIid);
    props.deleteProperty("apprnotified_" + firstIid);
  }
  scheduleApprovalWorker_(5000);

  return {
    success: true, processedCount: targetRows.length, failedCount: 0,
    message: targetRows.length + "件を承認しました(控えは数分以内に送信されます)"
  };
}

/**
 * 🆕 ダッシュボードから複数の依頼グループをまとめて承認する。
 * groupsRowsList: [[行番号,...], [行番号,...], ...] の配列。approverEmailは承認権限者のいずれか。
 */
function approvePendingGroupsFromDashboard(groupsRowsList, approverEmail) {
  const lock = LockService.getScriptLock();
  const gotLock = lock.tryLock(15 * 1000);
  if (!gotLock) return { success: false, message: "⏳ 他の処理が実行中です。少し待ってからもう一度お試しください。" };
  try {
    if (!approverEmail) return { success: false, message: "承認者を選択してください。" };
    const approvers = getApprovers_();
    if (approvers.indexOf(approverEmail) === -1) return { success: false, message: "選択された承認者には承認権限がありません。" };

    let totalProcessed = 0, totalFailed = 0, groupCount = 0, errorCount = 0;
    (groupsRowsList || []).forEach(function(rows) {
      const ids = (rows || []).map(function(r) { return parseInt(r, 10); }).filter(function(r) { return !isNaN(r) && r >= 2; });
      if (ids.length === 0) return;
      try {
        const res = approveRowsDirect_(ids, approverEmail);
        if (res.success) { groupCount++; totalProcessed += res.processedCount; totalFailed += res.failedCount; }
        else { errorCount++; }
      } catch (err) {
        errorCount++;
      }
    });

    return {
      success: groupCount > 0,
      message: `✅ ${groupCount}件の依頼を承認しました${totalFailed > 0 ? `(${totalFailed}ページで一部失敗)` : ""}${errorCount > 0 ? `、${errorCount}件は処理できませんでした` : ""}`
    };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 🆕 重複依頼チェック。同じ部門から、同じ品名を含む依頼が既に出ていないか調べ、
 * 見つかった場合だけメール本文に差し込む注意文言を返す(無ければ空文字)。
 *
 * 判定基準:
 *   ・同じ依頼部門であること
 *   ・品名が一致すること(前後の空白と大文字小文字の違いは無視)
 *   ・ステータスが「取消」以外であること(承認待ち・承認済み・送信完了はすべて対象)
 *   ・自分自身の依頼(同じ親依頼IDで束ねた全行)は当然除外する
 * ※日付や配送先は見ていないため、別日・別場所への正当な依頼も候補に挙がる。
 *   あくまで「確認を促す」ための文言であり、送信をブロックはしない。
 */
function buildDuplicateWarning_(deptName, itemNames, ownRows) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
    if (!sheet || sheet.getLastRow() < 2) return "";
    const dept = String(deptName || "").trim();
    if (!dept) return "";

    const wanted = {};
    (itemNames || []).forEach(function(n) {
      const key = String(n || "").trim().toLowerCase();
      if (key) wanted[key] = String(n).trim();
    });
    if (Object.keys(wanted).length === 0) return "";

    const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, CO31_TOTAL_COLS).getValues();

    // 自分自身の依頼は除外する。親依頼IDで束ねた兄弟行もまとめて除外(修正の再承認で誤検知しないように)
    const ownSet = {};
    (ownRows || []).forEach(function(r) { ownSet[Number(r)] = true; });
    const ownGroupIds = {};
    for (let i = 0; i < values.length; i++) {
      if (!ownSet[i + 2]) continue;
      const gid = String(values[i][CO31_COL_GROUPID_IDX] || "").trim();
      if (gid) ownGroupIds[gid] = true;
    }

    const hits = {};
    for (let i = 0; i < values.length; i++) {
      const rowNum = i + 2;
      const r = values[i];
      if (ownSet[rowNum]) continue;
      const gid = String(r[CO31_COL_GROUPID_IDX] || "").trim();
      if (gid && ownGroupIds[gid]) continue;
      if (String(r[4] || "").trim() !== dept) continue;
      const status = String(r[29] || "").trim();
      if (status === "取消") continue;

      const dispNo = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim() || "未採番";
      collectItems_(r).forEach(function(it) {
        const key = String(it.name || "").trim().toLowerCase();
        if (!key || !wanted[key]) return;
        if (!hits[key]) hits[key] = [];
        const label = `No.${dispNo}(${status || "状態不明"})`;
        if (hits[key].indexOf(label) === -1) hits[key].push(label);
      });
    }

    const keys = Object.keys(hits);
    if (keys.length === 0) return "";
    const lines = keys.slice(0, 10).map(function(k) {
      return `  ・${wanted[k]} … ${hits[k].slice(0, 5).join("、")}`;
    });
    return `
━━━━━━━━━━━━━━━━━━━━━━━
⚠️ 重複の可能性はありませんか?
同じ部門から、以下の品名を含む依頼が既に出ています。
${lines.join("\n")}
二重の手配になっていないかご確認ください。
(搬入日や配送先が異なる別の依頼であれば問題ありません)
━━━━━━━━━━━━━━━━━━━━━━━
`;
  } catch (e) {
    return "";
  }
}

▼ 承認する場合は以下のリンクをタップしてください
${approvalUrl}
━━━━━━━━━━━━━━━━━━━━━━━

  // 🔒 承認依頼メール(承認リンク付き)は、運送・用具部門の承認者(Dept_Master D14:E15)だけに送る。
  //    依頼部門(Dept_MasterのB列)には絶対に送らない ── 自分で承認できてしまうため。
  //    依頼部門が受け取るのは、承認後に sendGroupCO31_ が送る控えメール(承認リンクなし)だけ。
  const ccList = approvers.slice(1);

  sendMail_({
    to: approvers[0],
    cc: ccList.join(","),
    subject: `${isEdit ? "【承認依頼・CO-31 修正】" : "【承認依頼・CO-31】"}${dept}からの配送依頼${pageCount > 1 ? `(全${pageCount}ページ)` : ""}`,
    body: body
  });
}

function showApprovalPage_(token, userEmail) {
  const props = PropertiesService.getScriptProperties();
  const dataStr = props.getProperty("appr_" + token);

  if (!dataStr) {
    return buildSimplePage_("❌ リンクが無効です", "このリンクは存在しないか、すでに無効になっています。\n(依頼が修正された場合、古い承認リンクは無効になります)");
  }
  const data = JSON.parse(dataStr);

  if (data.status === "approved") {
    const approvedAt = data.approvedAt ? new Date(data.approvedAt).toLocaleString("ja-JP") : "不明";
