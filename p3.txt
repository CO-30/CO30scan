 let changeSummaryText = "";
  const revResults = results.filter(function(r) { return r.isRev && r.changeSummary; });
  if (revResults.length > 0) {
    const parts = revResults.map(function(r) {
      return results.length > 1 ? `【No.${r.display}】\n${r.changeSummary}` : r.changeSummary;
    });
    changeSummaryText = `\n※修正版を含みます。控えPDF内で黄色くハイライトされている箇所が、前回からの変更点です。\n\n■ 今回変更された箇所\n${parts.join("\n")}\n`;
  }

  const failNote = failedPages.length > 0
    ? `\n⚠️ この依頼のうち${failedPages.length}ページで控えの作成に失敗しました。不足分は運送・用具部門にお問い合わせください。\n`
    : "";

  // ---- 重複依頼の注意文言 ----
  let dupNote = "";
  try {
    if (typeof buildDuplicateWarning_ === 'function') {
      const dupNames = [];
      rows.forEach(function(rn) {
        const rr = responseSheet.getRange(rn, 1, 1, CO31_TOTAL_COLS).getValues()[0];
        collectItems_(rr).forEach(function(it) { if (it.name) dupNames.push(it.name); });
      });
      dupNote = buildDuplicateWarning_(toDept, dupNames, rows);
    }
  } catch (e) {}

  // ---- 控えメール(依頼部門宛) ----
  if (!copyAlreadySent) {
    try {
      const editLink = `${getFormBaseUrl_()}?mode=edit&dept=${encodeURIComponent(toDept)}&reqNo=${encodeURIComponent(displays[0])}`;

      const copyBody = `${first.approverName} 兄弟

お疲れ様です。運送・用具部門の${transportName}です。
配送依頼(CO-31)が承認されましたので控えをお送りします。

番号: ${displayJoined}${changeSummaryText}${failNote}${dupNote}
① 依頼の修正が必要な時
・以下のリンクから直接修正フォームを開けます
　${editLink}

② 用具を受け取る時
・配送された用具をご確認ください。
・受領確認として、運送部門が持参するCO-31(コピー)へ受取担当者の署名をお願いします。
・お渡しするCO-31原本は、返却時まで部門で保管してください。

③ 使用後に運送部門へ返却する時
・用具を運送部門へ返却してください。
・保管していたCO-31原本に、返却確認の署名をお願いします。
・これで部門からの返却手続きは完了となります。

④ お願い
・CO-31は、配送・受領・返却の記録となる大切な書類です。紛失しないよう保管をお願いいたします。
・返却時は、借用時と同じ数量・状態でご返却くださいますようお願いいたします。
・数量や状態に変更があった場合は、返却時に運送部門へお知らせください。

兄弟たちのご協力に心から感謝しています。

※部門代表アドレスにもCCでお送りしています。`;

      const copyBlobs = [];
      results.forEach(function(res) { res.requesterBlobs.forEach(function(b) { copyBlobs.push(b); }); });

      const copyOpts = {
        to: mailTo,
        subject: `【${anyRev ? '修正版・' : ''}配送依頼受領】CO-31 大会配送依頼控え(${toDept} / No.${displayJoined})`,
        body: copyBody,
        attachments: copyBlobs
      };
      if (mailCc) copyOpts.cc = mailCc;
      sendMail_(copyOpts);
      // 🆕 送信できたことを記録し、再処理時に二重送信しないようにする
      resultRows.forEach(function(r) { props.setProperty("co31copy_" + r, "1"); });
    } catch (e) {
      console.error("依頼元へのメール送信でエラーが発生しました: " + e.message);
    }
  }

  // ---- 印刷用メール(運送部門宛) ----
  if (!printAlreadySent) {
    try {
      const printBlobs = [];
      results.forEach(function(res) { res.storageBlobs.forEach(function(b) { printBlobs.push(b); }); });
      sendMail_({
        to: truckingEmail,
        subject: `【要対応・CO-31印刷用】${toDept} 大会配送依頼(No.${displayJoined})`,
        body: `運送・用具部門 担当者様\n\n${toDept}の配送依頼(No.${displayJoined})が承認されました。印刷用CO-31を添付します(全${totalPageCount}ページ構成)。\nGoogleドライブの印刷待ちフォルダにも保存済みです。${anyRev ? '\n※修正版を含みます。同じ依頼の古いPDFはドライブ内で削除済みです。' : ''}${failNote}\n※この部門のCO-30受取書は、最終締切後にメニューから一括生成されます。`,
        attachments: printBlobs
      });
      resultRows.forEach(function(r) { props.setProperty("co31print_" + r, "1"); });
    } catch (e) {
      console.error("運送部門への印刷用メール送信でエラーが発生しました: " + e.message);
    }
  }

  // ---- ステータスを確定し、一時的な記録を消す ----
  results.forEach(function(res) {
    responseSheet.getRange(res.row, 30).setValue("CO30作成待ち");
    responseSheet.getRange(res.row, CO31_COL_CHANGED_IDX + 1).setValue("");
    props.deleteProperty("co31copy_" + res.row);
    props.deleteProperty("co31print_" + res.row);
  });

  return { processedCount: results.length, failedPages: failedPages, displays: displays };
}

}

/** 🆕 修正版を保存する前に、ファイル名に「旧番号」を含むDrive上の古いPDFをゴミ箱へ移動する(番号で厳密に絞り込むため誤削除のリスクは低い) */
function removeOldDriveFilesForRequest_(folder, oldNumber) {
  if (!oldNumber) return;
  try {
    const esc = String(oldNumber).replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    const re = new RegExp("No" + esc + "(?!\\d)");
    const iter = folder.getFiles();
    while (iter.hasNext()) {
      const f = iter.next();
      const name = f.getName();
      if (re.test(name)) {
        try { f.setTrashed(true); } catch (e2) {}
      }
    }
  } catch (e) {}
}

/**
 * 🆕 CO-30「再印刷」専用: 同じ拠点の古いCO-30(通常版・再印刷版どちらも)をゴミ箱に移す。
 * 再印刷は「今の状態のまま同じ内容で作り直す」機能なので、古い版は必ず内容が古くなる。
 * ⚠️ 4時間おきの自動バッチ(exportAllCO30Combined)には絶対に使わないこと。
 *    バッチ生成のCO-30は毎回「新しく溜まった依頼」を含む別内容のため、消すと依頼ごと消えてしまう。
 */
function removeOldCO30FilesForLocation_(folder, fromLocation) {
  if (!fromLocation) return;
  try {
    const marker = `[${fromLocation}]`;
    const iter = folder.getFilesByType(MimeType.PDF);
    while (iter.hasNext()) {
      const f = iter.next();
      const name = f.getName();
      if (name.indexOf("CO-30") !== -1 && name.indexOf(marker) !== -1) {
        try { f.setTrashed(true); } catch (e2) {}
      }
    }
  } catch (e) {}
}

function exportAllCO30Combined() {
  // 🆕 呼び出し元(メニュー/トリガー/ダッシュボード)によって結果の見せ方が異なるため、
  //    ここではUIアラートなどを直接呼ばず、常に結果オブジェクト { success, message, supersededCount } を返す。
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const responseSheet = ss.getSheetByName("CO31_Responses");

  const templateSheet30 = ss.getSheetByName("CO30_Template");
  // 🆕 別紙テンプレート。存在しない場合はnullのままとし、従来通りCO30_Templateの繰り返しにフォールバックする。
  const templateSheetSuppl = ss.getSheetByName(CO30_SUPPL_SHEET_NAME);
  const ownerSheetCandidates = ["Depot_Master", "Owner_Master"];
  let ownerSheet = null;
  for (let i = 0; i < ownerSheetCandidates.length; i++) {
    const s = ss.getSheetByName(ownerSheetCandidates[i]);
    if (s) { ownerSheet = s; break; }
  }

  const truckingEmail = "2026trucking@gmail.com";

  if (!responseSheet || !templateSheet30) {
    return { success: false, message: "エラー：必要なシートが見つかりません。", supersededCount: 0 };
  }
  const lastRow = responseSheet.getLastRow();
  if (lastRow < 2) { return { success: false, message: "対象データがありません。", supersededCount: 0 }; }

  const dataRange = responseSheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();
  let targetRows = [];
  // 拠点ごとに: { lines:[{code,desc,dept,lenderDest,reqRef}], remarks:[] }
  let locationGroup = {};
  let inspectionRows = [];

  let ownerMap = {};
  if (ownerSheet) {
    const oData = ownerSheet.getDataRange().getValues();
    for (let i = 1; i < oData.length; i++) {
      let locName = oData[i][0] ? String(oData[i][0]).trim() : "";
      // 🆕 C列は識別記号(用具番号用)に使うため、ここでは電話番号のみ拾う。住所はItem_MasterのH/I列から別途取得する。
      if (locName) ownerMap[locName] = { phone: oData[i][1] || "", address: "" };
    }
  }
  // 🆕 拠点の住所をItem_MasterのH列(場所名)・I列(住所)から取得し、ownerMapへ上書きする。
  // フォームで「その他(手動入力)」により配送元が自由入力された場合はH列に一致しないため、住所は空欄のままになる。
  const locationAddressMap = getLocationAddressMap_();
  Object.keys(locationAddressMap).forEach(function(locName) {
    if (!ownerMap[locName]) ownerMap[locName] = { phone: "", address: "" };
    ownerMap[locName].address = locationAddressMap[locName];
  });

  for (let i = 0; i < dataRange.length; i++) {
    let rowNum = i + 2;
    let rowArr = dataRange[i];
    let status = rowArr[29];
    if (status !== "CO30作成待ち") continue;

    targetRows.push(rowNum);

    let fromLocation = rowArr[3] ? String(rowArr[3]).trim() : "未指定拠点";
    let deptName = rowArr[4] ? String(rowArr[4]).trim() : "不明部門";
    let toLocation = rowArr[5] ? String(rowArr[5]).trim() : "";
    let note = rowArr[24] ? String(rowArr[24]).trim() : "";
    let team = rowArr[30] ? String(rowArr[30]).trim() : determineTeam_(fromLocation);
    let internalId = rowArr[1] ? String(rowArr[1]).trim() : "";
    if (!internalId) internalId = assignInternalRequestId_(responseSheet, rowNum);
    let displayPage = String(rowArr[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    if (!displayPage) {
      // 🆕 何らかの理由(ステータス欄の手動書き換え等)で番号が未採番のまま「CO30作成待ち」になっている行を
      //    ここで検知し、その場で番号を採番してシートにも書き戻す(以後は正しい番号が表示される)。
      const backfilled = assignCO31DisplayNumber_(responseSheet, rowNum, deptName);
      displayPage = backfilled.display;
      rowArr[CO31_COL_DISPLAYPAGE_IDX] = displayPage; // このループ内で後から参照される場合に備えて更新
    }
    let reqRef = displayPage;

    if (!locationGroup[fromLocation]) locationGroup[fromLocation] = { lines: [], remarks: [] };
    if (note) {
      // 🆕 同じ依頼(親IDで束ねた複数行)は同じ依頼メモを持つため、同一文言の備考は1回だけ載せる
      const remarkLine = `【${deptName}】${note}`;
      if (locationGroup[fromLocation].remarks.indexOf(remarkLine) === -1) {
        locationGroup[fromLocation].remarks.push(remarkLine);
      }
    }

    // 🆕 CO31の数量・品名・サイズ→CO30の説明欄へ結合
    const reqItems = collectItems_(rowArr);
    reqItems.forEach(function(it) {
      const qty = parseInt(it.qty, 10) || 0;
      const name = it.name;
      if (!name || qty <= 0) return;

      let baseDesc = name;
      if (it.size) baseDesc += ` [${it.size}]`;

      if (qty === 1) {
        locationGroup[fromLocation].lines.push({
          desc: baseDesc, dept: deptName, lenderDest: toLocation, reqRef: reqRef,
          plan: 1, internalId: internalId, suffixMode: "none", requestNote: note
        });
      } else if (qty <= 5) {
        for (let b = 0; b < qty; b++) {
          locationGroup[fromLocation].lines.push({
            desc: baseDesc, dept: deptName, lenderDest: toLocation, reqRef: reqRef,
            plan: 1, internalId: internalId, suffixMode: "letter", suffixIndex: b, requestNote: note
          });
        }
      } else {
        // 数量6以上: 枝番なしで1行にまとめ、説明欄に数量を明記
        locationGroup[fromLocation].lines.push({
          desc: `${baseDesc} ×${qty}`, dept: deptName, lenderDest: toLocation, reqRef: reqRef,
          plan: qty, internalId: internalId, suffixMode: "star", requestNote: note
        });
      }
    });
  }

  let locations = Object.keys(locationGroup);
  if (locations.length === 0) {
    return { success: false, message: "現在「CO30作成待ち」のデータはありません。", supersededCount: 0 };
  }

  // 🆕 拠点ごとに識別記号を取得し、用具番号(物品コード)を採番
  const now = new Date();
  const autoSignDate = Utilities.formatDate(now, "JST", "yyyy/MM/dd");
  const autoSignTime = Utilities.formatDate(now, "JST", "HH:mm:ss"); // 🆕 ファイル名の衝突防止用
  const uniqueKeyBase = Utilities.formatDate(now, "JST", "yyyyMMdd_HHmmss");
  const token = ScriptApp.getOAuthToken();
  const linesPerPage = 5;

  locations.forEach(function(fromLocation) {
    const depotCode = getDepotCode_(fromLocation);
    let seq = getMaxToolSeqForDepotCode_(depotCode);
    locationGroup[fromLocation].lines.forEach(function(line) {
      // 🆕 用具番号の先頭に「依頼部門の略語」を付け、番号を見ただけでどの部門の依頼か分かるようにする
      const deptAbbrev = getDeptAbbrev_(line.dept) || String(line.dept || "").slice(0, 1);
      if (line.suffixMode === "none") {
        seq++;
        line.code = `${deptAbbrev}${depotCode}${seq}`;
      } else if (line.suffixMode === "letter") {
        // 同一商品の最初のユニットでseqを進め、以降の枝番は同じseqを使う
        if (line.suffixIndex === 0) { seq++; }
        line.code = `${deptAbbrev}${depotCode}${seq}${String.fromCharCode(65 + line.suffixIndex)}`;
      } else { // star
        seq++;
        line.code = `${deptAbbrev}${depotCode}${seq}*`;
      }
    });

    // inspectionRows へ変換(17列): 拠点,物品コード,物品名,予定数量,検品済数量,ステータス,
    //   最終更新時刻,担当TM,依頼部門,依頼番号,配送先,ライフサイクル,返却日,返却依頼番号,備考,内部依頼ID,依頼メモ
    locationGroup[fromLocation].lines.forEach(function(line) {
      inspectionRows.push([
        fromLocation, line.code, line.desc, line.plan, 0, "未完了", "",
        determineTeam_(fromLocation), line.dept, line.reqRef,
        line.lenderDest, "未処理", "", "", "", line.internalId, line.requestNote || ""
      ]);
    });
  });

  // 🆕 既存の同一依頼(内部依頼ID)の検品行は、まだ現場で処理されていない(未処理のままの)行だけ
  //    削除して作り直す。既に配送検品済などの進捗がある行は、取消の場合と同じ考え方で、
  //    削除せずに「差し替え」の印を備考に残す(現物のQRラベルは既に貼られている可能性があり、
  //    追跡情報を失うと現物を回収し忘れるリスクがあるため)。
  let supersededCount = 0;
  if (inspectionRows.length > 0) {
    let inspectSheet = ss.getSheetByName("CO30_Inspection");
    if (!inspectSheet) {
      inspectSheet = ss.insertSheet("CO30_Inspection");
    }
    ensureDashboardHeaders_(inspectSheet);

    const idSet = {};
    inspectionRows.forEach(function(r) { if (r[15]) idSet[r[15]] = true; });
    const vals = inspectSheet.getDataRange().getValues();
    for (let r = vals.length - 1; r >= 1; r--) {
      const rid = vals[r][15] ? String(vals[r][15]).trim() : "";
      if (!rid || !idSet[rid]) continue;
      const lifecycle = vals[r][11] ? String(vals[r][11]).trim() : "未処理";
      if (lifecycle === "未処理") {
        inspectSheet.deleteRow(r + 1);
      } else {
        const currentRemarks = vals[r][14] ? String(vals[r][14]).trim() : "";
        if (currentRemarks.indexOf("⚠️内容修正により差し替え") === -1) {
          const newRemarks = currentRemarks ? `⚠️内容修正により差し替え / ${currentRemarks}` : "⚠️内容修正により差し替え";
          inspectSheet.getRange(r + 1, 15).setValue(newRemarks);
          supersededCount++;
        }
      }
    }
    inspectSheet.getRange(inspectSheet.getLastRow() + 1, 1, inspectionRows.length, 17).setValues(inspectionRows);
  }

  locations.forEach(function(fromLocation, locIdx) {
    let group = locationGroup[fromLocation];
    let flattenedLines = group.lines;
    let allRemarks = group.remarks;
    if (flattenedLines.length === 0) return;

    let ownerInfo = ownerMap[fromLocation] || { phone: "", address: "" };

    // 🆕 1ページ目はCO30_Template(5件まで)。それを超える分は、別紙テンプレートがあれば
    //    CO30_別紙_Template(49件/ページ)、無ければ従来通りCO30_Templateの繰り返し(5件/ページ)を使う。
    //
    // ⚠️ メインページと別紙は「列の使い方(幅)」が根本的に異なるため、1つのシートに結合すると
    //    列幅がどちらかに引っ張られて崩れる。そのため、メイン用・別紙用で PDF ファイルを分けて生成する。
    const mainCount = Math.min(flattenedLines.length, linesPerPage);
    const remainingCount = flattenedLines.length - mainCount;
    const useSuppl = remainingCount > 0 && !!templateSheetSuppl;
    const remainderCapacity = useSuppl ? CO30_SUPPL_CAPACITY : linesPerPage;
    const remainderPages = remainingCount > 0 ? Math.ceil(remainingCount / remainderCapacity) : 0;
    const totalPages = 1 + remainderPages;

    // ---- メインページ用の結合シート(1ページ目、および別紙テンプレートが無い場合の2ページ目以降) ----
    const mainPagesCount = useSuppl ? 1 : totalPages;
    let printMainSheetName = `PRINT_CO30_LOC_${locIdx}_${uniqueKeyBase}`;
    let printMainSheet = ss.insertSheet(printMainSheetName);
    if (printMainSheet.getMaxColumns() < 40) {
      printMainSheet.insertColumnsAfter(printMainSheet.getMaxColumns(), 40 - printMainSheet.getMaxColumns());
    }
    for (let col = 1; col <= 40; col++) {
      printMainSheet.setColumnWidth(col, templateSheet30.getColumnWidth(col));
    }

    // ---- 別紙用の結合シート(必要な場合のみ作成。メインとは別の列幅を使う) ----
    let printSupplSheet = null;
    if (useSuppl) {
      printSupplSheet = ss.insertSheet(`PRINT_CO30SUPPL_LOC_${locIdx}_${uniqueKeyBase}`);
      const supplTemplateCols = templateSheetSuppl.getMaxColumns();
      if (printSupplSheet.getMaxColumns() < supplTemplateCols) {
        printSupplSheet.insertColumnsAfter(printSupplSheet.getMaxColumns(), supplTemplateCols - printSupplSheet.getMaxColumns());
      }
      for (let col = 1; col <= supplTemplateCols; col++) {
        printSupplSheet.setColumnWidth(col, templateSheetSuppl.getColumnWidth(col));
      }
    }

    let mainPageNumber = null;    // 🆕 1ページ目の番号を控えておき、別紙にも同じ番号を入れる
    let mainCursorRow = 1;        // メイン結合シートの積み上げ位置
    let supplCursorRow = 1;       // 別紙結合シートの積み上げ位置

    for (let p = 0; p < totalPages; p++) {
      const isLastPage = (p === totalPages - 1);

      if (p === 0) {
        // ---- 1ページ目: CO30_Template(従来通り) ----
        let tempSheet30 = templateSheet30.copyTo(ss);
        tempSheet30.setName(`TEMP_CO30_Loc${locIdx}_P${p + 1}_${uniqueKeyBase}`);
        if (tempSheet30.getMaxColumns() < 40) {
          tempSheet30.insertColumnsAfter(tempSheet30.getMaxColumns(), 40 - tempSheet30.getMaxColumns());
        }

        mainPageNumber = nextCO30PageNumber_(fromLocation); // 拠点ごと・ページごとの通し番号(永続)

        setCell_(tempSheet30, CO30_NUMBER_CELL, mainPageNumber, "@");
        tempSheet30.getRange("D3").setNumberFormat("@").setValue(autoSignDate);
        tempSheet30.getRange("E5").setValue(fromLocation);
        tempSheet30.getRange("AF5").setNumberFormat("@").setValue(ownerInfo.phone);
        tempSheet30.getRange("D7").setValue(ownerInfo.address);
        tempSheet30.getRange("D22").setValue(allRemarks.join("\n") || "特になし");
        tempSheet30.getRange("T31").setNumberFormat("@").setValue(`${p + 1} / ${totalPages}`); // 🆕 総ページ数は別紙込み

        for (let idx = 0; idx < mainCount; idx++) {
          let lineData = flattenedLines[idx];
          let rowNum = CO30_ITEM_ROWS[idx];
          setCell_(tempSheet30, `${CO30_COL_TOOLNO}${rowNum}`, lineData.code);
          setCell_(tempSheet30, `${CO30_COL_DESC}${rowNum}`, lineData.desc);
          setCell_(tempSheet30, `${CO30_COL_LENDER}${rowNum}`, lineData.lenderDest || "");
          setCell_(tempSheet30, `${CO30_COL_REQREF}${rowNum}`, lineData.reqRef || "");
        }

        SpreadsheetApp.flush();
        const mainIsLastOfItsPdf = useSuppl ? true : isLastPage; // メインPDFとしてはここで最後(別紙は別PDF)
        for (let r = 1; r <= 31; r++) {
          printMainSheet.setRowHeight(mainCursorRow + r - 1, templateSheet30.getRowHeight(r));
        }
        if (!mainIsLastOfItsPdf) {
          printMainSheet.setRowHeight(mainCursorRow + 31, 20);
          printMainSheet.setRowHeight(mainCursorRow + 32, 20);
        }
        tempSheet30.getRange("A1:AN31").copyTo(printMainSheet.getRange(mainCursorRow, 1, 31, 40));

        let lastInternalId = null; // 🆕 このページ内で、同じ依頼の2行目以降は備考を空欄にする(部門の1行目だけに表示)

        SpreadsheetApp.flush();
        const supplSpan = CO30_SUPPL_PRINT_LAST_ROW; // 🆕 C49(ページ数表示)を含めるため49行分を1ページとして扱う

        try { ss.deleteSheet(tempSheetSuppl); } catch (e) {}
        supplCursorRow += isLastPage ? supplSpan : (supplSpan + 2);

      } else {
        // ---- 2ページ目以降(別紙テンプレートが見つからない場合のフォールバック): 従来通りCO30_Templateを繰り返す ----
        let tempSheet30 = templateSheet30.copyTo(ss);

        const startIdx = mainCount + (p - 1) * linesPerPage;
        const endIdx = Math.min(startIdx + linesPerPage, flattenedLines.length);
        for (let idx = startIdx; idx < endIdx; idx++) {
          let lineData = flattenedLines[idx];

          setCell_(tempSheet30, `${CO30_COL_TOOLNO}${rowNum}`, lineData.code);

          setCell_(tempSheet30, `${CO30_COL_REQREF}${rowNum}`, lineData.reqRef || "");

        for (let r = 1; r <= 31; r++) {
          printMainSheet.setRowHeight(mainCursorRow + r - 1, templateSheet30.getRowHeight(r));
        }
        if (!isLastPage) {
          printMainSheet.setRowHeight(mainCursorRow + 31, 20);
          printMainSheet.setRowHeight(mainCursorRow + 32, 20);
        }
        tempSheet30.getRange("A1:AN31").copyTo(printMainSheet.getRange(mainCursorRow, 1, 31, 40));
        try { ss.deleteSheet(tempSheet30); } catch (e) {}
        mainCursorRow += isLastPage ? 31 : 33;
      }
    }

    SpreadsheetApp.flush();

    // ---- メインPDFを書き出す ----
    const mainRowsCount = mainCursorRow - 1;
    const urlMain = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printMainSheet.getSheetId()}&r1=0&c1=0&r2=${mainRowsCount}&c2=39`;
    const blobMain = UrlFetchApp.fetch(urlMain, { headers: { 'Authorization': `Bearer ${token}` } })
      .getBlob().setName(`CO-30_用具受取書_[${fromLocation}]_${autoSignDate} ${autoSignTime}(全${mainPagesCount}頁).pdf`);
    try { ss.deleteSheet(printMainSheet); } catch (e) {}

    // ---- 別紙PDFを書き出す(ある場合のみ) ----
    let blobSuppl = null;
    if (useSuppl) {
      const supplRowsCount = supplCursorRow - 1;
      const supplColsForExport = 5; // 🆕 用具番号/説明/貸出先/備考/依頼番号の5列固定。getLastColumn()は環境によって不安定なため直接指定する
      const urlSuppl = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printSupplSheet.getSheetId()}&r1=0&c1=0&r2=${supplRowsCount}&c2=${supplColsForExport}`;

function getInspectionData(locationName, teamFilter) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("CO30_Inspection");
  if (!sheet) return [];

  const values = sheet.getDataRange().getValues();
  const items = [];

  for (let i = 1; i < values.length; i++) {
    if (locationName && values[i][0] !== locationName) continue;

    const itemTeam = values[i][7] ? String(values[i][7]).trim() : "";
    if (teamFilter && teamFilter !== "両方" && itemTeam && itemTeam !== teamFilter) continue;

    const updatedAtRaw = values[i][6];
    let updatedAt = 0;
    if (updatedAtRaw instanceof Date) {
      updatedAt = updatedAtRaw.getTime();
    } else if (updatedAtRaw) {
      const parsed = new Date(updatedAtRaw);
      updatedAt = isNaN(parsed.getTime()) ? 0 : parsed.getTime();
    }

    const remarks = values[i][14] ? String(values[i][14]).trim() : "";
    const requestNote = values[i][16] ? String(values[i][16]).trim() : ""; // 🆕 依頼メモ(CO31の特記事項)

    items.push({
      rowId: i + 1,
      location: values[i][0] ? String(values[i][0]).trim() : "",
      code: String(values[i][1]).trim(),
      name: String(values[i][2]).trim(),
      plan: parseInt(values[i][3], 10) || 0,
      actual: parseInt(values[i][4], 10) || 0,
      updatedAt: updatedAt,
      team: itemTeam,
      dept: values[i][8] ? String(values[i][8]).trim() : "",
      reqNo: values[i][9] ? String(values[i][9]).trim() : "",
      lifecycle: values[i][11] ? String(values[i][11]).trim() : "未処理",
      remarks: remarks,
      requestNote: requestNote,
      cancelled: remarks.indexOf("⚠️取消済み") !== -1 // 🆕 QRラベル・スキャナー側で警告表示するためのフラグ
    });
  }
  return items;
}

function updateInspectionQty(locationName, itemCode, newQty, rowId) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("CO30_Inspection");
  if (!sheet) return { success: false, error: "検品シートが見つかりません" };

  if (rowId && rowId >= 2) {
    const rowValues = sheet.getRange(rowId, 1, 1, 6).getValues()[0];
    const sheetCode = String(rowValues[1]).trim();
    const sheetLocation = rowValues[0];
    const locationMatches = !locationName || sheetLocation === locationName;

    if (sheetCode === itemCode && locationMatches) {
      const planQty = rowValues[3];
      const now = new Date();
      sheet.getRange(rowId, 5).setValue(newQty);
      sheet.getRange(rowId, 6).setValue(newQty >= planQty ? "完了" : "未完了");
      sheet.getRange(rowId, 7).setValue(now);
      return { success: true, updatedAt: now.getTime(), rowId: rowId };
    }
  }

  const values = sheet.getDataRange().getValues();
  for (let i = 1; i < values.length; i++) {
    const locationMatches = !locationName || values[i][0] === locationName;
    if (locationMatches && String(values[i][1]).trim() === itemCode) {
      const row = i + 1;
      const planQty = values[i][3];
      const now = new Date();
      sheet.getRange(row, 5).setValue(newQty);
      sheet.getRange(row, 6).setValue(newQty >= planQty ? "完了" : "未完了");
      sheet.getRange(row, 7).setValue(now);
      return { success: true, updatedAt: now.getTime(), rowId: row };
    }
  }
  return { success: false, error: "該当する物品コードが見つかりません" };
}

/**
 * 拠点別・備品別表示(親子チェック)からの一括更新用。
 * rowIds(非連続可)すべてに、指定フィールド(lifecycle/remarks)とG列=最終更新時刻をセットする。
 */
function updateInspectionFieldBatch(rowIds, field, value) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("CO30_Inspection");
  if (!sheet) return { success: false, error: "シートが見つかりません" };

  const fieldToCol = { lifecycle: 12, remarks: 15 };
  const col = fieldToCol[field];
  if (!col) return { success: false, error: "不明なフィールド: " + field };

  const ids = (rowIds || [])
    .map(function(r) { return parseInt(r, 10); })
    .filter(function(r) { return !isNaN(r) && r >= 2; });
  if (ids.length === 0) return { success: true, updatedRows: [] };

  const lock = LockService.getScriptLock();
  const gotLock = lock.tryLock(10 * 1000);
  if (!gotLock) return { success: false, error: "他の処理が実行中です。少し待ってからもう一度お試しください。" };
  try {
    ensureDashboardHeaders_(sheet);
    const now = new Date();
    const lastRow = sheet.getLastRow();
    ids.forEach(function(rowId) {
      if (rowId > lastRow) return;
      sheet.getRange(rowId, col).setValue(value);
      sheet.getRange(rowId, 7).setValue(now);
    });
    return { success: true, updatedRows: ids };
  } catch (err) {
    return { success: false, error: err.message };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 🆕 ダッシュボード用: CO-30がまだ作成されていない依頼(承認待ち・CO30作成待ち)を一覧で返す。
 * 用具番号はまだ存在しないため含まない(拠点ごとに一括採番されるのはCO30作成時のため)。
 * 送信完了・取消の依頼は含めない。
 */
function listPendingRequests() {
  // 🆕 親依頼ID単位で集約(件数は「行数」ではなく「依頼数」になる)。
  //    グループ内のいずれかの行が承認待ち/CO30作成待ちなら1件として含める。
  const out = [];
  collectCO31Groups_().forEach(function(g) {
    const hasWait = g.statuses.indexOf("承認待ち") !== -1;
    const hasReady = g.statuses.indexOf("CO30作成待ち") !== -1;
    if (!hasWait && !hasReady) return;
    out.push({
      dept: g.dept,
      fromLocation: g.fromLocation,
      toLocation: g.toLocation,
      status: hasWait ? "承認待ち" : "CO30作成待ち",
      reqNo: g.reqNos.join("、"),
      rows: g.rows,
      items: g.items.map(function(it) { return { name: it.name, qty: it.qty || "1" }; }),
      timestamp: g.timestamp
    });
  });
  out.sort(function(a, b) { return b.timestamp - a.timestamp; });
  return out;
}

    const st = sheet.getRange(2, 30, sheet.getLastRow() - 1, 1).getValues();

    }
    return false;
  } catch (e) { return true; } // 判定できない時は念のため本処理へ
}

/**
 * 🆕 ステータスが「承認」のまま残っている依頼を、親依頼IDごとに処理する。
 * ・新しいグループの着手は開始から2.5分まで(残りは次回へ持ち越し)
 * ・2回目以降の再挑戦は1ページずつに分割し、確実に完了させる
 * ・3回連続で完了しない場合のみ打ち切り、運送・用具部門へ通知する
 */
function processApprovalQueue_() {
  deleteTriggersByName_(APPROVAL_WORKER_FN_);

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30 * 1000)) {
    scheduleApprovalWorker_(60 * 1000);
    return { busy: true, processed: 0 };
  }

  const started = Date.now();
  const NEW_GROUP_LIMIT_MS = 150 * 1000; // 2.5分: 最大4ページ(PDF8回)が入る余白を残す
  const SPLIT_ROW_LIMIT_MS = 240 * 1000; // 4分: 分割処理で次の1ページに着手してよい上限
  let remaining = false, processed = 0, failed = 0;

  try {
    try { cleanupCO31TempSheets_(); } catch (e) {}

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("CO31_Responses");
    if (!sheet || sheet.getLastRow() < 2) return { processed: 0 };

    const props = PropertiesService.getScriptProperties();
    const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, CO31_TOTAL_COLS).getValues();

    const groups = {};
    const order = [];
    for (let i = 0; i < values.length; i++) {
      if (String(values[i][29] || "").trim() !== "承認") continue;
      const rowNum = i + 2;
      const gid = String(values[i][CO31_COL_GROUPID_IDX] || "").trim() || ("r" + rowNum);
      if (!groups[gid]) {
        groups[gid] = { rows: [], iid: String(values[i][1] || "").trim() || ("r" + rowNum) };
        order.push(gid);
      }
      groups[gid].rows.push(rowNum);
    }
    if (order.length === 0) return { processed: 0 };

    for (let g = 0; g < order.length; g++) {
      if (Date.now() - started > NEW_GROUP_LIMIT_MS) { remaining = true; break; }

      const rows = groups[order[g]].rows;
      const iid = groups[order[g]].iid;
      const failKey = "apprfail_" + iid;
      const notifiedKey = "apprnotified_" + iid;
      const attempts = parseInt(props.getProperty(failKey) || "0", 10);

      if (attempts >= APPROVAL_MAX_ATTEMPTS_) {
        if (props.getProperty(notifiedKey) !== "1") {
          notifyApprovalFailure_(rows, `処理が${APPROVAL_MAX_ATTEMPTS_}回連続で完了しませんでした(実行時間超過の可能性があります)。`);
          props.setProperty(notifiedKey, "1");
        }
        continue;
      }

      const approverName = props.getProperty("apprby_id_" + iid) || "運送部門担当者";
      try {
        props.setProperty(failKey, String(attempts + 1)); // 先に加算(ハードタイムアウトでも記録が残る)

        if (attempts >= 1 && rows.length > 1) {
          // 🆕 2回目以降は1ページずつ処理して確実に完了させる
          for (let k = 0; k < rows.length; k++) {
            if (Date.now() - started > SPLIT_ROW_LIMIT_MS) { remaining = true; break; }
            sendGroupCO31_(sheet, [rows[k]], approverName);
          }
        } else {
          sendGroupCO31_(sheet, rows, approverName);
        }

        props.deleteProperty(failKey);
        props.deleteProperty(notifiedKey);
        props.deleteProperty("apprby_id_" + iid);
        processed++;
      } catch (err) {
        failed++;
        console.error("承認キューの処理に失敗(行" + rows.join(",") + "): " + err.message);
        if (attempts + 1 >= APPROVAL_MAX_ATTEMPTS_ && props.getProperty(notifiedKey) !== "1") {
          notifyApprovalFailure_(rows, err.message);
          props.setProperty(notifiedKey, "1");
        }
      }
    }
    return { processed: processed, failed: failed, remaining: remaining };
  } finally {
    lock.releaseLock();
    if (remaining) scheduleApprovalWorker_(10 * 1000);
  }
}

/**
 * 🆕 複数の内部依頼IDについて、CO30_Inspectionの備考に「取消済み」の印をまとめて付ける。
 * 1セルずつsetValueすると遅いため、列を丸ごと読んで一括で書き戻す。
 */
function markCO30ItemsAsCancelledBatch_(internalIds) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO30_Inspection");
  if (!sheet || sheet.getLastRow() < 2) return 0;

  const set = {};
  (internalIds || []).forEach(function(id) {
    const s = String(id || "").trim();
    if (s) set[s] = true;
  });
  if (Object.keys(set).length === 0) return 0;

  const n = sheet.getLastRow() - 1;
  const ids = sheet.getRange(2, 16, n, 1).getValues();      // P列=内部依頼ID
  const remarks = sheet.getRange(2, 15, n, 1).getValues();  // O列=備考

  let count = 0, changed = false;
  for (let i = 0; i < n; i++) {
    const rid = ids[i][0] ? String(ids[i][0]).trim() : "";
    if (!rid || !set[rid]) continue;
    const cur = remarks[i][0] ? String(remarks[i][0]).trim() : "";
    if (cur.indexOf("⚠️取消済み") !== -1) continue;
    remarks[i][0] = cur ? "⚠️取消済み / " + cur : "⚠️取消済み";
    count++; changed = true;
  }
  if (changed) sheet.getRange(2, 15, n, 1).setValues(remarks);
  return count;
}

function processCancelQueue_() {
  deleteTriggersByName_(CANCEL_WORKER_FN_);

  const props = PropertiesService.getScriptProperties();
  const keys = props.getKeys().filter(function(k) { return k.indexOf("cancjob_") === 0; }).sort();
  if (keys.length === 0) return { processed: 0 };

  const started = Date.now();
  const jobs = [];
  const allRows = [];
  const allIds = [];
  const allNos = [];

  for (let i = 0; i < keys.length; i++) {
    let job = null;
    try { job = JSON.parse(props.getProperty(keys[i])); } catch (e) {}
    props.deleteProperty(keys[i]);
    if (!job) continue;
    jobs.push(job);
    (job.rows || []).forEach(function(r) { allRows.push(r); });
    (job.internalIds || []).forEach(function(v) { allIds.push(v); });
    (job.reqNos || []).forEach(function(v) { if (allNos.indexOf(v) === -1) allNos.push(v); });
  }
  if (jobs.length === 0) return { processed: 0 };

  try { invalidateTokensForRows_(allRows); } catch (e) {}

  let markedTotal = 0;
  try { markedTotal = markCO30ItemsAsCancelledBatch_(allIds); } catch (e) {}

  try {
    if (allNos.length > 0) {
      const folder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
      const res = allNos.map(function(no) {
        const esc = String(no).replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        return new RegExp("No" + esc + "(?!\\d)");
      });
      const iter = folder.getFiles();
      while (iter.hasNext()) {
        if (Date.now() - started > 4 * 60 * 1000) break;
        const f = iter.next();
        const name = f.getName();
        if (res.some(function(re) { return re.test(name); })) {
          try { f.setTrashed(true); } catch (e) {}
        }
      }
    }
  } catch (e) {
    console.error("取消PDFの削除に失敗: " + e.message);
  }

  jobs.forEach(function(job) {
    try {
      const truckingEmail = "2026trucking@gmail.com";
      const dept = job.dept || "不明部門";
      const deptEmail = getDeptEmail_(dept);
      const ccList = [];
      if (deptEmail && deptEmail !== truckingEmail) ccList.push(deptEmail);
      if (job.contactEmail && job.contactEmail.indexOf("@") !== -1
          && job.contactEmail !== truckingEmail && ccList.indexOf(job.contactEmail) === -1) {
        ccList.push(job.contactEmail);
      }

      const statuses = job.oldStatuses || [];
      const wasApproved = statuses.some(function(s) {
        return s === "承認" || s === "CO30作成待ち" || s === "送信完了";
      });
      const urgencyNote = wasApproved
        ? "⚠️ この依頼は既に承認済み(または送信済み)でした。運送・用具部門は準備を進めている場合、作業を中止してください。\n⚠️ 依頼部門・運送・用具部門とも、既にお手元にあるこの依頼のCO-31(控えメールの添付PDF・印刷済みの紙)は、今後の手続きには使用せず破棄してください。"
        : "この依頼はまだ承認前でした。";
      const co30Note = markedTotal > 0
        ? `\n\n※既にCO-30が作成されており、${markedTotal}件の物品に「取消済み」の印を付けました。現場で使用しないようご確認ください。`
        : "";
      const reqNo = (job.reqNos || []).join("、");

      const mailOpts = {
        to: truckingEmail,
        subject: `【取消】${dept}の配送依頼が取り消されました${reqNo ? "(No." + reqNo + ")" : ""}`,
        body: `運送・用具部門 ご担当者様\n${dept} ご担当者様\n\n以下のCO-31配送依頼が取り消されましたのでお知らせします。\n\n` +
              `依頼番号: ${reqNo || "未採番(承認前)"}\n依頼部門: ${dept}\n配送元　: ${job.fromLoc || ""}\n` +
              `配送先　: ${job.toLoc || ""}\n担当者　: ${job.contactInfo || "未記入"}\n` +
              `取消前のステータス: ${aggregateGroupStatus_(statuses) || "不明"}\n\n${urgencyNote}${co30Note}`
      };
      if (ccList.length > 0) mailOpts.cc = ccList.join(",");
      sendMail_(mailOpts);
    } catch (e) {
      console.error("取消通知メールの送信に失敗: " + e.message);
    }
  });

  return { processed: jobs.length };
}
