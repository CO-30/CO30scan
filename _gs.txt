 *
 * 🆕 今回(4回目)の大規模改修:
 *   A. 番号体系を全面刷新
 *      - CO31_Responses B列: 内部依頼ID(全体通し番号。修正時も不変。CO30_Inspectionとの紐付け専用)
 *      - CO31_Responses 末尾列: 表示用ページ番号(部門ごと・PDF1枚ごとに1から連番。承認のたびに再計算)

 *      - CO30テンプレート AF3(結合 AF3:AK3) に「拠点ごと・PDF1枚ごとの通し番号」を印字(スクリプトプロパティで永続化)

// すべての自動メールはこのラッパー経由で送る（name/replyTo を一括付与）。
// ※内部は MailApp["sendEmail"] とブラケット記法。これにより手順2の一括置換で
//   ラッパー自身が置換されず、自己再帰（無限ループ）を防ぐ。
function sendMail_(recipientOrOptions, subject, body, options) {
  const opts = (typeof recipientOrOptions === 'string')
    ? Object.assign({ to: recipientOrOptions, subject: subject, body: body }, options || {})
    : Object.assign({}, recipientOrOptions);   // 既存の to/cc/subject/body/attachments はそのまま保持
  if (!opts.name)    opts.name = MAIL_SENDER_NAME;      // 呼び出し側で個別指定があればそちら優先
  if (!opts.replyTo) opts.replyTo = MAIL_REPLY_TO;
  MailApp["sendEmail"](opts);
}

function determineTeam_(fromLocation) {
  const loc = String(fromLocation || "").trim();
  if (!loc) return "運送";

  // 🆕 Item_MasterのH列(拠点名)・J列(担当TM)で拠点ごとに担当TMを指定できるようにする。
  //    同じ拠点が複数行ある場合は、最初に見つかった行の値を採用する。
  //    空欄の拠点は従来どおり「運送」扱い。拠点が増えてもコード変更なしで対応できる。
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Item_Master");
    if (sheet) {
      const values = sheet.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][7] || "").trim() !== loc) continue; // H列=拠点名
        const team = String(values[i][9] || "").trim();          // J列=担当TM
        if (team === "レンタル" || team === "運送") return team;
      }
    }
  } catch (e) {}

  // マスタに指定が無い場合のフォールバック(旧仕様の互換)
  if (loc === "コンベンションセンター") return "レンタル";
  return "運送";
}

// =========================================================
// 結合セル対応の安全な書き込みヘルパー
// =========================================================

function setCell_(sheet, a1, value, numberFormat) {
  const range = sheet.getRange(a1);
  const merged = range.getMergedRanges();
  const target = (merged && merged.length > 0) ? merged[0] : range;
  if (numberFormat) target.setNumberFormat(numberFormat);
  target.setValue(value);
}

/** 🆕 setCell_と同じだが、highlightがtrueならセル背景を黄色にする(修正箇所の可視化用) */
function setCellHL_(sheet, a1, value, highlight, numberFormat) {
  const range = sheet.getRange(a1);
  const merged = range.getMergedRanges();
  const target = (merged && merged.length > 0) ? merged[0] : range;
  if (numberFormat) target.setNumberFormat(numberFormat);
  target.setValue(value);
  target.setBackground(highlight ? "#fff3b0" : null);
}

function doGet(e) {
  if (e && e.parameter && e.parameter.page === 'dashboard') {
    return HtmlService.createHtmlOutputFromFile('dashboard')
      .setTitle('CO-30 監督者ダッシュボード')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
  }
  if (e && e.parameter && e.parameter.action === 'approvalPage') {
    const userEmail = Session.getActiveUser().getEmail();
    return showApprovalPage_(e.parameter.token, userEmail);
  }
  if (e && e.parameter && e.parameter.action) {
    return handleApiRequest_(e.parameter.action, e.parameter);
  }
  // 🆕 メール内の「修正リンク」から開かれた場合、mode/dept/reqNoをテンプレートに埋め込んで渡す
  // (Apps ScriptのWebアプリはiframe化されるため、クライアント側でURLパラメータを直接読めないことがある)
  const tmpl = HtmlService.createTemplateFromFile('index');
  tmpl.initialMode  = (e && e.parameter && e.parameter.mode)  ? e.parameter.mode  : '';
  tmpl.initialDept  = (e && e.parameter && e.parameter.dept)  ? e.parameter.dept  : '';
  tmpl.initialReqNo = (e && e.parameter && e.parameter.reqNo) ? e.parameter.reqNo : '';
  return tmpl.evaluate()
      .setTitle('CO-31 大会配送依頼 Webフォーム')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// doPost: 承認フォーム送信 / GitHub Pages API
// =========================================================

function doPost(e) {
  if (e.parameter && e.parameter.action === 'doApprove') {
    // 🆕 Session.getActiveUser()は組織外アカウントで空になることがあるため、
    //    承認ページでプルダウン選択された値を優先し、無ければセッションのメールにフォールバックする。
    const selectedEmail = e.parameter.approverEmail || "";
    const sessionEmail = Session.getActiveUser().getEmail();
    const userEmail = selectedEmail || sessionEmail;
    return processApprovalByToken_(e.parameter.token, userEmail);
  }
  try {
    const body = JSON.parse(e.postData.contents);
    return handleApiRequest_(body.action, body.payload || {});
  } catch (err) {
    return jsonResponse_({ success: false, error: "リクエストの解析に失敗: " + err.message });
  }
}

function handleApiRequest_(action, params) {
  try {
    if (action === 'getInspectionData') {
      const displayItems = getInspectionData(params.location || "", params.team || "");
      const allItems = getInspectionData(params.location || "", "");
      return jsonResponse_({ display: displayItems, all: allItems });
    }
    if (action === 'listRequests') {
      return jsonResponse_({ success: true, requests: listRequestsForScanner() });
    }
    if (action === 'listPrintFolderFiles') {
      const result = listPrintFolderFiles_();
      return jsonResponse_({ success: true, files: result.files, totalCount: result.totalCount });
    }
    if (action === 'updateInspectionQty') {
      const result = updateInspectionQty(
        params.location || "",
        params.itemCode,
        parseInt(params.newQty, 10),
        params.rowId ? parseInt(params.rowId, 10) : null
      );
      return jsonResponse_(result);
    }
    if (action === 'updateLifecycle') {
      // 🆕 5モード(未処理/配送検品済/配送済/回収検品済/返却済)いずれも同じAPIで処理
      const result = updateLifecycle(
        params.location || "",
        params.itemCode,
        params.lifecycle || "未処理"
      );
      return jsonResponse_(result);
    }
    if (action === 'updateFieldBatch') {
      // 🆕 拠点別・備品別ロット表示(親子チェック)からの一括更新用。ダッシュボードのupdateInspectionFieldBatchをJSON API経由で呼べるようにする中継。
      const rowIds = Array.isArray(params.rowIds) ? params.rowIds : String(params.rowIds || '').split(',').filter(Boolean);
      const result = updateInspectionFieldBatch(rowIds, params.field || '', params.value || '');
      return jsonResponse_(result);
    }
    if (action === 'updateRemarks') {
      // 🆕 スキャナー側(依頼一覧・スキャン履歴)から備考(どこから取ったか等)を更新する
      const result = updateInspectionRemarks_(params.location || "", params.itemCode, params.remarks || "");
      return jsonResponse_(result);
    }
    if (action === 'getRentalRemarkOptions') {
      // 🆕 Item_MasterのK列(レンタル備考候補)を返す
      return jsonResponse_({ success: true, options: getRentalRemarkOptions_() });
    }
    return jsonResponse_({ success: false, error: "不明なアクション: " + action });
  } catch (err) {
    return jsonResponse_({ success: false, error: err.message });
  }
}

function jsonResponse_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/** 🆕 印刷待ちフォルダ内のPDFを新しい順に返す(QRラベル印刷ページと統合した印刷センター用) */
/** 🆕 現在「取消」になっているCO31依頼の表示番号一覧を返す(印刷センターでの取消フラグ付け用) */
function getCancelledRequestNumbers_() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!sheet || sheet.getLastRow() < 2) return [];
  const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, CO31_TOTAL_COLS).getValues();
  const nums = [];
  values.forEach(function(r) {
    const status = String(r[29] || "").trim();
    if (status !== "取消") return;
    const reqNo = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    if (reqNo) nums.push(reqNo);
  });
  return nums;
}

function listPrintFolderFiles_() {
  try {
    const cancelledNumbers = getCancelledRequestNumbers_();
    const folder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
    const iter = folder.getFilesByType(MimeType.PDF);
    const files = [];
    while (iter.hasNext()) {
      const f = iter.next();
      const name = f.getName();
      const kind = name.indexOf("CO-30") === 0 || name.indexOf("CO30") !== -1 ? "CO30"
                 : (name.indexOf("CO31") !== -1 || name.indexOf("CO-31") !== -1) ? "CO31" : "その他";
      // 🆕 「No1」が「No10」等に誤マッチしないよう、後ろに数字が続かない場合だけ一致とみなす
      const cancelled = cancelledNumbers.some(function(num) {
        const esc = String(num).replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        return new RegExp("No" + esc + "(?!\\d)").test(name);
      });
      files.push({
        name: name,
        url: f.getUrl(),
        kind: kind,
        cancelled: cancelled,
        createdAt: f.getDateCreated().getTime()
      });
    }
    files.sort(function(a, b) { return b.createdAt - a.createdAt; });
    const CAP = 500; // 🆕 150→500に引き上げ
    return { files: files.slice(0, CAP), totalCount: files.length };
  } catch (e) {
    return { files: [], totalCount: 0 };
  }
}

    const qty  = rowArr[i]     ? String(rowArr[i]).trim()     : "";
    const name = rowArr[i + 1] ? String(rowArr[i + 1]).trim() : "";
    const size = rowArr[i + 2] ? String(rowArr[i + 2]).trim() : "";
    if (name || qty) items.push({ qty: qty, name: name, size: size });
  });
  for (let i = 33; i <= 75; i += 3) { // 物品6〜20
    const qty  = rowArr[i]     ? String(rowArr[i]).trim()     : "";
    const name = rowArr[i + 1] ? String(rowArr[i + 1]).trim() : "";
    const size = rowArr[i + 2] ? String(rowArr[i + 2]).trim() : "";
    if (name || qty) items.push({ qty: qty, name: name, size: size });
  }
  return items;
}

/** グループ内の行ステータスが混在している場合の表示用集約(未完了寄りを優先) */
function aggregateGroupStatus_(statuses) {
  const uniq = [];
  statuses.forEach(function(s) {
    if (s && uniq.indexOf(s) === -1) uniq.push(s);
  });
  if (uniq.length <= 1) return uniq[0] || "";
  const priority = ["承認待ち", "CO30作成待ち", "承認", "送信完了", "取消"];
  for (let i = 0; i < priority.length; i++) {
    if (uniq.indexOf(priority[i]) !== -1) return priority[i];
  }
  return uniq[0];
}

/**

function getDeptAbbrev_(deptName) {

/**
 * この部門で現在使われている最大の「基準番号」(末尾の -N 改訂サフィックスは除く)を調べる。
 */
function getMaxBaseNumberForDept_(sheet, deptName, excludeRow) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return 0;
  const values = sheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();
  let max = 0;
  for (let i = 0; i < values.length; i++) {
    const rowNum = i + 2;
    if (rowNum === excludeRow) continue;
    if (String(values[i][4] || "").trim() !== String(deptName).trim()) continue;
    const disp = String(values[i][CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    if (!disp) continue;
    const m = disp.match(/(\d+)/); // 略語の直後の数字部分=基準番号
    if (m) max = Math.max(max, parseInt(m[1], 10));
  }
  return max;
}

/**
 * 🆕 承認のたびに呼び出す。
 * 初回承認: 部門ごとに新しい基準番号を採番する。例: "設5"
 * 修正後の再承認: 既存の基準番号を維持したまま、末尾に改訂回数を付与する。
 *   例: "設5" → 1回目の修正で "設5-1" → 2回目の修正で "設5-2"
 * これにより、番号を見ただけで「修正版である」ことが一目で分かる。
 */
function assignCO31DisplayNumber_(sheet, row, deptName, isRevisionRequested) {
  const existing = String(sheet.getRange(row, CO31_COL_DISPLAYPAGE_IDX + 1).getValue() || "").trim();

  if (existing) {
    // 🆕 既に番号があっても、修正マークが無ければ「失敗した処理のやり直し」とみなし番号を据え置く。
    //    (これが無いと、再処理のたびに -1, -2 と改訂番号が進んでしまう)
    if (!isRevisionRequested) {
      return { display: existing, isRevision: false };
    }
    const m = existing.match(/^(.*?)(?:-(\d+))?$/);
    let base = existing, revision = 0;
    if (m) {
      base = m[1] || existing;
      revision = m[2] ? parseInt(m[2], 10) : 0;
    }
    const display = `${base}-${revision + 1}`;
    sheet.getRange(row, CO31_COL_DISPLAYPAGE_IDX + 1).setNumberFormat("@").setValue(display);
    return { display: display, isRevision: true };
  }

  const abbrev = getDeptAbbrev_(deptName) || String(deptName || "").trim().slice(0, 1);
  const maxSoFar = getMaxBaseNumberForDept_(sheet, deptName, row);
  const display = `${abbrev}${maxSoFar + 1}`;
  sheet.getRange(row, CO31_COL_DISPLAYPAGE_IDX + 1).setNumberFormat("@").setValue(display);
  return { display: display, isRevision: false };
}

/**
 * 🆕 修正(再承認)の際、旧データと新データを比較して変更されたフィールドを検出する。
 * 戻り値: { fields: ["fromLocation", ...], itemIndices: [0, 2, ...] }
 * fieldsは書式済みの項目名の配列、itemIndicesはcollectItems_後の物品リストの位置(0始まり)。
 */
function detectChangedFields_(oldRow, newRow) {
  const changed = { fields: [], itemIndices: [] };
  const scalarMap = {
    docDate: 2, fromLocation: 3, toLocation: 5, arrival: 6, cost: 7,
    totalNote: 24, approver: 25, contactInfo: 31, contactEmail: 32
  };
  Object.keys(scalarMap).forEach(function(key) {
    const idx = scalarMap[key];
    if (String(oldRow[idx] || "") !== String(newRow[idx] || "")) changed.fields.push(key);
  });

  const oldItems = collectItems_(oldRow);
  const newItems = collectItems_(newRow);
  const maxLen = Math.max(oldItems.length, newItems.length);
  for (let i = 0; i < maxLen; i++) {
    const o = oldItems[i] || { qty: "", name: "", size: "" };
    const n = newItems[i] || { qty: "", name: "", size: "" };
    if (o.qty !== n.qty || o.name !== n.name || o.size !== n.size) changed.itemIndices.push(i);
  }
  return changed;
}

/** 🆕 変更フィールドのキー→日本語ラベル */
const CO31_FIELD_LABELS_ = {
  docDate: "依頼日", fromLocation: "配送元", toLocation: "配送先(場所)",
  arrival: "搬入希望日時", cost: "費用", totalNote: "特記事項",
  approver: "責任者氏名", contactInfo: "担当者・連絡先", contactEmail: "担当者メール"
};

/**
 * 🆕 detectChangedFields_の結果を、メール本文に載せられる日本語の箇条書きサマリーに変換する。
 * newItemsには変更後(collectItems_後)の物品リストを渡す。
 */
function buildChangeSummary_(changedInfo, newItems) {
  const lines = [];
  (changedInfo.fields || []).forEach(function(key) {
    lines.push(`・${CO31_FIELD_LABELS_[key] || key}`);
  });
  if (changedInfo.itemIndices && changedInfo.itemIndices.length > 0) {
    const descs = changedInfo.itemIndices.map(function(idx) {
      const it = newItems[idx];
      return it && it.name ? `${idx + 1}品目目(${it.name})` : `${idx + 1}品目目(削除)`;
    });
    lines.push(`・物品: ${descs.join("、")}`);
  }
  return lines.length > 0 ? lines.join("\n") : "(変更箇所は検出されませんでした)";
}

// 🆕 拠点(Depot_Master)識別記号 と 用具番号の採番
// =========================================================

function getDepotCode_(depotName) {
  const name = String(depotName || "").trim();

  // 🆕 まずItem_MasterのH列(拠点名)・G列(識別記号)を見る。
  //    同じ拠点が複数行ある場合は、最初に見つかった行の値を採用する。
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Item_Master");
    if (sheet) {
      const values = sheet.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][7] || "").trim() !== name) continue; // H列=拠点名
        const code = String(values[i][6] || "").trim();           // G列=識別記号
        if (code) return code;
      }
    }
  } catch (e) {}

  // 従来のDepot_Master / Owner_MasterのC列も引き続き参照する(既存設定を無駄にしないため)
  const candidateSheets = ["Depot_Master", "Owner_Master"];
  for (let s = 0; s < candidateSheets.length; s++) {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(candidateSheets[s]);
    if (!sheet) continue;
    const values = sheet.getDataRange().getValues();
    for (let i = 1; i < values.length; i++) {
      if (String(values[i][0]).trim() === name) {
        const code = values[i][2] ? String(values[i][2]).trim() : "";
        if (code) return code;
      }
    }
  }

  // どこにも無ければ拠点名の先頭1文字を識別記号として使う
  return name.slice(0, 1) || "拠";
}

/** CO30_Inspectionに既にある物品コードから、この識別記号の最大連番を調べる(部門略語プレフィックスが前に付いても検出できるよう、先頭固定はしない) */
function getMaxToolSeqForDepotCode_(depotCode) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO30_Inspection");
  if (!sheet || sheet.getLastRow() < 2) return 0;
  const codes = sheet.getRange(2, 2, sheet.getLastRow() - 1, 1).getValues();
  const escaped = depotCode.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const re = new RegExp(escaped + "(\\d+)");
  let max = 0;
  codes.forEach(function(c) {
    const m = String(c[0] || "").match(re);
    if (m) max = Math.max(max, parseInt(m[1], 10));
  });
  return max;
}

/** CO30拠点ごとの通し番号(印刷用)。スクリプトプロパティで永続化し、バッチ生成をまたいで増え続ける。 */
function nextCO30PageNumber_(depotName) {
  const props = PropertiesService.getScriptProperties();
  const key = "co30pn_" + depotName;
  let cur = parseInt(props.getProperty(key) || "0", 10);
  cur += 1;
  props.setProperty(key, String(cur));
  return cur;
}

/**
 * 🆕 指定した拠点の「現在のCO30_Inspectionの内容」だけを使って、CO30 PDFを作り直す(再印刷)。
 * 用具番号・ライフサイクル・依頼番号など、既存のデータは一切変更しない(読み取るだけ)。
 * PDFは一度作ると内容が更新されない静止画のため、「番号や部門名が古いまま」といったズレに
 * 気づいたときに、最新の正しい内容で印刷用PDFだけを作り直すための機能。
 * ダッシュボードから google.script.run 経由で呼ばれる。
 */
function reprintCO30ForLocation(fromLocation) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const templateSheet30 = ss.getSheetByName("CO30_Template");
    // 🆕 別紙テンプレート。exportAllCO30Combinedと同じルールで、あれば49件/ページの別紙を使う。
    const templateSheetSuppl = ss.getSheetByName(CO30_SUPPL_SHEET_NAME);
    const inspSheet = ss.getSheetByName("CO30_Inspection");
    if (!templateSheet30 || !inspSheet) {
      return { success: false, message: "必要なシートが見つかりません。" };
    }

    // 現在のCO30_Inspectionから、この拠点の行をそのまま読み取る(番号などは一切再計算しない)
    const values = inspSheet.getDataRange().getValues();
    const flattenedLines = [];
    const remarksSet = {};
    for (let i = 1; i < values.length; i++) {
      const row = values[i];
      if (String(row[0] || "").trim() !== fromLocation) continue;
      const code = row[1] ? String(row[1]).trim() : "";
      if (!code) continue;
      const dept = row[8] ? String(row[8]).trim() : "";
      const reqNo = row[9] ? String(row[9]).trim() : "";
      const reqRef = reqNo;
      const internalId = row[15] ? String(row[15]).trim() : ""; // 🆕 別紙の備考重複表示防止に使用
      const requestNote = row[16] ? String(row[16]).trim() : "";
      if (requestNote && dept) remarksSet[`【${dept}】${requestNote}`] = true;
      flattenedLines.push({
        code: code,
        desc: row[2] ? String(row[2]).trim() : "",
        lenderDest: row[10] ? String(row[10]).trim() : "",
        reqRef: reqRef,
        internalId: internalId,       // 🆕
        requestNote: requestNote      // 🆕
      });
    }

    if (flattenedLines.length === 0) {
      return { success: false, message: `拠点「${fromLocation}」の物品がCO30_Inspectionに見つかりませんでした。` };
    }

    // 所有者情報(電話・住所)
    const ownerSheetCandidates = ["Depot_Master", "Owner_Master"];
    let ownerInfo = { phone: "", address: "" };
    for (let s = 0; s < ownerSheetCandidates.length; s++) {
      const sheet = ss.getSheetByName(ownerSheetCandidates[s]);
      if (!sheet) continue;
      const oData = sheet.getDataRange().getValues();
      for (let i = 1; i < oData.length; i++) {
        if (String(oData[i][0]).trim() === fromLocation) {
          ownerInfo = { phone: oData[i][1] || "", address: oData[i][2] || "" };
          break;
        }
      }
    }

    const truckingEmail = "2026trucking@gmail.com";
    const token = ScriptApp.getOAuthToken();
    const now = new Date();
    const autoSignDate = Utilities.formatDate(now, "JST", "yyyy/MM/dd");
    const autoSignTime = Utilities.formatDate(now, "JST", "HH:mm:ss");
    const uniqueKeyBase = "REPRINT_" + Utilities.formatDate(now, "JST", "yyyyMMdd_HHmmss");
    const linesPerPage = 5;

    // 🆕 exportAllCO30Combinedと同じ分割ルール: 1ページ目はCO30_Template(5件)、
    //    残りは別紙テンプレートがあればCO30_別紙_Template(49件/ページ)、無ければ本紙の繰り返し(5件/ページ)。
    const mainCount = Math.min(flattenedLines.length, linesPerPage);
    const remainingCount = flattenedLines.length - mainCount;
    const useSuppl = remainingCount > 0 && !!templateSheetSuppl;
    const remainderCapacity = useSuppl ? CO30_SUPPL_CAPACITY : linesPerPage;
    const remainderPages = remainingCount > 0 ? Math.ceil(remainingCount / remainderCapacity) : 0;
    const totalPages = 1 + remainderPages;
    const mainPagesCount = useSuppl ? 1 : totalPages;

    // ---- メインページ用の結合シート ----
    const printMainSheet = ss.insertSheet(`PRINT_CO30_RP_${uniqueKeyBase}`);
    if (printMainSheet.getMaxColumns() < 40) {
      printMainSheet.insertColumnsAfter(printMainSheet.getMaxColumns(), 40 - printMainSheet.getMaxColumns());
    }
    for (let col = 1; col <= 40; col++) {
      printMainSheet.setColumnWidth(col, templateSheet30.getColumnWidth(col));
    }

    // ---- 別紙用の結合シート(必要な場合のみ) ----
    let printSupplSheet = null;
    if (useSuppl) {
      printSupplSheet = ss.insertSheet(`PRINT_CO30SUPPL_RP_${uniqueKeyBase}`);
      const supplTemplateCols = templateSheetSuppl.getMaxColumns();
      if (printSupplSheet.getMaxColumns() < supplTemplateCols) {
        printSupplSheet.insertColumnsAfter(printSupplSheet.getMaxColumns(), supplTemplateCols - printSupplSheet.getMaxColumns());
      }
      for (let col = 1; col <= supplTemplateCols; col++) {
        printSupplSheet.setColumnWidth(col, templateSheetSuppl.getColumnWidth(col));
      }
    }

    let mainPageNumber = null;
    let mainCursorRow = 1;
    let supplCursorRow = 1;

    for (let p = 0; p < totalPages; p++) {
      const isLastPage = (p === totalPages - 1);

      if (p === 0) {

        const tempSheet30 = templateSheet30.copyTo(ss);
        tempSheet30.setName(`TEMP_CO30_RP_P${p + 1}_${uniqueKeyBase}`);
        if (tempSheet30.getMaxColumns() < 40) {

        tempSheet30.getRange("T31").setNumberFormat("@").setValue(`${p + 1} / ${totalPages}`);

        for (let idx = 0; idx < mainCount; idx++) {
          const lineData = flattenedLines[idx];
          const rowNum = CO30_ITEM_ROWS[idx];
          setCell_(tempSheet30, `${CO30_COL_TOOLNO}${rowNum}`, lineData.code);
          setCell_(tempSheet30, `${CO30_COL_DESC}${rowNum}`, lineData.desc);

          setCell_(tempSheet30, `${CO30_COL_REQREF}${rowNum}`, lineData.reqRef || "");

        }
        tempSheet30.getRange("A1:AN31").copyTo(printMainSheet.getRange(mainCursorRow, 1, 31, 40));
        try { ss.deleteSheet(tempSheet30); } catch (e) {}
        mainCursorRow += mainIsLastOfItsPdf ? 31 : 33;

      } else if (useSuppl) {

        const tempSheetSuppl = templateSheetSuppl.copyTo(ss);

        }

        const pageNumber = nextCO30PageNumber_(fromLocation);
        setCell_(tempSheet30, CO30_NUMBER_CELL, `${pageNumber}(再印刷)`, "@");
        tempSheet30.getRange("D3").setNumberFormat("@").setValue(autoSignDate);
        tempSheet30.getRange("E5").setValue(fromLocation);
        tempSheet30.getRange("AF5").setNumberFormat("@").setValue(ownerInfo.phone);
        tempSheet30.getRange("D7").setValue(ownerInfo.address);
        tempSheet30.getRange("D22").setValue(Object.keys(remarksSet).join("\n") || "特になし");
        tempSheet30.getRange("T31").setNumberFormat("@").setValue(`${p + 1} / ${totalPages}`);

        const startIdx = mainCount + (p - 1) * linesPerPage;
        const endIdx = Math.min(startIdx + linesPerPage, flattenedLines.length);
        for (let idx = startIdx; idx < endIdx; idx++) {
          const lineData = flattenedLines[idx];
          const rowNum = CO30_ITEM_ROWS[idx - startIdx];
          setCell_(tempSheet30, `${CO30_COL_TOOLNO}${rowNum}`, lineData.code);
          setCell_(tempSheet30, `${CO30_COL_DESC}${rowNum}`, lineData.desc);
          setCell_(tempSheet30, `${CO30_COL_LENDER}${rowNum}`, lineData.lenderDest || "");
          setCell_(tempSheet30, `${CO30_COL_REQREF}${rowNum}`, lineData.reqRef || "");

        for (let r = 1; r <= 31; r++) {

        }
        if (!isLastPage) {
          printMainSheet.setRowHeight(mainCursorRow + 31, 20);
          printMainSheet.setRowHeight(mainCursorRow + 32, 20);
        }
        tempSheet30.getRange("A1:AN31").copyTo(printMainSheet.getRange(mainCursorRow, 1, 31, 40));
        try { ss.deleteSheet(tempSheet30); } catch (e) {}
        mainCursorRow += isLastPage ? 31 : 33;
      }
    }

    SpreadsheetApp.flush();

    // ---- メインPDFを書き出す ----
    const mainRowsCount = mainCursorRow - 1;
    const urlMain = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printMainSheet.getSheetId()}&r1=0&c1=0&r2=${mainRowsCount}&c2=39`;
    const blobMain = UrlFetchApp.fetch(urlMain, { headers: { 'Authorization': `Bearer ${token}` } })
      .getBlob().setName(`CO-30_再印刷_[${fromLocation}]_${autoSignDate} ${autoSignTime}(全${mainPagesCount}頁).pdf`);
    try { ss.deleteSheet(printMainSheet); } catch (e) {}

    // ---- 別紙PDFを書き出す(ある場合のみ) ----
    let blobSuppl = null;
    if (useSuppl) {
      const supplRowsCount = supplCursorRow - 1;
      const supplColsForExport = 5; // 🆕 用具番号/説明/貸出先/備考/依頼番号の5列固定。getLastColumn()は環境によって不安定なため直接指定する
      const urlSuppl = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printSupplSheet.getSheetId()}&r1=0&c1=0&r2=${supplRowsCount}&c2=${supplColsForExport}`;
      blobSuppl = UrlFetchApp.fetch(urlSuppl, { headers: { 'Authorization': `Bearer ${token}` } })
        .getBlob().setName(`CO-30_別紙_再印刷_[${fromLocation}]_${autoSignDate} ${autoSignTime}(全${remainderPages}頁).pdf`);
      try { ss.deleteSheet(printSupplSheet); } catch (e) {}
    }

    try {
      const printFolder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
      removeOldCO30FilesForLocation_(printFolder, fromLocation); // 同じ拠点の古いCO-30(本紙・別紙とも)をゴミ箱へ
      printFolder.createFile(blobMain.copyBlob());
      if (blobSuppl) printFolder.createFile(blobSuppl.copyBlob());
    } catch (driveErr) {
      console.error("CO30再印刷のGoogle Drive保存に失敗しました: " + driveErr.message);
    }

    const supplNote = blobSuppl ? `\n\n※物品数が多いため、別紙(CO-30_別紙_再印刷_...)も別ファイルで添付しています。` : "";
    sendMail_({
      to: truckingEmail,
      subject: `【再印刷】CO-30 用具受取書(拠点: ${fromLocation})`,
      body: `運送・用具部門 担当者様\n\nダッシュボードから、拠点「${fromLocation}」のCO-30を最新のデータで再印刷しました。\n番号・検品状態など既存のデータは変更していません。古いPDFと差し替えてご利用ください。${supplNote}`,
      attachments: blobSuppl ? [blobMain, blobSuppl] : [blobMain]
    });

    return { success: true, message: `✅ 拠点「${fromLocation}」のCO-30を再印刷し、メール送信・Driveへの保存を行いました。` };
  } catch (err) {
    return { success: false, message: "❌ 再印刷に失敗しました: " + err.message };
  }
}

/**
 * 🆕 Dept_Masterで「運送・用具部門」の行を"行ごと"に走査し、各行のD列(メール)・E列(氏名)を
 * ペアで返す。以前は1セルにカンマ区切りで複数メールを入れていたが、
 * 承認者ごとに行を分けて入力する方式に変更した。
 * 例:
 *   運送・用具部門 | ... | ... | tanaka@example.com | 田村裕
 *   運送・用具部門 | ... | ... | gima@example.com    | 儀間悟
 */
function getApproversWithNames_() {
  const fallback = [{ email: "2026trucking@gmail.com", name: "" }];
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Dept_Master");
    if (!sheet) return fallback;
    const lastRow = sheet.getLastRow();
    if (lastRow < 1) return fallback;
    const values = sheet.getRange(1, 1, lastRow, 5).getValues(); // A〜E列

    // 🆕 A列(部門名)が結合セルになっている場合、Apps Scriptは結合範囲の1行目にしか値を返さない。
    //    2行目以降が見た目は「運送・用具部門」でも実際には空欄というケースに対応するため、
    //    結合範囲の中では上のセルの値を引き継がせる。
    const effectiveDeptA = values.map(function(r) { return String(r[0] || "").trim(); });
    const mergedRanges = sheet.getRange(1, 1, lastRow, 1).getMergedRanges();
    mergedRanges.forEach(function(mr) {
      const topRow = mr.getRow();
      const numRows = mr.getNumRows();
      const anchorVal = effectiveDeptA[topRow - 1];
      for (let r = topRow; r < topRow + numRows; r++) {
        effectiveDeptA[r - 1] = anchorVal;
      }
    });

    const list = [];
    for (let i = 0; i < values.length; i++) {
      if (effectiveDeptA[i] !== "運送・用具部門") continue;
      const cellD = values[i][3] ? String(values[i][3]).trim() : "";
      const cellE = values[i][4] ? String(values[i][4]).trim() : "";
      if (!cellD) continue;
      // 🆕 「1セルにカンマ区切りで複数メール」という旧形式が残っていても読めるようにする
      const emails = cellD.split(",").map(function(e) { return e.trim(); }).filter(function(e) { return e.includes("@"); });
      if (emails.length === 1) {
        list.push({ email: emails[0], name: cellE });
      } else if (emails.length > 1) {
        // 旧形式(1セルに複数メール)の場合、氏名を個別に紐づけられないため空のまま登録する
        emails.forEach(function(email) { list.push({ email: email, name: "" }); });
      }
    }
    return list.length > 0 ? list : fallback;
  } catch (e) {}
  return fallback;
}

function getApprovers_() {
  return getApproversWithNames_().map(function(a) { return a.email; });
}

/** 🆕 承認ボタンを押したアカウントのメールから、Dept_MasterのE列(氏名)を逆引きする(前後空白・大文字小文字ゆれを吸収) */
function getApproverNameByEmail_(email) {
  if (!email) return "";
  const target = String(email).trim().toLowerCase();
  const found = getApproversWithNames_().find(function(a) { return a.email.trim().toLowerCase() === target; });
  return found ? found.name : "";
}

// 🆕 1グループ=1トークン: トークンは複数行(rowIds)を束ねて承認する。
//    旧形式トークン({rowId: n})がScriptPropertiesに残っていても読めるよう、
//    読み取り側は必ず tokenRowIds_() を通して行番号の配列を取り出すこと。
function generateApprovalToken_(rowIds) {
  const token = Utilities.getUuid().replace(/-/g, "");
  PropertiesService.getScriptProperties().setProperty(
    "appr_" + token,
    JSON.stringify({ rowIds: rowIds, created: Date.now(), status: "pending" })
  );
  return token;
}

/** トークンデータから対象行番号の配列を取り出す(旧形式 {rowId} との後方互換) */
function tokenRowIds_(data) {
  if (data && data.rowIds && data.rowIds.length) return data.rowIds.map(function(r) { return Number(r); });
  if (data && data.rowId) return [Number(data.rowId)];
  return [];
}

function invalidateTokensForRow_(rowId) {
  invalidateTokensForRows_([rowId]);
}

/**
 * 🆕 複数行のトークンをまとめて無効化する。
 * getKeys()+getProperty()をキーの数だけ呼ぶと通信回数が膨大になるため、
 * getProperties()で一度に全件取得してからメモリ上で判定する。
 */
function invalidateTokensForRows_(rowIds) {
  const props = PropertiesService.getScriptProperties();
  const set = {};
  (rowIds || []).forEach(function(r) { set[Number(r)] = true; });
  if (Object.keys(set).length === 0) return;

  let all = {};
  try { all = props.getProperties(); } catch (e) { return; }

  Object.keys(all).forEach(function(k) {
    if (k.indexOf("co31copy_") === 0 || k.indexOf("co31print_") === 0) {
      const n = Number(k.substring(k.indexOf("_") + 1));
      if (set[n]) { try { props.deleteProperty(k); } catch (e) {} }
      return;
    }
    if (k.indexOf("appr_") !== 0) return;
    try {
      const d = JSON.parse(all[k]);
      if (d && d.status === "pending" && tokenRowIds_(d).some(function(r) { return set[r]; })) {
        props.deleteProperty(k);
      }
    } catch (e) {}
  });
}

/**
 * 🆕 トークンを介さず、行番号配列を直接承認する共通コア処理。
 * ダッシュボードからの一括承認で使う(メールリンク経由のapproveByToken_と処理内容は同じ)。
 */
function approveRowsDirect_(rows, approverEmail) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!sheet) return { success: false, message: "CO31_Responsesが見つかりません" };
  if (!rows || rows.length === 0) return { success: false, message: "対象行がありません" };

  const props = PropertiesService.getScriptProperties();
  const approverDisplayName = getApproverNameByEmail_(approverEmail) || approverEmail || "運送部門担当者";

  const targetRows = rows.filter(function(r) {
    const status = String(sheet.getRange(r, 30).getValue() || "").trim();
    return status !== "CO30作成待ち" && status !== "送信完了" && status !== "取消" && status !== "承認";
  });
  if (targetRows.length === 0) return { success: false, message: "承認待ちの行がありませんでした(既に処理済みの可能性があります)" };

  // 🆕 ここでもPDF生成はせず、キューに任せる
  let firstIid = "";
  targetRows.forEach(function(r) {
    const iid = assignInternalRequestId_(sheet, r);
    if (!firstIid) firstIid = iid;
    props.setProperty("apprby_id_" + iid, approverDisplayName);
    sheet.getRange(r, 30).setValue("承認");
    invalidateTokensForRow_(r);
  });
  if (firstIid) {
    props.deleteProperty("apprfail_" + firstIid);
    props.deleteProperty("apprnotified_" + firstIid);
  }
  scheduleApprovalWorker_(5000);

  return {
    success: true, processedCount: targetRows.length, failedCount: 0,
    message: targetRows.length + "件を承認しました(控えは数分以内に送信されます)"
  };
}

/**
 * 🆕 ダッシュボードから複数の依頼グループをまとめて承認する。
 * groupsRowsList: [[行番号,...], [行番号,...], ...] の配列。approverEmailは承認権限者のいずれか。
 */
function approvePendingGroupsFromDashboard(groupsRowsList, approverEmail) {
  const lock = LockService.getScriptLock();
  const gotLock = lock.tryLock(15 * 1000);
  if (!gotLock) return { success: false, message: "⏳ 他の処理が実行中です。少し待ってからもう一度お試しください。" };
  try {
    if (!approverEmail) return { success: false, message: "承認者を選択してください。" };
    const approvers = getApprovers_();
    if (approvers.indexOf(approverEmail) === -1) return { success: false, message: "選択された承認者には承認権限がありません。" };

    let totalProcessed = 0, totalFailed = 0, groupCount = 0, errorCount = 0;
    (groupsRowsList || []).forEach(function(rows) {
      const ids = (rows || []).map(function(r) { return parseInt(r, 10); }).filter(function(r) { return !isNaN(r) && r >= 2; });
      if (ids.length === 0) return;
      try {
        const res = approveRowsDirect_(ids, approverEmail);
        if (res.success) { groupCount++; totalProcessed += res.processedCount; totalFailed += res.failedCount; }
        else { errorCount++; }
      } catch (err) {
        errorCount++;
      }
    });

    return {
      success: groupCount > 0,
      message: `✅ ${groupCount}件の依頼を承認しました${totalFailed > 0 ? `(${totalFailed}ページで一部失敗)` : ""}${errorCount > 0 ? `、${errorCount}件は処理できませんでした` : ""}`
    };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 🆕 重複依頼チェック。同じ部門から、同じ品名を含む依頼が既に出ていないか調べ、
 * 見つかった場合だけメール本文に差し込む注意文言を返す(無ければ空文字)。
 *
 * 判定基準:
 *   ・同じ依頼部門であること
 *   ・品名が一致すること(前後の空白と大文字小文字の違いは無視)
 *   ・ステータスが「取消」以外であること(承認待ち・承認済み・送信完了はすべて対象)
 *   ・自分自身の依頼(同じ親依頼IDで束ねた全行)は当然除外する
 * ※日付や配送先は見ていないため、別日・別場所への正当な依頼も候補に挙がる。
 *   あくまで「確認を促す」ための文言であり、送信をブロックはしない。
 */
function buildDuplicateWarning_(deptName, itemNames, ownRows) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
    if (!sheet || sheet.getLastRow() < 2) return "";
    const dept = String(deptName || "").trim();
    if (!dept) return "";

    const wanted = {};
    (itemNames || []).forEach(function(n) {
      const key = String(n || "").trim().toLowerCase();
      if (key) wanted[key] = String(n).trim();
    });
    if (Object.keys(wanted).length === 0) return "";

    const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, CO31_TOTAL_COLS).getValues();

    // 自分自身の依頼は除外する。親依頼IDで束ねた兄弟行もまとめて除外(修正の再承認で誤検知しないように)
    const ownSet = {};
    (ownRows || []).forEach(function(r) { ownSet[Number(r)] = true; });
    const ownGroupIds = {};
    for (let i = 0; i < values.length; i++) {
      if (!ownSet[i + 2]) continue;
      const gid = String(values[i][CO31_COL_GROUPID_IDX] || "").trim();
      if (gid) ownGroupIds[gid] = true;
    }

    const hits = {};
    for (let i = 0; i < values.length; i++) {
      const rowNum = i + 2;
      const r = values[i];
      if (ownSet[rowNum]) continue;
      const gid = String(r[CO31_COL_GROUPID_IDX] || "").trim();
      if (gid && ownGroupIds[gid]) continue;
      if (String(r[4] || "").trim() !== dept) continue;
      const status = String(r[29] || "").trim();
      if (status === "取消") continue;

      const dispNo = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim() || "未採番";
      collectItems_(r).forEach(function(it) {
        const key = String(it.name || "").trim().toLowerCase();
        if (!key || !wanted[key]) return;
        if (!hits[key]) hits[key] = [];
        const label = `No.${dispNo}(${status || "状態不明"})`;
        if (hits[key].indexOf(label) === -1) hits[key].push(label);
      });
    }

    const keys = Object.keys(hits);
    if (keys.length === 0) return "";
    const lines = keys.slice(0, 10).map(function(k) {
      return `  ・${wanted[k]} … ${hits[k].slice(0, 5).join("、")}`;
    });
    return `
━━━━━━━━━━━━━━━━━━━━━━━
⚠️ 重複の可能性はありませんか?
同じ部門から、以下の品名を含む依頼が既に出ています。
${lines.join("\n")}
二重の手配になっていないかご確認ください。
(搬入日や配送先が異なる別の依頼であれば問題ありません)
━━━━━━━━━━━━━━━━━━━━━━━
`;
  } catch (e) {
    return "";
  }
}

▼ 承認する場合は以下のリンクをタップしてください
${approvalUrl}
━━━━━━━━━━━━━━━━━━━━━━━

  // 🔒 承認依頼メール(承認リンク付き)は、運送・用具部門の承認者(Dept_Master D14:E15)だけに送る。
  //    依頼部門(Dept_MasterのB列)には絶対に送らない ── 自分で承認できてしまうため。
  //    依頼部門が受け取るのは、承認後に sendGroupCO31_ が送る控えメール(承認リンクなし)だけ。
  const ccList = approvers.slice(1);

  sendMail_({
    to: approvers[0],
    cc: ccList.join(","),
    subject: `${isEdit ? "【承認依頼・CO-31 修正】" : "【承認依頼・CO-31】"}${dept}からの配送依頼${pageCount > 1 ? `(全${pageCount}ページ)` : ""}`,
    body: body
  });
}

function showApprovalPage_(token, userEmail) {
  const props = PropertiesService.getScriptProperties();
  const dataStr = props.getProperty("appr_" + token);

  if (!dataStr) {
    return buildSimplePage_("❌ リンクが無効です", "このリンクは存在しないか、すでに無効になっています。\n(依頼が修正された場合、古い承認リンクは無効になります)");
  }
  const data = JSON.parse(dataStr);

  if (data.status === "approved") {
    const approvedAt = data.approvedAt ? new Date(data.approvedAt).toLocaleString("ja-JP") : "不明";
    return buildSimplePage_("✅ すでに承認済みです", `この依頼は ${approvedAt} に承認されました。`);
  }

/**
 * 🆕 承認処理のコアロジック(HTMLを組み立てない、シンプルな結果オブジェクトを返す)。
 * doPost経由(フォールバック)・google.script.run経由(通常経路)の両方から呼ばれる共通処理。
 * 戻り値: { success, code, message, approverDisplayName, userEmail }
 */
function approveByToken_(token, userEmail) {
  const props = PropertiesService.getScriptProperties();
  const key = "appr_" + token;
  const dataStr = props.getProperty(key);

  if (!dataStr) return { success: false, code: "invalid", message: "このトークンは存在しません。" };
  const data = JSON.parse(dataStr);
  if (data.status === "approved") return { success: false, code: "already", message: "この依頼はすでに承認されています。" };

  if (userEmail) {
    const approvers = getApprovers_();
    if (!approvers.includes(userEmail)) {
      return { success: false, code: "forbidden", message: `${userEmail} には承認権限がありません。` };
    }
  }

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("CO31_Responses");
    const rows = tokenRowIds_(data);
    if (rows.length === 0) throw new Error("トークンに承認対象の行が含まれていません。");

    const approverDisplayName = getApproverNameByEmail_(userEmail) || userEmail || "運送部門担当者";

    // 🆕 PDF生成・メール送信は重く実行時間上限に抵触するため、ここではマークだけ付ける。
    //    実際の処理は processApprovalQueue_ がバックグラウンドで行う(独立した実行枠を使える)。
    let marked = 0;
    let firstIid = "";
    rows.forEach(function(r) {
      const st = String(sheet.getRange(r, 30).getValue() || "").trim();
      if (st === "CO30作成待ち" || st === "送信完了" || st === "取消") return;
      // 承認者名は行番号ではなく内部依頼ID(不変)に紐づける
      const iid = assignInternalRequestId_(sheet, r);
      if (!firstIid) firstIid = iid;
      props.setProperty("apprby_id_" + iid, approverDisplayName);
      sheet.getRange(r, 30).setValue("承認");
      marked++;
    });

    if (firstIid) {
      props.deleteProperty("apprfail_" + firstIid);
      props.deleteProperty("apprnotified_" + firstIid);
    }

    // 行への書き込みが成功してからトークンを承認済みにする(途中失敗時は再承認できるようにするため)
    data.status = "approved";
    data.approvedAt = Date.now();
    data.approvedBy = userEmail || "不明";
    props.setProperty(key, JSON.stringify(data));

    scheduleApprovalWorker_(5000);

    return {
      success: true, queued: true, markedCount: marked,
      approverDisplayName: approverDisplayName, userEmail: userEmail || "",
      partialFailure: false, failedCount: 0
    };
  } catch (err) {
    return { success: false, code: "error", message: err.message };
  }
}

/** doPost(素のフォームPOST)からのフォールバック用。HTMLページを組み立てて返す。 */

    return buildSimplePage_("❌ 承認処理でエラーが発生しました", result.message + "\n\nスプレッドシートから手動で承認してください。");
  }

  const html = `<!DOCTYPE html>
<html lang="ja"><head><meta charset="UTF-8"><meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>承認完了</title>
<style>body{font-family:-apple-system,sans-serif;text-align:center;padding:60px 20px;background:#f4f6f7;}
.icon{font-size:64px;margin-bottom:16px;} h1{color:#27ae60;font-size:22px;} p{color:#666;font-size:14px;line-height:1.6;}</style>
</head><body>
<div class="icon">${result.partialFailure ? "⚠️" : "✅"}</div><h1>承認が完了しました</h1>
<p>承認を受け付けました。<br>CO-31の控えは、数分以内に担当者・依頼部門へ自動送信されます。<br>この画面は閉じていただいて構いません。</p>

  return HtmlService.createHtmlOutput(html).setTitle("承認完了");

/**
 * 定期トリガーの実行先。LockServiceで多重実行を防ぎ、UIが無い実行環境でも安全に動くようにする。
 * ロックが取れない場合は、人が今まさに手動実行中と判断し、静かにスキップする(ログにのみ残す)。
 */
function runScheduledCO30Batch_() {
  const lock = LockService.getScriptLock();
  const gotLock = lock.tryLock(5 * 1000);
  if (!gotLock) {
    console.log("CO30定期実行: 他の処理が実行中のため、この回はスキップしました。");
    return;
  }
  try {
    const result = exportAllCO30Combined();
    console.log(result.message);
    if (result.supersededCount > 0) {
      try {
        sendMail_({
          to: "2026trucking@gmail.com",
          subject: `【要確認】CO30自動作成で${result.supersededCount}件の差し替えがありました`,
          body: `運送・用具部門 担当者様\n\n定期実行によるCO-30の一括作成で、修正前の内容が既に現場で処理済みだった物品が${result.supersededCount}件ありました。\n該当の行は削除せず、備考欄に「⚠️内容修正により差し替え」の印を付けて保持しています。\n\nダッシュボードで内容をご確認ください。`
        });
      } catch (e) {}
    }
  } catch (err) {
    console.error("CO30定期実行でエラーが発生しました: " + err.message);

      sendMail_({
        to: "2026trucking@gmail.com",

      });
    } catch (e2) {}
  } finally {
    lock.releaseLock();
  }
}

/** 自動作成トリガーを設定する共通ロジック(4時間おき)。既に設定済みなら二重に作らない。 */
function enableCO30Trigger_() {
  const existing = ScriptApp.getProjectTriggers().filter(function(t) {
    return t.getHandlerFunction() === 'runScheduledCO30Batch_';
  });
  if (existing.length > 0) {
    return { success: true, enabled: true, message: "✅ 既に自動作成は設定済みです(4時間おき)。" };
  }
  ScriptApp.newTrigger('runScheduledCO30Batch_').timeBased().everyHours(4).create();
  return { success: true, enabled: true, message: "✅ CO-30の自動作成を4時間おきに設定しました。" };
}

/** 自動作成トリガーを解除する共通ロジック */
function disableCO30Trigger_() {
  const triggers = ScriptApp.getProjectTriggers().filter(function(t) {
    return t.getHandlerFunction() === 'runScheduledCO30Batch_';
  });
  if (triggers.length === 0) {
    return { success: true, enabled: false, message: "自動作成は設定されていません。" };
  }
  triggers.forEach(function(t) { ScriptApp.deleteTrigger(t); });
  return { success: true, enabled: false, message: `✅ 自動作成(${triggers.length}件)を解除しました。` };
}

/** 🆕 ダッシュボードで現在の自動作成ON/OFF状態を表示するために使う */

  const existing = ScriptApp.getProjectTriggers().filter(function(t) {
    return t.getHandlerFunction() === 'runScheduledCO30Batch_';
  });
  return { enabled: existing.length > 0, intervalHours: 4 };
}

function submitCO31Form(formData) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("CO31_Responses");
  if (!sheet) throw new Error("エラー：シート「CO31_Responses」が見つかりません。");

  const team = determineTeam_(formData.fromLocation);
  const timestamp = new Date();

  let arrivalFullText = formData.arrivalDateStr || "";
  if (formData.arrivalTimeStr) { arrivalFullText += " " + formData.arrivalTimeStr; } else { arrivalFullText += " 指定なし"; }
  if (formData.arrivalDetailTimeStr && formData.arrivalDetailTimeStr.trim() !== "") { arrivalFullText += " " + formData.arrivalDetailTimeStr.trim(); } else { arrivalFullText += " なし"; }

  // フォームの品目スロット(1〜20)を順番どおりに読み取る
  const slots = [];
  for (let n = 1; n <= 20; n++) {
    slots.push({
      qty:  formData["qty" + n]  || "",
      name: formData["item" + n] || "",
      size: formData["size" + n] || ""
    });
  }

  // 🆕 1行分(最大5品目・物品1〜5列のみ)のrowFullを組み立てる。物品6〜20列(index33-77)は新形式では使わない
  function buildRow_(chunk) {
    const rowFull = new Array(CO31_TOTAL_COLS).fill("");
    rowFull[0] = timestamp;
    rowFull[2] = formData.docDate || "";
    rowFull[3] = formData.fromLocation || "";
    rowFull[4] = formData.deptName || "";
    rowFull[5] = formData.toLocation || "";
    rowFull[6] = arrivalFullText;
    rowFull[7] = formData.cost || "";               // 費用: 空欄OK
    const firstBlocks = [8, 11, 14, 17, 20];
    firstBlocks.forEach(function(base, idx) {
      const it = chunk[idx] || { qty: "", name: "", size: "" };
      rowFull[base]     = it.qty  || "";
      rowFull[base + 1] = it.name || "";
      rowFull[base + 2] = it.size || "";
    });
    rowFull[24] = formData.totalNote || "";
    rowFull[25] = formData.approver || "";
    rowFull[29] = "承認待ち";
    rowFull[30] = team;
    rowFull[31] = formData.contactInfo || "";        // AF: 担当者名・連絡先
    rowFull[32] = formData.contactEmail || "";       // AG: 担当者メール
    return rowFull;
  }

  function chunkItemsAll_(chunk) {
    return chunk.filter(function(it) { return it && (it.name || it.qty); });
  }

  const isEdit = formData.editRow && parseInt(formData.editRow, 10) >= 2;
  // 🆕 1送信=1トークン=1通: このトークンで承認されるべき全行と、メール本文用データを集める
  let tokenRowIds = [];
  let emailFormData = null;

  if (isEdit) {
    const anchorRow = parseInt(formData.editRow, 10);
    const anchorData = sheet.getRange(anchorRow, 1, 1, CO31_TOTAL_COLS).getValues()[0];
    const groupId = String(anchorData[CO31_COL_GROUPID_IDX] || "").trim();

    if (!groupId) {
      // ---- 旧形式(親IDなし・1行=1依頼・最大20品目)の修正: 従来通り1行に上書き ----
      const rowFull = buildRow_(slots.slice(0, 5));
      for (let n = 6; n <= 20; n++) {                // AH〜: 物品6〜20(旧形式のみ)
        const base = 33 + (n - 6) * 3;
        const it = slots[n - 1];
        rowFull[base]     = it.qty  || "";
        rowFull[base + 1] = it.name || "";
        rowFull[base + 2] = it.size || "";
      }
      const internalId = anchorData[1];
      rowFull[1] = internalId ? String(internalId).trim() : "";
      rowFull[CO31_COL_DISPLAYPAGE_IDX] = anchorData[CO31_COL_DISPLAYPAGE_IDX] || "";

      const changed = detectChangedFields_(anchorData, rowFull);
      rowFull[CO31_COL_CHANGED_IDX] = JSON.stringify(changed);

      sheet.getRange(anchorRow, 1, 1, CO31_TOTAL_COLS).setValues([rowFull]);
      invalidateTokensForRow_(anchorRow);

      tokenRowIds = [anchorRow];
      emailFormData = Object.assign({}, formData);
      emailFormData.itemsAll = collectItems_(rowFull).map(function(it) { return { name: it.name, qty: it.qty }; });
      emailFormData.changedSummary = buildChangeSummary_(changed, collectItems_(rowFull));
    } else {
      // ---- 🆕 新形式(親IDで束ねた複数行)の修正 ----
      const groupRows = getGroupRowNumbers_(sheet, groupId);
      const rowCount = groupRows.length;

      const existingSlots = slots.slice(0, rowCount * 5);
      const extraFilled = slots.slice(rowCount * 5).filter(function(it) { return it.name || it.qty; });

      // 🆕 グループ全体(既存ページ+追加分)の品目が0件になる修正は禁止する。全削除は「取消」で行ってもらう。
      const totalRemaining = existingSlots.filter(function(it) { return it && (it.name || it.qty); }).length + extraFilled.length;
      if (totalRemaining === 0) {
        throw new Error("すべての品目が削除されています。依頼全体が不要な場合は、修正ではなく「取消」をご利用ください。");
      }

      // ---- フェーズ1: 品目が全て空になったページを特定する(この時点ではまだ何も書き換えない) ----
      const emptyPageRows = []; // { rowNum, oldDisplay, oldInternalId }
      const remainingChunks = []; // 残すページの新しい品目チャンク(元の並び順を維持)
      for (let k = 0; k < rowCount; k++) {
        const chunk = existingSlots.slice(k * 5, k * 5 + 5);
        const filledCount = chunk.filter(function(it) { return it && (it.name || it.qty); }).length;
        if (filledCount === 0) {
          const rn = groupRows[k];
          const oldR = sheet.getRange(rn, 1, 1, CO31_TOTAL_COLS).getValues()[0];
          emptyPageRows.push({
            rowNum: rn,
            oldDisplay: String(oldR[CO31_COL_DISPLAYPAGE_IDX] || "").trim(),
            oldInternalId: oldR[1] ? String(oldR[1]).trim() : ""
          });
        } else {
          remainingChunks.push(chunk);
        }
      }

      // ---- フェーズ2: 空になったページを削除する(既にCO30化済みなら現物側に「取消済み」の印を付けてから削除) ----
      if (emptyPageRows.length > 0) {
        emptyPageRows.forEach(function(p) {
          if (p.oldInternalId) markCO30ItemsAsCancelled_(p.oldInternalId); // 既存無ければ何もしない(安全)
          invalidateTokensForRow_(p.rowNum);
        });
        // 行番号のズレを防ぐため、必ず降順(下の行から)で削除する
        emptyPageRows.slice().sort(function(a, b) { return b.rowNum - a.rowNum; }).forEach(function(p) {
          try { sheet.deleteRow(p.rowNum); } catch (e) {}
        });
        try {
          const printFolder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
          emptyPageRows.forEach(function(p) { if (p.oldDisplay) removeOldDriveFilesForRequest_(printFolder, p.oldDisplay); });
        } catch (e) {}
      }

      // ---- フェーズ3: 削除後の実際の行番号を取り直し、残ったページに新しい内容を書き込む ----
      const groupRowsAfter = getGroupRowNumbers_(sheet, groupId);
      const changedRows = [];
      const pendingUnchangedRows = [];
      const summaryParts = [];

      emptyPageRows.forEach(function(p) {
        summaryParts.push(`【ページ削除】${p.oldDisplay ? "No." + p.oldDisplay + " " : ""}このページの品目がすべて削除されたため、ページ自体を削除しました。`);
      });

      for (let k = 0; k < groupRowsAfter.length; k++) {
        const rowNum = groupRowsAfter[k];
        const oldRow = sheet.getRange(rowNum, 1, 1, CO31_TOTAL_COLS).getValues()[0];
        const oldStatus = String(oldRow[29] || "").trim();
        const chunk = remainingChunks[k];
        const rowFull = buildRow_(chunk);
        rowFull[1] = oldRow[1] ? String(oldRow[1]).trim() : "";
        rowFull[CO31_COL_GROUPID_IDX] = groupId;
        rowFull[CO31_COL_DISPLAYPAGE_IDX] = oldRow[CO31_COL_DISPLAYPAGE_IDX] || "";

        const changed = detectChangedFields_(oldRow, rowFull);
        if (changed.fields.length === 0 && changed.itemIndices.length === 0) {
          if (oldStatus === "承認待ち") pendingUnchangedRows.push(rowNum);
          continue;
        }

        rowFull[CO31_COL_CHANGED_IDX] = JSON.stringify(changed);
        sheet.getRange(rowNum, 1, 1, CO31_TOTAL_COLS).setValues([rowFull]);
        invalidateTokensForRow_(rowNum);
        changedRows.push(rowNum);

        const dispNo = String(oldRow[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
        const pagePrefix = groupRowsAfter.length > 1 ? `【${k + 1}ページ目${dispNo ? "・No." + dispNo : ""}】\n` : "";
        summaryParts.push(pagePrefix + buildChangeSummary_(changed, collectItems_(rowFull)));
      }

      // ---- フェーズ4: 既存ページを超える追加分を、新しいページとして末尾に追記する ----
      const newRowsAdded = [];
      if (extraFilled.length > 0) {
        for (let i = 0; i < extraFilled.length; i += 5) {
          const chunk = extraFilled.slice(i, i + 5);
          const rowFull = buildRow_(chunk);
          rowFull[CO31_COL_GROUPID_IDX] = groupId;
          sheet.appendRow(rowFull);
          const newRowNum = sheet.getLastRow();
          newRowsAdded.push(newRowNum);
          const newPageIdx = groupRowsAfter.length + Math.floor(i / 5) + 1;
          const addedNames = chunk.filter(function(it) { return it.name; }).map(function(it) { return it.name; }).join("、");
          summaryParts.push(`【${newPageIdx}ページ目・新規追加】\n・追加された品目: ${addedNames || "(品名なし)"}`);
        }
      }

      pendingUnchangedRows.forEach(function(rn) { invalidateTokensForRow_(rn); });
      tokenRowIds = changedRows.concat(pendingUnchangedRows).concat(newRowsAdded).sort(function(a, b) { return a - b; });

      if (tokenRowIds.length === 0) {
        // 🆕 ページ削除だけが発生し、残りのページに変更が無く再承認の必要も無いケース。
        //    再承認は不要だが、削除の事実は運送・用具部門【と依頼部門本人】の双方に知らせる。
        const delDeptEmail = getDeptEmail_(formData.deptName || "");
        const delContactEmail = (formData.contactEmail && formData.contactEmail.includes("@")) ? formData.contactEmail : "";
        const delCc = [];
        if (delDeptEmail) delCc.push(delDeptEmail);
        if (delContactEmail && delCc.indexOf(delContactEmail) === -1) delCc.push(delContactEmail);
        try {
          const delMailOpts = {
            to: "2026trucking@gmail.com",
            subject: `【CO-31 ページ削除】${formData.deptName || "不明部門"}の依頼から一部ページが削除されました`,
            body: `運送・用具部門 御中\n${formData.deptName || "不明部門"} 御中\n\n${formData.deptName || "不明部門"}からの修正操作により、以下のページが削除されました(残りのページに変更はありません)。\n\n${summaryParts.join("\n")}\n\n※削除したページの控えPDF・印刷済みの紙をお持ちの場合は、今後の手続きには使用せず破棄してください。`
          };
          if (delCc.length > 0) delMailOpts.cc = delCc.join(",");
          sendMail_(delMailOpts);
        } catch (e) {}
        return { success: true, edit: true, pagesDeleted: true, deletedCount: emptyPageRows.length };
      }

      emailFormData = Object.assign({}, formData);
      emailFormData.itemsAll = chunkItemsAll_(existingSlots).concat(extraFilled);
      emailFormData.changedSummary = summaryParts.join("\n");
    }
  } else {
    // ---- 🆕 新規: 空スロットを詰めて5品目ごとに行を分け、共通の親依頼IDで束ねる ----
    const filled = slots.filter(function(it) { return it.name || it.qty; });
    const chunks = [];
    for (let i = 0; i < filled.length; i += 5) chunks.push(filled.slice(i, i + 5));
    if (chunks.length === 0) chunks.push([]);

    const groupId = assignNewGroupId_(sheet);
    for (let k = 0; k < chunks.length; k++) {
      const rowFull = buildRow_(chunks[k]);
      rowFull[CO31_COL_GROUPID_IDX] = groupId;
      sheet.appendRow(rowFull);
      tokenRowIds.push(sheet.getLastRow());
    }

    emailFormData = Object.assign({}, formData);
    emailFormData.itemsAll = filled;
  }

  try {
    sendApprovalRequestEmail_(tokenRowIds, emailFormData, !!isEdit);
  } catch (emailErr) {
    console.error("承認依頼メールの送信に失敗しました: " + emailErr.message);
  }
  return { success: true, edit: !!isEdit };
}

/**
 * 🆕 CO31_Responsesを親依頼ID(AA列)単位にグルーピングして返す共通ヘルパー。
 * 親IDが空の行(旧形式)は「1行=1グループ」として扱う。
 * 返り値: 出現順のグループ配列。各グループは
 *   { row(先頭行), rows[], groupId, reqNos[], statuses[], items[], dept, docDate,
 *     fromLocation, toLocation, arrival, team, timestamp }
 */
function collectCO31Groups_() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!sheet || sheet.getLastRow() < 2) return [];
  const lastRow = sheet.getLastRow();
  const values = sheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();
  const arrivalDisp = sheet.getRange(2, 7, lastRow - 1, 1).getDisplayValues();

  const groups = {};
  const order = [];
  for (let i = 0; i < values.length; i++) {
    const r = values[i];
    if (!r[4]) continue;
    const gid = String(r[CO31_COL_GROUPID_IDX] || "").trim();
    const key = gid ? "g:" + gid : "r:" + (i + 2);
    if (!groups[key]) {
      groups[key] = {
        row: i + 2, rows: [], groupId: gid, reqNos: [], statuses: [], items: [],
        dept: String(r[4]).trim(),
        docDate: r[2] ? String(r[2]) : "",
        fromLocation: r[3] ? String(r[3]).trim() : "",
        toLocation: r[5] ? String(r[5]).trim() : "",
        arrival: arrivalDisp[i][0] || "",
        team: r[30] ? String(r[30]).trim() : "",
        timestamp: (r[0] instanceof Date) ? r[0].getTime() : 0
      };
      order.push(key);
    }
    const g = groups[key];
    g.rows.push(i + 2);
    const reqNo = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    if (reqNo) g.reqNos.push(reqNo);
    g.statuses.push(r[29] ? String(r[29]).trim() : "");
    collectItems_(r).filter(function(it) { return it.name; }).forEach(function(it) {
      g.items.push({ name: it.name, qty: it.qty });
    });
  }
  return order.map(function(key) { return groups[key]; });
}

function listAllCO31Requests_() {
  const out = collectCO31Groups_().map(function(g) {
    return {
      row: g.row,                       // 先頭行(アンカー)。修正・取消はこの行番号を起点に親IDで全行を辿る
      rows: g.rows,
      groupId: g.groupId,
      reqNo: g.reqNos.join("、"),       // 各行の番号を連結(例: "清1、清2"。未承認ならまだ空)
      dept: g.dept,
      docDate: g.docDate,
      fromLocation: g.fromLocation,
      toLocation: g.toLocation,
      arrival: g.arrival,
      status: aggregateGroupStatus_(g.statuses),
      team: g.team,
      items: g.items,
      timestamp: g.timestamp
    };
  });
  out.sort(function(a, b) { return b.timestamp - a.timestamp; });
  return out;
}

function getDeptRequests(deptName) {
  return listAllCO31Requests_().filter(function(r) { return r.dept === String(deptName).trim(); });
}

function getRequestDetail(row) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!sheet) throw new Error("CO31_Responsesが見つかりません");
  const r = sheet.getRange(row, 1, 1, CO31_TOTAL_COLS).getValues()[0];
  const groupId = String(r[CO31_COL_GROUPID_IDX] || "").trim();

  const items = [];
  const reqNos = [];
  const statuses = [];
  let rows = [row];

  if (groupId) {
    // 🆕 新形式: 親IDに紐づく全行から品目を行順に連結(各行は物品1〜5列のみ)
    rows = getGroupRowNumbers_(sheet, groupId);
    rows.forEach(function(rn) {
      const rr = (rn === row) ? r : sheet.getRange(rn, 1, 1, CO31_TOTAL_COLS).getValues()[0];
      [8, 11, 14, 17, 20].forEach(function(i) {
        items.push({ qty: String(rr[i] || ""), name: String(rr[i + 1] || ""), size: String(rr[i + 2] || "") });
      });
      const no = String(rr[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
      reqNos.push(no || ""); // 🆕 rows[i]とreqNos[i]が常に1対1対応するよう、未採番でも空文字列を入れる
      statuses.push(String(rr[29] || "").trim());
    });
  } else {
    // 旧形式: 1行から最大20品目を読む
    [8, 11, 14, 17, 20].forEach(function(i) {
      items.push({ qty: String(r[i] || ""), name: String(r[i + 1] || ""), size: String(r[i + 2] || "") });
    });
    for (let i = 33; i <= 75; i += 3) {
      items.push({ qty: String(r[i] || ""), name: String(r[i + 1] || ""), size: String(r[i + 2] || "") });
    }
    const no = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    reqNos.push(no || ""); // 🆕 新形式と同様、rows[i]とreqNos[i]が常に1対1対応するようにする
    statuses.push(String(r[29] || "").trim());
  }

  // フォームは常に20スロットあるため、20件になるまで空品目でパディングする
  while (items.length < 20) items.push({ qty: "", name: "", size: "" });

  return {
    row: rows[0],                                    // アンカーは常にグループ先頭行
    rows: rows,
    groupId: groupId,
    itemLimit: groupId ? rows.length * 5 : 20,       // 🆕 修正フォームで編集を許可する品目枠数
    reqNo: reqNos.filter(Boolean).join("、"),         // 表示用連結文字列は空文字列を除いてから繋げる
    reqNos: reqNos,                                   // 🆕 rows配列と同じ順序・同じ長さ(未採番の行は空文字列)
    docDate: String(r[2] || ""),
    fromLocation: String(r[3] || ""),
    deptName: String(r[4] || ""),
    toLocation: String(r[5] || ""),
    arrival: String(sheet.getRange(row, 7).getDisplayValue() || ""),
    cost: String(r[7] || ""),
    totalNote: String(r[24] || ""),
    approver: String(r[25] || ""),
    contactInfo: String(r[31] || ""),
    contactEmail: String(r[32] || ""),
    status: aggregateGroupStatus_(statuses),
    items: items
  };
}

/** 🆕 AD列(ステータス)の入力規則(プルダウン制限)に「取消」が含まれていない場合に自動追加する */
function ensureStatusValidationIncludesCancel_(sheet) {
  try {
    const rule = SpreadsheetApp.newDataValidation()
      .requireValueInList(["承認待ち", "承認", "送信完了", "CO30作成待ち", "取消"], true)
      .setAllowInvalid(false)
      .build();
    const lastRow = Math.max(sheet.getLastRow(), 1000); // 将来追加される行の分もあらかじめ余裕を持たせる
    sheet.getRange(2, 30, lastRow - 1, 1).setDataValidation(rule);
  } catch (e) {
    console.error("ステータス列の入力規則更新に失敗しました: " + e.message);
  }
}

/** 🆕 CO31が取消された際、既にCO30化(内部依頼IDで紐付け)されている物品の備考に「取消済み」の印を付ける */
function markCO30ItemsAsCancelled_(internalId) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO30_Inspection");
  if (!sheet || sheet.getLastRow() < 2) return 0;
  const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 16).getValues();
  let count = 0;
  for (let i = 0; i < values.length; i++) {
    const rowInternalId = values[i][15] ? String(values[i][15]).trim() : "";
    if (rowInternalId !== internalId) continue;
    const currentRemarks = values[i][14] ? String(values[i][14]).trim() : "";
    if (currentRemarks.indexOf("⚠️取消済み") !== -1) continue; // 既に印済みなら二重に付けない
    const newRemarks = currentRemarks ? `⚠️取消済み / ${currentRemarks}` : "⚠️取消済み";
    sheet.getRange(i + 2, 15).setValue(newRemarks); // O列=備考
    count++;
  }
  return count;
}

/**
 * 🩺 バックフィル用: 過去に「取消」になったが、まだCO30_Inspection側に印が付いていない依頼を
 * まとめて見つけて備考に印を付ける。Apps Scriptエディタから直接実行してください。
 * (取消済みの依頼はフォーム側にもう「取消」ボタンが出ないため、この関数で後追い対応する)
 */
function BACKFILL_markCancelledCO30Items() {
  const respSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
  if (!respSheet || respSheet.getLastRow() < 2) {
    SpreadsheetApp.getUi().alert("CO31_Responsesにデータがありません。");
    return;
  }
  const lastRow = respSheet.getLastRow();
  const values = respSheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();
  let totalMarked = 0;
  let cancelledRequestCount = 0;
  for (let i = 0; i < values.length; i++) {
    const status = String(values[i][29] || "").trim();
    if (status !== "取消") continue;
    const internalId = String(values[i][1] || "").trim();
    if (!internalId) continue;
    cancelledRequestCount++;
    totalMarked += markCO30ItemsAsCancelled_(internalId);
  }
  SpreadsheetApp.getUi().alert(`✅ 完了しました。\n取消済みの依頼: ${cancelledRequestCount}件を確認し、CO30_Inspection上の${totalMarked}件の物品に「⚠️取消済み」の印を新たに付けました。`);
}

function cancelRequest(row) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
    if (!sheet) return { success: false, error: "CO31_Responsesシートが見つかりません" };

    const rowNum = parseInt(row, 10);
    if (isNaN(rowNum) || rowNum < 2) return { success: false, error: "行番号が不正です(" + row + ")" };
    const lastRow = sheet.getLastRow();
    if (rowNum > lastRow) return { success: false, error: "対象の行が見つかりません(行" + rowNum + ")" };

    // 🆕 必要な範囲を1回だけまとめて読む(行ごとにgetRangeすると通信回数が増えて遅い)
    const cols = Math.min(CO31_TOTAL_COLS, sheet.getMaxColumns());
    const all = sheet.getRange(2, 1, lastRow - 1, cols).getValues();
    const get_ = function(rn) {
      const arr = (all[rn - 2] || []).slice();

      return arr;

      }

      oldStatuses.push(String(rr[29] || "").trim());

      if (iid) internalIds.push(iid);
      sheet.getRange(rn, 30).setValue("取消");
    });

    // 🆕 トークン無効化・CO30への印付け・Drive削除・通知メールはすべてバックグラウンドへ。
    //    同期で行うと実行時間の上限に達し、try-catchで捕捉できないエラーになる。
    try {
      const job = {
        dept: String(oldRow[4] || "不明部門").trim(),
        fromLoc: String(oldRow[3] || "").trim(),
        toLoc: String(oldRow[5] || "").trim(),
        contactInfo: String(oldRow[31] || "").trim(),
        contactEmail: String(oldRow[32] || "").trim(),
        rows: targetRows,
        internalIds: internalIds,
        reqNos: reqNos,
        oldStatuses: oldStatuses
      };
      PropertiesService.getScriptProperties()
        .setProperty("cancjob_" + Date.now() + "_" + rowNum, JSON.stringify(job));
      scheduleCancelWorker_(3000);
    } catch (e) {
      console.error("取消の後処理を予約できませんでした: " + e.message);
    }

    return { success: true, cancelledCount: targetRows.length };
  } catch (err) {
    const msg = (err && err.message) ? err.message : String(err);
    console.error("cancelRequestでエラー: " + msg);
    return { success: false, error: msg || "原因不明のエラーです。実行ログをご確認ください。" };
  }
}

/**
 * 🆕 検品スキャナー用の依頼一覧(シンプル版)。
 * 部門・物品名・数量・検品状況(物品コードごとにチェック済みか)だけを返す。
 * CO30_Inspectionの「内部依頼ID」列で紐付け、物品名でグルーピングする。
 * CO30がまだ作成されていない場合は codes が空配列になる。
 */
function listRequestsForScanner() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const respSheet = ss.getSheetByName("CO31_Responses");
  if (!respSheet || respSheet.getLastRow() < 2) return [];
  const lastRow = respSheet.getLastRow();
  const values = respSheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();

  // CO30_Inspectionを内部依頼ID(16列目)でグルーピング → 物品名ごとにコードをまとめる
  // 🆕 「検品済みか」は数量ベースではなく、そのコードの現在のライフサイクル状態で判定する(呼び出し側=scan.htmlで
  //     選択中のモードと比較して判定できるよう、生のlifecycle値をそのまま返す)
  const inspByReqId = {}; // { internalId: { itemName: [{code,lifecycle,remarks,location}] } }
  const inspSheet = ss.getSheetByName("CO30_Inspection");
  if (inspSheet && inspSheet.getLastRow() >= 2) {
    const iv = inspSheet.getRange(2, 1, inspSheet.getLastRow() - 1, 16).getValues();
    iv.forEach(function(r) {
      const code = r[1] ? String(r[1]).trim() : "";
      const name = r[2] ? String(r[2]).trim() : "";
      const internalId = r[15] ? String(r[15]).trim() : "";
      if (!code || !internalId) return;
      const location = r[0] ? String(r[0]).trim() : "";
      const lifecycle = r[11] ? String(r[11]).trim() : "未処理";
      const remarks = r[14] ? String(r[14]).trim() : "";
      if (!inspByReqId[internalId]) inspByReqId[internalId] = {};
      if (!inspByReqId[internalId][name]) inspByReqId[internalId][name] = [];
      inspByReqId[internalId][name].push({ code: code, lifecycle: lifecycle, remarks: remarks, location: location });
    });
  }

  const out = [];
  for (let i = 0; i < values.length; i++) {
    const r = values[i];
    if (!r[4]) continue;
    const status = r[29] ? String(r[29]).trim() : "";
    if (status === "取消") continue;

    const internalId = r[1] ? String(r[1]).trim() : "";
    const dept = String(r[4]).trim();
    const reqNo = String(r[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    const requestNote = String(r[24] || "").trim(); // 🆕 CO31の特記事項(Y列)
    const co30Groups = internalId ? inspByReqId[internalId] : null;

    let items;
    if (co30Groups) {
      items = Object.keys(co30Groups).map(function(name) {
        return { name: name, qty: String(co30Groups[name].length), codes: co30Groups[name] };
      });
    } else {
      items = collectItems_(r).filter(function(it) { return it.name; })
        .map(function(it) { return { name: it.name, qty: it.qty || "1", codes: [] }; });
    }

    out.push({
      reqNo: reqNo,
      dept: dept,
      requestNote: requestNote,
      items: items,
      timestamp: (r[0] instanceof Date) ? r[0].getTime() : 0
    });
  }
  out.sort(function(a, b) { return b.timestamp - a.timestamp; });
  return out;
}

function getEquipmentItems() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const masterSheet = ss.getSheetByName("Item_Master");
  if (!masterSheet) return [];
  const values = masterSheet.getDataRange().getValues();
  const items = [];
  for (let i = 1; i < values.length; i++) {
    const name    = values[i][0] ? String(values[i][0]).trim() : "";
    const maker   = values[i][1] ? String(values[i][1]).trim() : "";
    const purpose = values[i][2] ? String(values[i][2]).trim() : "";
    if (name !== "") items.push({ name: name, maker: maker, purpose: purpose });
  }
  return items;
}

/**
 * 🆕 Item_MasterのH列(場所名)・I列(住所)から、配送元フォームの候補と住所を取得する。
 * フォームの「配送元」プルダウンの候補として使う。
 */
function getLocationOptions_() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Item_Master");
  if (!sheet) return [];
  const values = sheet.getDataRange().getValues();
  const seen = {};
  const out = [];
  for (let i = 1; i < values.length; i++) {
    const name = values[i][7] ? String(values[i][7]).trim() : ""; // H列
    if (!name || seen[name]) continue;
    seen[name] = true;
    out.push({
      name: name,
      address: values[i][8] ? String(values[i][8]).trim() : "",   // I列
      code: values[i][6] ? String(values[i][6]).trim() : "",      // 🆕 G列=識別記号
      team: values[i][9] ? String(values[i][9]).trim() : ""       // 🆕 J列=担当TM
    });
  }
  return out;
}

/** getLocationOptions_()を「場所名→住所」のマップ形式にしたもの(CO30生成時の住所書き込みに使う) */
function getLocationAddressMap_() {
  const map = {};
  getLocationOptions_().forEach(function(loc) { map[loc.name] = loc.address; });
  return map;
}

/** google.script.runからフォーム(index.html)が呼び出す用。配送元候補一覧を返す。 */
function getFromLocationOptions() {
  return getLocationOptions_().map(function(loc) { return loc.name; });
}

/**
 * 🆕 1行分のCO31 PDF(PDF-A: 原本+コピー2 / PDF-B: コピー1)を生成して返す。
 * 採番(assignCO31DisplayNumber_)・internalId付与はここで行うが、
 * メール送信・Drive保存・ステータス更新は行わない(sendGroupCO31_側でまとめて実施)。
 */
function generateCO31PdfsForRow_(responseSheet, row, approverDisplayName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const templateSheet31 = ss.getSheetByName("CO31_Template");
  if (!templateSheet31) throw new Error("CO31_Templateシートが見つかりません。");

  const uniqueKey = "REQ_" + Utilities.formatDate(new Date(), "JST", "yyyyMMdd_HHmmss_SSS") + "_R" + row;

  const targetData = responseSheet.getRange(row, 1, 1, CO31_TOTAL_COLS).getValues()[0];
  const docDateStr      = targetData[2] ? String(targetData[2]) : "";
  const fromLocation    = targetData[3] ? String(targetData[3]).trim() : "未指定拠点";
  const toDept          = targetData[4] ? String(targetData[4]).trim() : "不明部門";
  const toLocation      = targetData[5];
  const requestDateTime = String(responseSheet.getRange(row, 7).getDisplayValue());
  const cost            = targetData[7];
  const totalQty        = targetData[23];
  const remarks         = targetData[24];
  const approverName    = targetData[25] ? String(targetData[25]).trim() : "責任者"; // 依頼部門の責任者氏名(フォーム入力値)
  const transportName   = approverDisplayName || "運送部門担当者"; // 🆕 承認ボタンを押した運送・用具部門担当者の氏名
  const contactInfo     = targetData[31] ? String(targetData[31]).trim() : "";
  const contactEmail    = targetData[32] ? String(targetData[32]).trim() : "";

  // 内部依頼ID(CO30連携用・不変)
  const internalId = assignInternalRequestId_(responseSheet, row);

  // 🆕 部門ごとの表示番号を採番する前に、旧番号を控えておく(修正時の古いDriveファイル特定用)
  const oldDisplayBeforeRevision = String(targetData[CO31_COL_DISPLAYPAGE_IDX] || "").trim();

  // 🆕 部門ごとの表示番号を採番(初回="設5"型 / 修正の再承認="設5-1"型)
  // 🆕 修正マーク(変更フィールドのJSON)がある場合だけ改訂版として番号を進める
  const changedRawForRev = targetData[CO31_COL_CHANGED_IDX];
  const isRevisionRequested = !!(changedRawForRev && String(changedRawForRev).trim());
  const numInfo = assignCO31DisplayNumber_(responseSheet, row, toDept, isRevisionRequested);
  const isRev = numInfo.isRevision;

  // 🆕 改訂番号を採番した「直後」にマークを消す。
  //    処理の最後で消すと、途中でタイムアウトした際に再処理で番号が二重に進んでしまう。
  if (isRev) {
    responseSheet.getRange(row, CO31_COL_CHANGED_IDX + 1).setValue("");
  }

  // 🆕 修正で変更されたフィールドを取得(初回承認時は空)
  let changedInfo = { fields: [], itemIndices: [] };
  try {
    const raw = targetData[CO31_COL_CHANGED_IDX];
    if (raw) changedInfo = JSON.parse(raw);
  } catch (e) {}
  function hl(key) { return isRev && changedInfo.fields && changedInfo.fields.indexOf(key) !== -1; }

  // 物品(最大20)→ 5品目/ページに分割
  let items = collectItems_(targetData).filter(function(it) { return it.name || it.qty; });
  if (items.length === 0) items = [{ qty: "", name: "", size: "" }];
  const pages = [];
  for (let i = 0; i < items.length; i += 5) pages.push(items.slice(i, i + 5));
  const totalPages = pages.length;

  const now = new Date();
  const autoSignDate = Utilities.formatDate(now, "JST", "yyyy/MM/dd");
  const autoSignTime = Utilities.formatDate(now, "JST", "HH:mm");

  const token = ScriptApp.getOAuthToken();
  const formattedDocDate = docDateStr ? Utilities.formatDate(new Date(docDateStr), "JST", "yyyy/MM/dd") : autoSignDate;

  let datePart = "", amPmPart = "指定なし", detailTimePart = "なし";
  if (requestDateTime) {
    if (requestDateTime.includes("午後")) { amPmPart = "午後"; let s = requestDateTime.split("午後"); datePart = s[0].trim(); detailTimePart = s[1] ? s[1].trim() : "なし"; }
    else if (requestDateTime.includes("午前")) { amPmPart = "午前"; let s = requestDateTime.split("午前"); datePart = s[0].trim(); detailTimePart = s[1] ? s[1].trim() : "なし"; }
    else { amPmPart = "指定なし"; datePart = requestDateTime.includes("指定なし") ? requestDateTime.split("指定なし")[0].trim() : requestDateTime.trim(); }
  }

  const pdfStorageBlobs = [];   // PDF-A: 原本+コピー2を1枚にまとめた保管・印刷用(運送部門宛)
  const pdfRequesterBlobs = []; // PDF-B: 単独1ページ構成の控え(コピー1・署名欄印字あり、依頼部門宛)

  for (let p = 0; p < totalPages; p++) {
    const pageSuffix = totalPages > 1 ? `_p${p + 1}` : "";
    // 印字する番号(複数ページなら「(1/3)」を添えるが、保存されている番号自体は全ページ共通)
    const printedNumber = totalPages > 1 ? `${numInfo.display} (${p + 1}/${totalPages})` : numInfo.display;

    // ---- 元になるテンプレート(この後、原本+コピーとして複製する) ----
    const tempSheet31 = templateSheet31.copyTo(ss);
    tempSheet31.setName(`TEMP_CO31_P${p + 1}_${uniqueKey}`);

    setCell_(tempSheet31, CO31_NUMBER_CELL, printedNumber, "@");  // 番号(修正時は自動でハイライト不要、番号自体が変わるので分かる)
    setCellHL_(tempSheet31, "S3", formattedDocDate, hl('docDate'), "@");
    setCellHL_(tempSheet31, "C4", fromLocation, hl('fromLocation'));
    setCell_(tempSheet31, "N4", toDept);        // 配送先(部門) ※部門は編集フォームでは固定のためハイライト対象外
    setCellHL_(tempSheet31, "M6", toLocation, hl('toLocation'));    // 配送先(場所)
    setCellHL_(tempSheet31, "M8", contactInfo, hl('contactInfo')); // 担当者名・連絡先
    setCellHL_(tempSheet31, "R9", cost || "", hl('cost'));

    setCellHL_(tempSheet31, "C9", datePart, hl('arrival'));
    if (!requestDateTime.includes("指定なし") && (requestDateTime.includes("午前") || requestDateTime.includes("午後"))) {
      if (amPmPart === "午前") setCellHL_(tempSheet31, "J9", detailTimePart !== "なし" ? detailTimePart : "✔", hl('arrival'));
      else if (amPmPart === "午後") setCellHL_(tempSheet31, "O9", detailTimePart !== "なし" ? detailTimePart : "✔", hl('arrival'));
    }

    const pageItems = pages[p];
    for (let i = 0; i < 5; i++) {
      const it = pageItems[i] || { qty: "", name: "", size: "" };
      const startRow = 12 + i;
      const globalIdx = p * 5 + i;
      const itemChanged = isRev && changedInfo.itemIndices && changedInfo.itemIndices.indexOf(globalIdx) !== -1;
      setCellHL_(tempSheet31, `B${startRow}`, it.qty || "", itemChanged);
      setCellHL_(tempSheet31, `E${startRow}`, it.name || "", itemChanged);
      setCellHL_(tempSheet31, `K${startRow}`, it.size || "", itemChanged);
    }
    if (p === 0) setCell_(tempSheet31, "Q12", totalQty || "");
    setCellHL_(tempSheet31, "C18", remarks || "特になし", hl('totalNote'));
    setCell_(tempSheet31, "C20", transportName || "運送部門担当者");
    setCell_(tempSheet31, "C22", autoSignDate);
    setCell_(tempSheet31, "G22", autoSignTime);
    setCellHL_(tempSheet31, "K20", approverName || "責任者 承認済", hl('approver'));

    // ---- PDF-A: 原本+コピー2を1枚にまとめたシートを作成(保管・印刷用。従来のprintTempSheet31と同じ構成) ----
    const printTempSheet31 = ss.insertSheet(`PRINT_TEMP_31_P${p + 1}_${uniqueKey}`);
    for (let col = 1; col <= 21; col++) printTempSheet31.setColumnWidth(col, tempSheet31.getColumnWidth(col));
    tempSheet31.getRange("A1:U25").copyTo(printTempSheet31.getRange("A1:U25"));
    tempSheet31.getRange("A1:U25").copyTo(printTempSheet31.getRange("A27:U51"));
    for (let r = 1; r <= 25; r++) {
      printTempSheet31.setRowHeight(r, tempSheet31.getRowHeight(r));
      printTempSheet31.setRowHeight(r + 26, tempSheet31.getRowHeight(r));
    }
    printTempSheet31.setRowHeight(26, 20);
    printTempSheet31.getRange("Q2").setValue("【原本】"); printTempSheet31.getRange("Q28").setValue("【コピー2】");
    printTempSheet31.getRange("C20").setValue(""); printTempSheet31.getRange("C22").setValue(""); printTempSheet31.getRange("G22").setValue("");
    printTempSheet31.getRange("C46").setValue(""); printTempSheet31.getRange("C48").setValue(""); printTempSheet31.getRange("G48").setValue("");
    SpreadsheetApp.flush();

    const url31Print = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printTempSheet31.getSheetId()}&r1=0&c1=0&r2=51&c2=21`;
    const storageBlob = UrlFetchApp.fetch(url31Print, { headers: { 'Authorization': `Bearer ${token}` } }).getBlob();
    storageBlob.setName(`CO-31_${toDept}_No${numInfo.display}${pageSuffix}.pdf`);
    pdfStorageBlobs.push(storageBlob);
    try { ss.deleteSheet(printTempSheet31); } catch (e) {}

    // ---- PDF-B: 依頼部門宛、単独1ページ構成(コピー1)。署名欄(C20/C22/G22)は空欄クリアせず、tempSheet31の値をそのまま印字 ----
    const printReqSheet31 = ss.insertSheet(`PRINT_REQ_31_P${p + 1}_${uniqueKey}`);
    for (let col = 1; col <= 21; col++) printReqSheet31.setColumnWidth(col, tempSheet31.getColumnWidth(col));
    tempSheet31.getRange("A1:U25").copyTo(printReqSheet31.getRange("A1:U25"));
    for (let r = 1; r <= 25; r++) {
      printReqSheet31.setRowHeight(r, tempSheet31.getRowHeight(r));
    }
    printReqSheet31.getRange("Q2").setValue("【コピー1】");
    SpreadsheetApp.flush();

    const url31Req = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printReqSheet31.getSheetId()}&r1=0&c1=0&r2=25&c2=21`;
    const requesterBlob = UrlFetchApp.fetch(url31Req, { headers: { 'Authorization': `Bearer ${token}` } }).getBlob();
    requesterBlob.setName(`CO-31_${toDept}_No${numInfo.display}${pageSuffix}.pdf`);
    pdfRequesterBlobs.push(requesterBlob);
    try { ss.deleteSheet(printReqSheet31); } catch (e) {}

    try { ss.deleteSheet(tempSheet31); } catch (e) {}
  }

  return {
    row: row,
    toDept: toDept,
    fromLocation: fromLocation,
    approverName: approverName,
    contactInfo: contactInfo,
    contactEmail: contactEmail,
    display: numInfo.display,
    isRev: isRev,
    oldDisplay: oldDisplayBeforeRevision,
    changeSummary: isRev ? buildChangeSummary_(changedInfo, items) : "",
    totalPages: totalPages,
    storageBlobs: pdfStorageBlobs,     // PDF-A: 原本+コピー2(保管・印刷用、運送部門宛)
    requesterBlobs: pdfRequesterBlobs  // PDF-B: コピー1(署名欄印字あり、依頼部門宛)
  };
}

/**
 * 🆕 グループ(複数行=複数ページ)のCO31をまとめて処理する本体。
 * 全行分のPDFを生成し、Drive保存・依頼部門宛1通・運送部門宛1通のメール送信・ステータス更新を行う。
 * 戻り値: { processedCount, failedPages: [{row, error}], displays: [...] }
 * 一部の行で失敗しても残りは処理を続け、失敗情報を戻り値で返す(呼び出し元が承認者に警告表示する)。
 */
function sendGroupCO31_(responseSheet, rows, approverDisplayName) {
  const truckingEmail = "2026trucking@gmail.com";
  const transportName = approverDisplayName || "運送部門担当者";
  const props = PropertiesService.getScriptProperties();

  const copyDone_  = function(r) { return props.getProperty("co31copy_" + r) === "1"; };
  const printDone_ = function(r) { return props.getProperty("co31print_" + r) === "1"; };

  // 🆕 前回の実行でメール送信まで終わっていた場合(ステータス更新直前に力尽きたケース)は、
  //    PDFの再生成もメールの再送もせず、ステータスの確定だけ行う。
  if (rows.length > 0 && rows.every(copyDone_) && rows.every(printDone_)) {
    rows.forEach(function(r) {
      const st = String(responseSheet.getRange(r, 30).getValue() || "").trim();
      if (st === "承認") {
        responseSheet.getRange(r, 30).setValue("CO30作成待ち");
        responseSheet.getRange(r, CO31_COL_CHANGED_IDX + 1).setValue("");
      }
      props.deleteProperty("co31copy_" + r);
      props.deleteProperty("co31print_" + r);
    });
    return { processedCount: rows.length, failedPages: [], displays: [], resumed: true };
  }

  const results = [];
  const failedPages = [];
  rows.forEach(function(row) {
    try {
      const status = String(responseSheet.getRange(row, 30).getValue() || "").trim();
      if (status === "CO30作成待ち" || status === "送信完了" || status === "取消") return;
      results.push(generateCO31PdfsForRow_(responseSheet, row, approverDisplayName));
    } catch (err) {
      failedPages.push({ row: row, error: err.message });
      console.error("CO31ページ処理に失敗しました(行" + row + "): " + err.message);
    }
  });

  if (results.length === 0) {
    return { processedCount: 0, failedPages: failedPages, displays: [] };
  }

  const first = results[0];
  const toDept = first.toDept;
  const resultRows = results.map(function(r) { return r.row; });
  const displays = results.map(function(r) { return r.display; });
  const displayJoined = displays.join("、");
  const anyRev = results.some(function(r) { return r.isRev; });
  const totalPageCount = results.reduce(function(s, r) { return s + r.totalPages; }, 0);
  const copyAlreadySent  = resultRows.every(copyDone_);
  const printAlreadySent = resultRows.every(printDone_);

  // ---- 印刷用PDFをGoogleドライブに保存 ----
  try {
    const printFolder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
    results.forEach(function(res) {
      if (res.isRev && res.oldDisplay) removeOldDriveFilesForRequest_(printFolder, res.oldDisplay);
    });
    results.forEach(function(res) {
      res.storageBlobs.forEach(function(blob) {
        removeDriveFilesByExactName_(printFolder, blob.getName()); // 🆕 再処理時の重複を防ぐ
        printFolder.createFile(blob);
      });
    });
  } catch (driveErr) {
    console.error("Google Driveへの保存に失敗しました(フォルダIDと共有権限を確認してください): " + driveErr.message);
  }

  // ---- 宛先 ----
  const deptEmailForCc = getDeptEmail_(toDept);
  let mailTo = (first.contactEmail && first.contactEmail.includes("@")) ? first.contactEmail : "";
  let mailCc = deptEmailForCc || "";
  if (!mailTo) { mailTo = mailCc || truckingEmail; mailCc = ""; }
  if (mailCc && mailCc === mailTo) mailCc = "";

  // ---- 変更サマリー ----
  let changeSummaryText = "";
  const revResults = results.filter(function(r) { return r.isRev && r.changeSummary; });
  if (revResults.length > 0) {
    const parts = revResults.map(function(r) {
      return results.length > 1 ? `【No.${r.display}】\n${r.changeSummary}` : r.changeSummary;
    });
    changeSummaryText = `\n※修正版を含みます。控えPDF内で黄色くハイライトされている箇所が、前回からの変更点です。\n\n■ 今回変更された箇所\n${parts.join("\n")}\n`;
  }

  const failNote = failedPages.length > 0
    ? `\n⚠️ この依頼のうち${failedPages.length}ページで控えの作成に失敗しました。不足分は運送・用具部門にお問い合わせください。\n`
    : "";

  // ---- 重複依頼の注意文言 ----
  let dupNote = "";
  try {
    if (typeof buildDuplicateWarning_ === 'function') {
      const dupNames = [];
      rows.forEach(function(rn) {
        const rr = responseSheet.getRange(rn, 1, 1, CO31_TOTAL_COLS).getValues()[0];
        collectItems_(rr).forEach(function(it) { if (it.name) dupNames.push(it.name); });
      });
      dupNote = buildDuplicateWarning_(toDept, dupNames, rows);
    }
  } catch (e) {}

  // ---- 控えメール(依頼部門宛) ----
  if (!copyAlreadySent) {
    try {
      const editLink = `${getFormBaseUrl_()}?mode=edit&dept=${encodeURIComponent(toDept)}&reqNo=${encodeURIComponent(displays[0])}`;

      const copyBody = `${first.approverName} 兄弟

お疲れ様です。運送・用具部門の${transportName}です。
配送依頼(CO-31)が承認されましたので控えをお送りします。

番号: ${displayJoined}${changeSummaryText}${failNote}${dupNote}
① 依頼の修正が必要な時
・以下のリンクから直接修正フォームを開けます
　${editLink}

② 用具を受け取る時
・配送された用具をご確認ください。
・受領確認として、運送部門が持参するCO-31(コピー)へ受取担当者の署名をお願いします。
・お渡しするCO-31原本は、返却時まで部門で保管してください。

③ 使用後に運送部門へ返却する時
・用具を運送部門へ返却してください。
・保管していたCO-31原本に、返却確認の署名をお願いします。
・これで部門からの返却手続きは完了となります。

④ お願い
・CO-31は、配送・受領・返却の記録となる大切な書類です。紛失しないよう保管をお願いいたします。
・返却時は、借用時と同じ数量・状態でご返却くださいますようお願いいたします。
・数量や状態に変更があった場合は、返却時に運送部門へお知らせください。

兄弟たちのご協力に心から感謝しています。

※部門代表アドレスにもCCでお送りしています。`;

      const copyBlobs = [];
      results.forEach(function(res) { res.requesterBlobs.forEach(function(b) { copyBlobs.push(b); }); });

      const copyOpts = {
        to: mailTo,
        subject: `【${anyRev ? '修正版・' : ''}配送依頼受領】CO-31 大会配送依頼控え(${toDept} / No.${displayJoined})`,
        body: copyBody,
        attachments: copyBlobs
      };
      if (mailCc) copyOpts.cc = mailCc;
      sendMail_(copyOpts);
      // 🆕 送信できたことを記録し、再処理時に二重送信しないようにする
      resultRows.forEach(function(r) { props.setProperty("co31copy_" + r, "1"); });
    } catch (e) {
      console.error("依頼元へのメール送信でエラーが発生しました: " + e.message);
    }
  }

  // ---- 印刷用メール(運送部門宛) ----
  if (!printAlreadySent) {
    try {
      const printBlobs = [];
      results.forEach(function(res) { res.storageBlobs.forEach(function(b) { printBlobs.push(b); }); });
      sendMail_({
        to: truckingEmail,
        subject: `【要対応・CO-31印刷用】${toDept} 大会配送依頼(No.${displayJoined})`,
        body: `運送・用具部門 担当者様\n\n${toDept}の配送依頼(No.${displayJoined})が承認されました。印刷用CO-31を添付します(全${totalPageCount}ページ構成)。\nGoogleドライブの印刷待ちフォルダにも保存済みです。${anyRev ? '\n※修正版を含みます。同じ依頼の古いPDFはドライブ内で削除済みです。' : ''}${failNote}\n※この部門のCO-30受取書は、最終締切後にメニューから一括生成されます。`,
        attachments: printBlobs
      });
      resultRows.forEach(function(r) { props.setProperty("co31print_" + r, "1"); });
    } catch (e) {
      console.error("運送部門への印刷用メール送信でエラーが発生しました: " + e.message);
    }
  }

  // ---- ステータスを確定し、一時的な記録を消す ----
  results.forEach(function(res) {
    responseSheet.getRange(res.row, 30).setValue("CO30作成待ち");
    responseSheet.getRange(res.row, CO31_COL_CHANGED_IDX + 1).setValue("");
    props.deleteProperty("co31copy_" + res.row);
    props.deleteProperty("co31print_" + res.row);
  });

  return { processedCount: results.length, failedPages: failedPages, displays: displays };
}

/**
 * 従来互換の薄いラッパー(シート上でAD列を手動「承認」にした場合のフォールバック等から呼ばれる)。
 * 1行だけのグループとして処理する。
 */
function sendIndividualCO31(responseSheet, row, approverDisplayName) {
  return sendGroupCO31_(responseSheet, [row], approverDisplayName);
}

/** 🆕 修正版を保存する前に、ファイル名に「旧番号」を含むDrive上の古いPDFをゴミ箱へ移動する(番号で厳密に絞り込むため誤削除のリスクは低い) */
function removeOldDriveFilesForRequest_(folder, oldNumber) {
  if (!oldNumber) return;
  try {
    const esc = String(oldNumber).replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    const re = new RegExp("No" + esc + "(?!\\d)");
    const iter = folder.getFiles();
    while (iter.hasNext()) {
      const f = iter.next();
      const name = f.getName();
      if (re.test(name)) {
        try { f.setTrashed(true); } catch (e2) {}
      }
    }
  } catch (e) {}
}

/**
 * 🆕 CO-30「再印刷」専用: 同じ拠点の古いCO-30(通常版・再印刷版どちらも)をゴミ箱に移す。
 * 再印刷は「今の状態のまま同じ内容で作り直す」機能なので、古い版は必ず内容が古くなる。
 * ⚠️ 4時間おきの自動バッチ(exportAllCO30Combined)には絶対に使わないこと。
 *    バッチ生成のCO-30は毎回「新しく溜まった依頼」を含む別内容のため、消すと依頼ごと消えてしまう。
 */
function removeOldCO30FilesForLocation_(folder, fromLocation) {
  if (!fromLocation) return;
  try {
    const marker = `[${fromLocation}]`;
    const iter = folder.getFilesByType(MimeType.PDF);
    while (iter.hasNext()) {
      const f = iter.next();
      const name = f.getName();
      if (name.indexOf("CO-30") !== -1 && name.indexOf(marker) !== -1) {
        try { f.setTrashed(true); } catch (e2) {}
      }
    }
  } catch (e) {}
}

// =========================================================

function exportAllCO30Combined() {
  // 🆕 呼び出し元(メニュー/トリガー/ダッシュボード)によって結果の見せ方が異なるため、
  //    ここではUIアラートなどを直接呼ばず、常に結果オブジェクト { success, message, supersededCount } を返す。
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const responseSheet = ss.getSheetByName("CO31_Responses");

  const templateSheet30 = ss.getSheetByName("CO30_Template");
  // 🆕 別紙テンプレート。存在しない場合はnullのままとし、従来通りCO30_Templateの繰り返しにフォールバックする。
  const templateSheetSuppl = ss.getSheetByName(CO30_SUPPL_SHEET_NAME);
  const ownerSheetCandidates = ["Depot_Master", "Owner_Master"];
  let ownerSheet = null;
  for (let i = 0; i < ownerSheetCandidates.length; i++) {
    const s = ss.getSheetByName(ownerSheetCandidates[i]);
    if (s) { ownerSheet = s; break; }
  }

  const truckingEmail = "2026trucking@gmail.com";

  if (!responseSheet || !templateSheet30) {
    return { success: false, message: "エラー：必要なシートが見つかりません。", supersededCount: 0 };
  }
  const lastRow = responseSheet.getLastRow();
  if (lastRow < 2) { return { success: false, message: "対象データがありません。", supersededCount: 0 }; }

  const dataRange = responseSheet.getRange(2, 1, lastRow - 1, CO31_TOTAL_COLS).getValues();
  let targetRows = [];
  // 拠点ごとに: { lines:[{code,desc,dept,lenderDest,reqRef}], remarks:[] }
  let locationGroup = {};
  let inspectionRows = [];

  let ownerMap = {};
  if (ownerSheet) {
    const oData = ownerSheet.getDataRange().getValues();
    for (let i = 1; i < oData.length; i++) {
      let locName = oData[i][0] ? String(oData[i][0]).trim() : "";
      // 🆕 C列は識別記号(用具番号用)に使うため、ここでは電話番号のみ拾う。住所はItem_MasterのH/I列から別途取得する。
      if (locName) ownerMap[locName] = { phone: oData[i][1] || "", address: "" };
    }
  }
  // 🆕 拠点の住所をItem_MasterのH列(場所名)・I列(住所)から取得し、ownerMapへ上書きする。
  // フォームで「その他(手動入力)」により配送元が自由入力された場合はH列に一致しないため、住所は空欄のままになる。
  const locationAddressMap = getLocationAddressMap_();
  Object.keys(locationAddressMap).forEach(function(locName) {
    if (!ownerMap[locName]) ownerMap[locName] = { phone: "", address: "" };
    ownerMap[locName].address = locationAddressMap[locName];
  });

  for (let i = 0; i < dataRange.length; i++) {
    let rowNum = i + 2;
    let rowArr = dataRange[i];
    let status = rowArr[29];
    if (status !== "CO30作成待ち") continue;

    targetRows.push(rowNum);

    let fromLocation = rowArr[3] ? String(rowArr[3]).trim() : "未指定拠点";
    let deptName = rowArr[4] ? String(rowArr[4]).trim() : "不明部門";
    let toLocation = rowArr[5] ? String(rowArr[5]).trim() : "";
    let note = rowArr[24] ? String(rowArr[24]).trim() : "";
    let team = rowArr[30] ? String(rowArr[30]).trim() : determineTeam_(fromLocation);
    let internalId = rowArr[1] ? String(rowArr[1]).trim() : "";
    if (!internalId) internalId = assignInternalRequestId_(responseSheet, rowNum);
    let displayPage = String(rowArr[CO31_COL_DISPLAYPAGE_IDX] || "").trim();
    if (!displayPage) {
      // 🆕 何らかの理由(ステータス欄の手動書き換え等)で番号が未採番のまま「CO30作成待ち」になっている行を
      //    ここで検知し、その場で番号を採番してシートにも書き戻す(以後は正しい番号が表示される)。
      const backfilled = assignCO31DisplayNumber_(responseSheet, rowNum, deptName);
      displayPage = backfilled.display;
      rowArr[CO31_COL_DISPLAYPAGE_IDX] = displayPage; // このループ内で後から参照される場合に備えて更新
    }
    let reqRef = displayPage;

    if (!locationGroup[fromLocation]) locationGroup[fromLocation] = { lines: [], remarks: [] };
    if (note) {
      // 🆕 同じ依頼(親IDで束ねた複数行)は同じ依頼メモを持つため、同一文言の備考は1回だけ載せる
      const remarkLine = `【${deptName}】${note}`;
      if (locationGroup[fromLocation].remarks.indexOf(remarkLine) === -1) {
        locationGroup[fromLocation].remarks.push(remarkLine);
      }
    }

    // 🆕 CO31の数量・品名・サイズ→CO30の説明欄へ結合
    const reqItems = collectItems_(rowArr);
    reqItems.forEach(function(it) {
      const qty = parseInt(it.qty, 10) || 0;
      const name = it.name;
      if (!name || qty <= 0) return;

      let baseDesc = name;
      if (it.size) baseDesc += ` [${it.size}]`;

      if (qty === 1) {
        locationGroup[fromLocation].lines.push({
          desc: baseDesc, dept: deptName, lenderDest: toLocation, reqRef: reqRef,
          plan: 1, internalId: internalId, suffixMode: "none", requestNote: note
        });
      } else if (qty <= 5) {
        for (let b = 0; b < qty; b++) {
          locationGroup[fromLocation].lines.push({
            desc: baseDesc, dept: deptName, lenderDest: toLocation, reqRef: reqRef,
            plan: 1, internalId: internalId, suffixMode: "letter", suffixIndex: b, requestNote: note
          });
        }
      } else {
        // 数量6以上: 枝番なしで1行にまとめ、説明欄に数量を明記
        locationGroup[fromLocation].lines.push({
          desc: `${baseDesc} ×${qty}`, dept: deptName, lenderDest: toLocation, reqRef: reqRef,
          plan: qty, internalId: internalId, suffixMode: "star", requestNote: note
        });
      }
    });
  }

  let locations = Object.keys(locationGroup);
  if (locations.length === 0) {
    return { success: false, message: "現在「CO30作成待ち」のデータはありません。", supersededCount: 0 };
  }

  // 🆕 拠点ごとに識別記号を取得し、用具番号(物品コード)を採番
  const now = new Date();
  const autoSignDate = Utilities.formatDate(now, "JST", "yyyy/MM/dd");
  const autoSignTime = Utilities.formatDate(now, "JST", "HH:mm:ss"); // 🆕 ファイル名の衝突防止用
  const uniqueKeyBase = Utilities.formatDate(now, "JST", "yyyyMMdd_HHmmss");
  const token = ScriptApp.getOAuthToken();
  const linesPerPage = 5;

  locations.forEach(function(fromLocation) {
    const depotCode = getDepotCode_(fromLocation);
    let seq = getMaxToolSeqForDepotCode_(depotCode);
    locationGroup[fromLocation].lines.forEach(function(line) {
      // 🆕 用具番号の先頭に「依頼部門の略語」を付け、番号を見ただけでどの部門の依頼か分かるようにする
      const deptAbbrev = getDeptAbbrev_(line.dept) || String(line.dept || "").slice(0, 1);
      if (line.suffixMode === "none") {
        seq++;
        line.code = `${deptAbbrev}${depotCode}${seq}`;
      } else if (line.suffixMode === "letter") {
        // 同一商品の最初のユニットでseqを進め、以降の枝番は同じseqを使う
        if (line.suffixIndex === 0) { seq++; }
        line.code = `${deptAbbrev}${depotCode}${seq}${String.fromCharCode(65 + line.suffixIndex)}`;
      } else { // star
        seq++;
        line.code = `${deptAbbrev}${depotCode}${seq}*`;
      }
    });

    // inspectionRows へ変換(17列): 拠点,物品コード,物品名,予定数量,検品済数量,ステータス,
    //   最終更新時刻,担当TM,依頼部門,依頼番号,配送先,ライフサイクル,返却日,返却依頼番号,備考,内部依頼ID,依頼メモ
    locationGroup[fromLocation].lines.forEach(function(line) {
      inspectionRows.push([
        fromLocation, line.code, line.desc, line.plan, 0, "未完了", "",
        determineTeam_(fromLocation), line.dept, line.reqRef,
        line.lenderDest, "未処理", "", "", "", line.internalId, line.requestNote || ""
      ]);
    });
  });

  // 🆕 既存の同一依頼(内部依頼ID)の検品行は、まだ現場で処理されていない(未処理のままの)行だけ
  //    削除して作り直す。既に配送検品済などの進捗がある行は、取消の場合と同じ考え方で、
  //    削除せずに「差し替え」の印を備考に残す(現物のQRラベルは既に貼られている可能性があり、
  //    追跡情報を失うと現物を回収し忘れるリスクがあるため)。
  let supersededCount = 0;
  if (inspectionRows.length > 0) {
    let inspectSheet = ss.getSheetByName("CO30_Inspection");
    if (!inspectSheet) {
      inspectSheet = ss.insertSheet("CO30_Inspection");
    }
    ensureDashboardHeaders_(inspectSheet);

    const idSet = {};
    inspectionRows.forEach(function(r) { if (r[15]) idSet[r[15]] = true; });
    const vals = inspectSheet.getDataRange().getValues();
    for (let r = vals.length - 1; r >= 1; r--) {
      const rid = vals[r][15] ? String(vals[r][15]).trim() : "";
      if (!rid || !idSet[rid]) continue;
      const lifecycle = vals[r][11] ? String(vals[r][11]).trim() : "未処理";
      if (lifecycle === "未処理") {
        inspectSheet.deleteRow(r + 1);
      } else {
        const currentRemarks = vals[r][14] ? String(vals[r][14]).trim() : "";
        if (currentRemarks.indexOf("⚠️内容修正により差し替え") === -1) {
          const newRemarks = currentRemarks ? `⚠️内容修正により差し替え / ${currentRemarks}` : "⚠️内容修正により差し替え";
          inspectSheet.getRange(r + 1, 15).setValue(newRemarks);
          supersededCount++;
        }
      }
    }
    inspectSheet.getRange(inspectSheet.getLastRow() + 1, 1, inspectionRows.length, 17).setValues(inspectionRows);
  }

  locations.forEach(function(fromLocation, locIdx) {
    let group = locationGroup[fromLocation];
    let flattenedLines = group.lines;
    let allRemarks = group.remarks;
    if (flattenedLines.length === 0) return;

    let ownerInfo = ownerMap[fromLocation] || { phone: "", address: "" };

    // 🆕 1ページ目はCO30_Template(5件まで)。それを超える分は、別紙テンプレートがあれば
    //    CO30_別紙_Template(49件/ページ)、無ければ従来通りCO30_Templateの繰り返し(5件/ページ)を使う。
    //
    // ⚠️ メインページと別紙は「列の使い方(幅)」が根本的に異なるため、1つのシートに結合すると
    //    列幅がどちらかに引っ張られて崩れる。そのため、メイン用・別紙用で PDF ファイルを分けて生成する。
    const mainCount = Math.min(flattenedLines.length, linesPerPage);
    const remainingCount = flattenedLines.length - mainCount;
    const useSuppl = remainingCount > 0 && !!templateSheetSuppl;
    const remainderCapacity = useSuppl ? CO30_SUPPL_CAPACITY : linesPerPage;
    const remainderPages = remainingCount > 0 ? Math.ceil(remainingCount / remainderCapacity) : 0;
    const totalPages = 1 + remainderPages;

    // ---- メインページ用の結合シート(1ページ目、および別紙テンプレートが無い場合の2ページ目以降) ----
    const mainPagesCount = useSuppl ? 1 : totalPages;
    let printMainSheetName = `PRINT_CO30_LOC_${locIdx}_${uniqueKeyBase}`;
    let printMainSheet = ss.insertSheet(printMainSheetName);
    if (printMainSheet.getMaxColumns() < 40) {
      printMainSheet.insertColumnsAfter(printMainSheet.getMaxColumns(), 40 - printMainSheet.getMaxColumns());
    }
    for (let col = 1; col <= 40; col++) {
      printMainSheet.setColumnWidth(col, templateSheet30.getColumnWidth(col));
    }

    // ---- 別紙用の結合シート(必要な場合のみ作成。メインとは別の列幅を使う) ----
    let printSupplSheet = null;
    if (useSuppl) {
      printSupplSheet = ss.insertSheet(`PRINT_CO30SUPPL_LOC_${locIdx}_${uniqueKeyBase}`);
      const supplTemplateCols = templateSheetSuppl.getMaxColumns();
      if (printSupplSheet.getMaxColumns() < supplTemplateCols) {
        printSupplSheet.insertColumnsAfter(printSupplSheet.getMaxColumns(), supplTemplateCols - printSupplSheet.getMaxColumns());
      }
      for (let col = 1; col <= supplTemplateCols; col++) {
        printSupplSheet.setColumnWidth(col, templateSheetSuppl.getColumnWidth(col));
      }
    }

    let mainPageNumber = null;    // 🆕 1ページ目の番号を控えておき、別紙にも同じ番号を入れる
    let mainCursorRow = 1;        // メイン結合シートの積み上げ位置
    let supplCursorRow = 1;       // 別紙結合シートの積み上げ位置

    for (let p = 0; p < totalPages; p++) {
      const isLastPage = (p === totalPages - 1);

      if (p === 0) {
        // ---- 1ページ目: CO30_Template(従来通り) ----
        let tempSheet30 = templateSheet30.copyTo(ss);
        tempSheet30.setName(`TEMP_CO30_Loc${locIdx}_P${p + 1}_${uniqueKeyBase}`);
        if (tempSheet30.getMaxColumns() < 40) {
          tempSheet30.insertColumnsAfter(tempSheet30.getMaxColumns(), 40 - tempSheet30.getMaxColumns());
        }

        mainPageNumber = nextCO30PageNumber_(fromLocation); // 拠点ごと・ページごとの通し番号(永続)

        setCell_(tempSheet30, CO30_NUMBER_CELL, mainPageNumber, "@");
        tempSheet30.getRange("D3").setNumberFormat("@").setValue(autoSignDate);
        tempSheet30.getRange("E5").setValue(fromLocation);
        tempSheet30.getRange("AF5").setNumberFormat("@").setValue(ownerInfo.phone);
        tempSheet30.getRange("D7").setValue(ownerInfo.address);
        tempSheet30.getRange("D22").setValue(allRemarks.join("\n") || "特になし");
        tempSheet30.getRange("T31").setNumberFormat("@").setValue(`${p + 1} / ${totalPages}`); // 🆕 総ページ数は別紙込み

        for (let idx = 0; idx < mainCount; idx++) {
          let lineData = flattenedLines[idx];
          let rowNum = CO30_ITEM_ROWS[idx];
          setCell_(tempSheet30, `${CO30_COL_TOOLNO}${rowNum}`, lineData.code);
          setCell_(tempSheet30, `${CO30_COL_DESC}${rowNum}`, lineData.desc);
          setCell_(tempSheet30, `${CO30_COL_LENDER}${rowNum}`, lineData.lenderDest || "");

        const mainIsLastOfItsPdf = useSuppl ? true : isLastPage; // メインPDFとしてはここで最後(別紙は別PDF)

        }

        const supplSpan = CO30_SUPPL_PRINT_LAST_ROW; // 🆕 C49(ページ数表示)を含めるため49行分を1ページとして扱う

        let tempSheet30 = templateSheet30.copyTo(ss);

        const startIdx = mainCount + (p - 1) * linesPerPage;
        const endIdx = Math.min(startIdx + linesPerPage, flattenedLines.length);
        for (let idx = startIdx; idx < endIdx; idx++) {
          let lineData = flattenedLines[idx];
          let rowNum = CO30_ITEM_ROWS[idx - startIdx];
          setCell_(tempSheet30, `${CO30_COL_TOOLNO}${rowNum}`, lineData.code);

        for (let r = 1; r <= 31; r++) {
          printMainSheet.setRowHeight(mainCursorRow + r - 1, templateSheet30.getRowHeight(r));

          printMainSheet.setRowHeight(mainCursorRow + 31, 20);
          printMainSheet.setRowHeight(mainCursorRow + 32, 20);
        }
        tempSheet30.getRange("A1:AN31").copyTo(printMainSheet.getRange(mainCursorRow, 1, 31, 40));
        try { ss.deleteSheet(tempSheet30); } catch (e) {}
        mainCursorRow += isLastPage ? 31 : 33;
      }
    }

    SpreadsheetApp.flush();

    // ---- メインPDFを書き出す ----
    const mainRowsCount = mainCursorRow - 1;
    const urlMain = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printMainSheet.getSheetId()}&r1=0&c1=0&r2=${mainRowsCount}&c2=39`;

      .getBlob().setName(`CO-30_用具受取書_[${fromLocation}]_${autoSignDate} ${autoSignTime}(全${mainPagesCount}頁).pdf`);
    try { ss.deleteSheet(printMainSheet); } catch (e) {}

    // ---- 別紙PDFを書き出す(ある場合のみ) ----
    let blobSuppl = null;
    if (useSuppl) {
      const supplRowsCount = supplCursorRow - 1;
      const supplColsForExport = 5; // 🆕 用具番号/説明/貸出先/備考/依頼番号の5列固定。getLastColumn()は環境によって不安定なため直接指定する
      const urlSuppl = `${ss.getUrl().replace(/edit$/, '')}export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&gid=${printSupplSheet.getSheetId()}&r1=0&c1=0&r2=${supplRowsCount}&c2=${supplColsForExport}`;

function getInspectionData(locationName, teamFilter) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("CO30_Inspection");
  if (!sheet) return [];

  const values = sheet.getDataRange().getValues();
  const items = [];

  for (let i = 1; i < values.length; i++) {
    if (locationName && values[i][0] !== locationName) continue;

    const itemTeam = values[i][7] ? String(values[i][7]).trim() : "";
    if (teamFilter && teamFilter !== "両方" && itemTeam && itemTeam !== teamFilter) continue;

    const updatedAtRaw = values[i][6];
    let updatedAt = 0;
    if (updatedAtRaw instanceof Date) {
      updatedAt = updatedAtRaw.getTime();
    } else if (updatedAtRaw) {
      const parsed = new Date(updatedAtRaw);
      updatedAt = isNaN(parsed.getTime()) ? 0 : parsed.getTime();
    }

    const remarks = values[i][14] ? String(values[i][14]).trim() : "";
    const requestNote = values[i][16] ? String(values[i][16]).trim() : ""; // 🆕 依頼メモ(CO31の特記事項)

    items.push({
      rowId: i + 1,
      location: values[i][0] ? String(values[i][0]).trim() : "",
      code: String(values[i][1]).trim(),
      name: String(values[i][2]).trim(),
      plan: parseInt(values[i][3], 10) || 0,
      actual: parseInt(values[i][4], 10) || 0,
      updatedAt: updatedAt,
      team: itemTeam,
      dept: values[i][8] ? String(values[i][8]).trim() : "",
      reqNo: values[i][9] ? String(values[i][9]).trim() : "",
      lifecycle: values[i][11] ? String(values[i][11]).trim() : "未処理",
      remarks: remarks,
      requestNote: requestNote,
      cancelled: remarks.indexOf("⚠️取消済み") !== -1 // 🆕 QRラベル・スキャナー側で警告表示するためのフラグ
    });
  }
  return items;
}

function updateInspectionQty(locationName, itemCode, newQty, rowId) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("CO30_Inspection");
  if (!sheet) return { success: false, error: "検品シートが見つかりません" };

  if (rowId && rowId >= 2) {
    const rowValues = sheet.getRange(rowId, 1, 1, 6).getValues()[0];
    const sheetCode = String(rowValues[1]).trim();
    const sheetLocation = rowValues[0];
    const locationMatches = !locationName || sheetLocation === locationName;

    if (sheetCode === itemCode && locationMatches) {
      const planQty = rowValues[3];
      const now = new Date();
      sheet.getRange(rowId, 5).setValue(newQty);
      sheet.getRange(rowId, 6).setValue(newQty >= planQty ? "完了" : "未完了");
      sheet.getRange(rowId, 7).setValue(now);
      return { success: true, updatedAt: now.getTime(), rowId: rowId };
    }
  }

  const values = sheet.getDataRange().getValues();
  for (let i = 1; i < values.length; i++) {
    const locationMatches = !locationName || values[i][0] === locationName;
    if (locationMatches && String(values[i][1]).trim() === itemCode) {
      const row = i + 1;
      const planQty = values[i][3];
      const now = new Date();
      sheet.getRange(row, 5).setValue(newQty);
      sheet.getRange(row, 6).setValue(newQty >= planQty ? "完了" : "未完了");
      sheet.getRange(row, 7).setValue(now);
      return { success: true, updatedAt: now.getTime(), rowId: row };
    }
  }
  return { success: false, error: "該当する物品コードが見つかりません" };
}

/**
 * 拠点別・備品別表示(親子チェック)からの一括更新用。
 * rowIds(非連続可)すべてに、指定フィールド(lifecycle/remarks)とG列=最終更新時刻をセットする。
 */
function updateInspectionFieldBatch(rowIds, field, value) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("CO30_Inspection");
  if (!sheet) return { success: false, error: "シートが見つかりません" };

  const fieldToCol = { lifecycle: 12, remarks: 15 };
  const col = fieldToCol[field];
  if (!col) return { success: false, error: "不明なフィールド: " + field };

  const ids = (rowIds || [])
    .map(function(r) { return parseInt(r, 10); })
    .filter(function(r) { return !isNaN(r) && r >= 2; });
  if (ids.length === 0) return { success: true, updatedRows: [] };

  const lock = LockService.getScriptLock();
  const gotLock = lock.tryLock(10 * 1000);
  if (!gotLock) return { success: false, error: "他の処理が実行中です。少し待ってからもう一度お試しください。" };
  try {
    ensureDashboardHeaders_(sheet);
    const now = new Date();
    const lastRow = sheet.getLastRow();
    ids.forEach(function(rowId) {
      if (rowId > lastRow) return;
      sheet.getRange(rowId, col).setValue(value);
      sheet.getRange(rowId, 7).setValue(now);
    });
    return { success: true, updatedRows: ids };
  } catch (err) {
    return { success: false, error: err.message };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 🆕 ダッシュボード用: CO-30がまだ作成されていない依頼(承認待ち・CO30作成待ち)を一覧で返す。
 * 用具番号はまだ存在しないため含まない(拠点ごとに一括採番されるのはCO30作成時のため)。
 * 送信完了・取消の依頼は含めない。
 */
function listPendingRequests() {
  // 🆕 親依頼ID単位で集約(件数は「行数」ではなく「依頼数」になる)。
  //    グループ内のいずれかの行が承認待ち/CO30作成待ちなら1件として含める。
  const out = [];
  collectCO31Groups_().forEach(function(g) {
    const hasWait = g.statuses.indexOf("承認待ち") !== -1;
    const hasReady = g.statuses.indexOf("CO30作成待ち") !== -1;
    if (!hasWait && !hasReady) return;
    out.push({
      dept: g.dept,
      fromLocation: g.fromLocation,
      toLocation: g.toLocation,
      status: hasWait ? "承認待ち" : "CO30作成待ち",
      reqNo: g.reqNos.join("、"),
      rows: g.rows,
      items: g.items.map(function(it) { return { name: it.name, qty: it.qty || "1" }; }),
      timestamp: g.timestamp
    });
  });
  out.sort(function(a, b) { return b.timestamp - a.timestamp; });
  return out;
}

/** 🆕 未処理の承認があるかだけを高速に判定する(AD列1列のみ読む) */
function hasPendingApprovals_() {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO31_Responses");
    if (!sheet || sheet.getLastRow() < 2) return false;
    const st = sheet.getRange(2, 30, sheet.getLastRow() - 1, 1).getValues();
    for (let i = 0; i < st.length; i++) {
      if (String(st[i][0] || "").trim() === "承認") return true;

    return false;

}

/**
 * 🆕 ステータスが「承認」のまま残っている依頼を、親依頼IDごとに処理する。
 * ・新しいグループの着手は開始から2.5分まで(残りは次回へ持ち越し)
 * ・2回目以降の再挑戦は1ページずつに分割し、確実に完了させる
 * ・3回連続で完了しない場合のみ打ち切り、運送・用具部門へ通知する
 */
function processApprovalQueue_() {
  deleteTriggersByName_(APPROVAL_WORKER_FN_);

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30 * 1000)) {
    scheduleApprovalWorker_(60 * 1000);
    return { busy: true, processed: 0 };
  }

  const started = Date.now();
  const NEW_GROUP_LIMIT_MS = 150 * 1000; // 2.5分: 最大4ページ(PDF8回)が入る余白を残す
  const SPLIT_ROW_LIMIT_MS = 240 * 1000; // 4分: 分割処理で次の1ページに着手してよい上限
  let remaining = false, processed = 0, failed = 0;

  try {
    try { cleanupCO31TempSheets_(); } catch (e) {}

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("CO31_Responses");
    if (!sheet || sheet.getLastRow() < 2) return { processed: 0 };

    const props = PropertiesService.getScriptProperties();
    const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, CO31_TOTAL_COLS).getValues();

    const groups = {};
    const order = [];
    for (let i = 0; i < values.length; i++) {
      if (String(values[i][29] || "").trim() !== "承認") continue;
      const rowNum = i + 2;
      const gid = String(values[i][CO31_COL_GROUPID_IDX] || "").trim() || ("r" + rowNum);
      if (!groups[gid]) {
        groups[gid] = { rows: [], iid: String(values[i][1] || "").trim() || ("r" + rowNum) };
        order.push(gid);
      }
      groups[gid].rows.push(rowNum);
    }
    if (order.length === 0) return { processed: 0 };

    for (let g = 0; g < order.length; g++) {
      if (Date.now() - started > NEW_GROUP_LIMIT_MS) { remaining = true; break; }

      const rows = groups[order[g]].rows;
      const iid = groups[order[g]].iid;
      const failKey = "apprfail_" + iid;
      const notifiedKey = "apprnotified_" + iid;
      const attempts = parseInt(props.getProperty(failKey) || "0", 10);

      if (attempts >= APPROVAL_MAX_ATTEMPTS_) {
        if (props.getProperty(notifiedKey) !== "1") {
          notifyApprovalFailure_(rows, `処理が${APPROVAL_MAX_ATTEMPTS_}回連続で完了しませんでした(実行時間超過の可能性があります)。`);
          props.setProperty(notifiedKey, "1");
        }
        continue;
      }

      const approverName = props.getProperty("apprby_id_" + iid) || "運送部門担当者";
      try {
        props.setProperty(failKey, String(attempts + 1)); // 先に加算(ハードタイムアウトでも記録が残る)

        if (attempts >= 1 && rows.length > 1) {
          // 🆕 2回目以降は1ページずつ処理して確実に完了させる
          for (let k = 0; k < rows.length; k++) {
            if (Date.now() - started > SPLIT_ROW_LIMIT_MS) { remaining = true; break; }
            sendGroupCO31_(sheet, [rows[k]], approverName);
          }
        } else {
          sendGroupCO31_(sheet, rows, approverName);
        }

        props.deleteProperty(failKey);
        props.deleteProperty(notifiedKey);
        props.deleteProperty("apprby_id_" + iid);
        processed++;
      } catch (err) {
        failed++;
        console.error("承認キューの処理に失敗(行" + rows.join(",") + "): " + err.message);
        if (attempts + 1 >= APPROVAL_MAX_ATTEMPTS_ && props.getProperty(notifiedKey) !== "1") {
          notifyApprovalFailure_(rows, err.message);
          props.setProperty(notifiedKey, "1");
        }
      }
    }
    return { processed: processed, failed: failed, remaining: remaining };
  } finally {
    lock.releaseLock();
    if (remaining) scheduleApprovalWorker_(10 * 1000);
  }
}

/**
 * 🆕 複数の内部依頼IDについて、CO30_Inspectionの備考に「取消済み」の印をまとめて付ける。
 * 1セルずつsetValueすると遅いため、列を丸ごと読んで一括で書き戻す。
 */
function markCO30ItemsAsCancelledBatch_(internalIds) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("CO30_Inspection");
  if (!sheet || sheet.getLastRow() < 2) return 0;

  const set = {};
  (internalIds || []).forEach(function(id) {
    const s = String(id || "").trim();
    if (s) set[s] = true;
  });
  if (Object.keys(set).length === 0) return 0;

  const n = sheet.getLastRow() - 1;
  const ids = sheet.getRange(2, 16, n, 1).getValues();      // P列=内部依頼ID
  const remarks = sheet.getRange(2, 15, n, 1).getValues();  // O列=備考

  let count = 0, changed = false;
  for (let i = 0; i < n; i++) {
    const rid = ids[i][0] ? String(ids[i][0]).trim() : "";
    if (!rid || !set[rid]) continue;
    const cur = remarks[i][0] ? String(remarks[i][0]).trim() : "";
    if (cur.indexOf("⚠️取消済み") !== -1) continue;
    remarks[i][0] = cur ? "⚠️取消済み / " + cur : "⚠️取消済み";
    count++; changed = true;
  }
  if (changed) sheet.getRange(2, 15, n, 1).setValues(remarks);
  return count;
}

function processCancelQueue_() {
  deleteTriggersByName_(CANCEL_WORKER_FN_);

  const props = PropertiesService.getScriptProperties();
  const keys = props.getKeys().filter(function(k) { return k.indexOf("cancjob_") === 0; }).sort();
  if (keys.length === 0) return { processed: 0 };

  const started = Date.now();
  const jobs = [];
  const allRows = [];
  const allIds = [];
  const allNos = [];

  for (let i = 0; i < keys.length; i++) {
    let job = null;
    try { job = JSON.parse(props.getProperty(keys[i])); } catch (e) {}
    props.deleteProperty(keys[i]);
    if (!job) continue;
    jobs.push(job);
    (job.rows || []).forEach(function(r) { allRows.push(r); });
    (job.internalIds || []).forEach(function(v) { allIds.push(v); });
    (job.reqNos || []).forEach(function(v) { if (allNos.indexOf(v) === -1) allNos.push(v); });
  }
  if (jobs.length === 0) return { processed: 0 };

  try { invalidateTokensForRows_(allRows); } catch (e) {}

  let markedTotal = 0;
  try { markedTotal = markCO30ItemsAsCancelledBatch_(allIds); } catch (e) {}

  try {
    if (allNos.length > 0) {
      const folder = DriveApp.getFolderById(CO31_PRINT_FOLDER_ID);
      const res = allNos.map(function(no) {
        const esc = String(no).replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
        return new RegExp("No" + esc + "(?!\\d)");
      });
      const iter = folder.getFiles();
      while (iter.hasNext()) {
        if (Date.now() - started > 4 * 60 * 1000) break;
        const f = iter.next();
        const name = f.getName();
        if (res.some(function(re) { return re.test(name); })) {
          try { f.setTrashed(true); } catch (e) {}
        }
      }
    }
  } catch (e) {
    console.error("取消PDFの削除に失敗: " + e.message);
  }

  jobs.forEach(function(job) {
    try {
      const truckingEmail = "2026trucking@gmail.com";
      const dept = job.dept || "不明部門";
      const deptEmail = getDeptEmail_(dept);
      const ccList = [];
      if (deptEmail && deptEmail !== truckingEmail) ccList.push(deptEmail);
      if (job.contactEmail && job.contactEmail.indexOf("@") !== -1
          && job.contactEmail !== truckingEmail && ccList.indexOf(job.contactEmail) === -1) {
        ccList.push(job.contactEmail);
      }

      const statuses = job.oldStatuses || [];
      const wasApproved = statuses.some(function(s) {
        return s === "承認" || s === "CO30作成待ち" || s === "送信完了";
      });
      const urgencyNote = wasApproved
        ? "⚠️ この依頼は既に承認済み(または送信済み)でした。運送・用具部門は準備を進めている場合、作業を中止してください。\n⚠️ 依頼部門・運送・用具部門とも、既にお手元にあるこの依頼のCO-31(控えメールの添付PDF・印刷済みの紙)は、今後の手続きには使用せず破棄してください。"
        : "この依頼はまだ承認前でした。";
      const co30Note = markedTotal > 0
        ? `\n\n※既にCO-30が作成されており、${markedTotal}件の物品に「取消済み」の印を付けました。現場で使用しないようご確認ください。`
        : "";
      const reqNo = (job.reqNos || []).join("、");

      const mailOpts = {
        to: truckingEmail,
        subject: `【取消】${dept}の配送依頼が取り消されました${reqNo ? "(No." + reqNo + ")" : ""}`,
        body: `運送・用具部門 ご担当者様\n${dept} ご担当者様\n\n以下のCO-31配送依頼が取り消されましたのでお知らせします。\n\n` +
              `依頼番号: ${reqNo || "未採番(承認前)"}\n依頼部門: ${dept}\n配送元　: ${job.fromLoc || ""}\n` +
              `配送先　: ${job.toLoc || ""}\n担当者　: ${job.contactInfo || "未記入"}\n` +
              `取消前のステータス: ${aggregateGroupStatus_(statuses) || "不明"}\n\n${urgencyNote}${co30Note}`
      };
      if (ccList.length > 0) mailOpts.cc = ccList.join(",");
      sendMail_(mailOpts);
    } catch (e) {
      console.error("取消通知メールの送信に失敗: " + e.message);
    }
  });

  return { processed: jobs.length };
}
